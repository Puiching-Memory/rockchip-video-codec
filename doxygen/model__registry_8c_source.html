38static int package_model_dir(char *out, size_t cap) {
39 Dl_info info;
40 char *slash;
41
42 if (!dladdr(&rkvc_model_registry_anchor, &info) || !info.dli_fname)
43 return -1;
44 if (strlen(info.dli_fname) >= cap)
45 return -1;
46 strcpy(out, info.dli_fname);
47 slash = strrchr(out, '/');
48 if (!slash)
49 return -1;
50 *slash = '\0';
51 if (strlen(out) + strlen("/../share/rkvc/models") + 1 > cap)
52 return -1;
53 strcat(out, "/../share/rkvc/models");
54 return 0;
55}

58static void scan_dir_recursive(const char *dir, unsigned depth, char **paths,
59 size_t *npaths) {
60 DIR *d = opendir(dir);
61 struct dirent *de;
62
63 if (!d || depth > 16)
64 return;
65 while (*npaths < RKMODEL_MAX_SCAN_FILES && (de = readdir(d)) != NULL) {
66 size_t len = strlen(de->d_name);
67 char *full;
68 struct stat st;
69 if (strcmp(de->d_name, ".") == 0 || strcmp(de->d_name, "..") == 0)
70 continue;
71 full = rkvc_g_calloc(1, strlen(dir) + len + 2);
72 if (!full)
73 break;
74 sprintf(full, "%s/%s", dir, de->d_name);
75 /* 发现可信内容时不跟随符号链接。 */
76 if (lstat(full, &st) != 0 || S_ISLNK(st.st_mode)) {
77 rkvc_g_free(full);
78 continue;
79 }
80 if (S_ISDIR(st.st_mode)) {
81 scan_dir_recursive(full, depth + 1, paths, npaths);
82 rkvc_g_free(full);
83 continue;
84 }
85 if (S_ISREG(st.st_mode) && len >= 9 &&
86 strcmp(de->d_name + len - 8, ".rkmodel") == 0)
87 paths[(*npaths)++] = full;
88 else
89 rkvc_g_free(full);
90 }
91 closedir(d);
92}

94rkvc_status rkvc_model_registry_scan(rkvc_context *ctx) {
95 char *paths[RKMODEL_MAX_SCAN_FILES];
96 size_t npaths = 0;
97 char dirs[RKVC_MAX_MODEL_DIRS][PATH_MAX];
98 size_t ndirs = 0;
99 size_t i;
100
101 if (!ctx)
102 return RKVC_STATUS_INVALID;
103 if (ctx->model_dir_override) {
104 snprintf(dirs[ndirs++], PATH_MAX, "%s", ctx->model_dir_override);
105 } else {
106 if (package_model_dir(dirs[0], PATH_MAX) == 0)
107 ndirs = 1;
108 snprintf(dirs[ndirs++], PATH_MAX, "/usr/local/share/rkvc/models");
109 snprintf(dirs[ndirs++], PATH_MAX, "/usr/share/rkvc/models");
110 for (i = 0; i < ctx->paths.model_dir_count &&
111 ndirs < RKVC_MAX_MODEL_DIRS; ++i)
112 snprintf(dirs[ndirs++], PATH_MAX, "%s", ctx->paths.model_dirs[i]);
113 }
114
115 for (i = 0; i < ndirs; ++i)
116 scan_dir_recursive(dirs[i], 0, paths, &npaths);
117 qsort(paths, npaths, sizeof(paths[0]), path_cmp);
118
119 ctx->models = rkvc_g_calloc(npaths ? npaths : 1, sizeof(*ctx->models));
120 if (!ctx->models) {
121 for (i = 0; i < npaths; ++i)
122 rkvc_g_free(paths[i]);
123 return RKVC_STATUS_NOMEM;
124 }
125 for (i = 0; i < npaths; ++i) {
126 char err[160] = {0};
127 rkvc_rkmodel *m = &ctx->models[ctx->model_count];
128 if (rkvc_rkmodel_open(paths[i], m, err, sizeof(err)) ==
129 RKVC_STATUS_OK) {
130 if (m->min_runtime_abi > RKVC_ABI_VERSION)
131 ctx->model_skipped++;
132 else
133 ctx->model_count++;
134 } else {
135 ctx->model_skipped++;
136 }
137 rkvc_g_free(paths[i]);
138 }
139 return RKVC_STATUS_OK;
140}

152const rkvc_rkmodel *rkvc_model_registry_select(const rkvc_context *ctx,
153 const rkvc_request *req,
154 rkvc_diag **diag) {
155 const rkvc_rkmodel *best = NULL;
156 const char *role;
157 int best_score = -1;
158 size_t i;
159 if (!ctx || !req)
160 return NULL;
161 role = role_for_operation(req->operation);
162 for (i = 0; i < ctx->model_count; ++i) {
163 const rkvc_rkmodel *m = &ctx->models[i];
164 int score = 0;
165 if (req->model_id && strcmp(req->model_id, m->info.id) != 0)
166 continue;
167 /* 显式 model_id 也必须过 role 过滤：同 id 的 encoder/decoder 模型
168 * 并存时，跳过 role 会按扫描序选中错误角色的模型。 */
169 if (role && strcmp(role, m->info.role) != 0)
170 continue;
171 /* rknn_target 是 SoC 族名（如 rk3576），而 caps.soc 来自设备树
172 * compatible，可能带板级后缀（rk3576-evb1-v10）：按前缀匹配。 */
173 if (m->info.rknn_target[0] && ctx->caps.soc[0] &&
174 strncmp(m->info.rknn_target, ctx->caps.soc,
175 strlen(m->info.rknn_target)) != 0)
176 continue;
177 if (m->info.rknn_target[0] && ctx->caps.soc[0]) score += 50;
178 if (!best || score > best_score ||
179 (score == best_score && strcmp(m->info.id, best->info.id) < 0)) {
180 best = m;
181 best_score = score;
182 }
183 }
184 if (!best && diag)
185 rkvc_diag_push(diag, RKVC_STATUS_NOT_FOUND, 2,
186 "model-registry", "no compatible model candidate");
187 return best;
188}
