66rkvc_err rkvc_session_reconfigure(rkvc_session *session,
67 const rkvc_reconfig_desc *desc)
68{
69 if (!session || !desc)
70 return RKVC_ERR_INVALID;
71 if (desc->flags == 0)
72 return RKVC_OK;
73
74 if ((desc->flags & RKVC_RECONFIG_BITRATE) && desc->bitrate <= 0)
75 return RKVC_ERR_INVALID;
76 if ((desc->flags & RKVC_RECONFIG_GOP) && desc->gop_size < 1)
77 return RKVC_ERR_INVALID;
78 if (session_reconfig_unsupported(session))
79 return RKVC_ERR_INVALID;
80
81 pthread_mutex_lock(&session->lock);
82 if (desc->flags & RKVC_RECONFIG_BITRATE)
83 session->desc.bitrate = desc->bitrate;
84 if (desc->flags & RKVC_RECONFIG_GOP)
85 session->desc.gop_size = desc->gop_size;
86 session->reconfig_pending |= desc->flags;
87 pthread_mutex_unlock(&session->lock);
88 return RKVC_OK;
89}

97rkvc_err rkvc_session_apply_reconfig(rkvc_session *s, int *force_idr)
98{
99 if (!s)
100 return RKVC_ERR_INVALID;
101 if (force_idr)
102 *force_idr = 0;
103
104 pthread_mutex_lock(&s->lock);
105 unsigned pending = s->reconfig_pending;
106 int64_t bitrate = s->desc.bitrate;
107 int gop = s->desc.gop_size;
108 if (pending)
109 s->reconfig_pending = 0;
110 pthread_mutex_unlock(&s->lock);
111
112 if (!pending)
113 return RKVC_OK;
114
115 if (force_idr && (pending & RKVC_RECONFIG_IDR))
116 *force_idr = 1;
117
118 if (s->enc) {
119 rkvc_err err = rkvc_mpp_enc_apply_rc(s->enc, bitrate, gop);
120 if (err != RKVC_OK) {
121 pthread_mutex_lock(&s->lock);
122 s->reconfig_pending |= pending;
123 pthread_mutex_unlock(&s->lock);
124 if (force_idr)
125 *force_idr = 0;
126 return err;
127 }
128 }
129 return RKVC_OK;
130}
