3/*
4 * lib/rans.h — rANS 熵编解码器（纯 C 实现）。
5 *
6 * 完整移植自 Microsoft msrtc_rans（mlvc/packages/msrtc_rans，C++17 模板），
7 * 覆盖全部功能：
8 * - 两种 rANS 变体：RansByte（uint32_t state / uint8_t unit）和 Rans64（uint64_t / uint32_t）
9 * - PMF 表 → 预计算编码符号表（定点倒数，消除 Put 中的除法）
10 * - PMF 表 → 解码 CDF 累积频率表（二分查找符号）
11 * - 流式编码（多 coder 写入同一流）+ Flush 产出码流
12 * - 流式解码（Open → 多次 Decode）
13 * - 一次性 Encode/Decode（缓冲版本）
14 * - Bypass 编解码（超出符号范围的离群值）
15 * - 可增长堆缓冲（HeapResizableBuffer 等价）
16 * - EOF / 码流完整性校验
17 *
18 * 算法来源：ryg_rans / msrtc_rans（MIT License, Copyright (c) Microsoft Corporation）。
19 */

115/* ── 流式编码器 ──────────────────────────────────────────────────── */
116/*
117 * 编码顺序与 C++ 一致：数据逆序写入（从末尾到开头），
118 * Flush 时写入 state 字节，最终码流为 [ptr, end) 字节范围。
119 *
120 * 典型用法：
121 * rkvc_rans_enc_stream s;
122 * rkvc_rans_enc_stream_init(&s, RKVC_RANS_BYTE);
123 * rkvc_rans_enc_stream_encode(&s, &gaussian, s1, y1, n);
124 * rkvc_rans_enc_stream_encode(&s, &gaussian, s0, y0, n);
125 * rkvc_rans_enc_stream_encode(&s, &bitest, zi, z, n);
126 * size_t sz; const uint8_t *out = rkvc_rans_enc_stream_flush(&s, &sz);
127 */
128
129typedef struct {
130 rkvc_rans_variant variant;
131 uint64_t state; /* rANS 状态（RansByte 仅用低 32 位）*/
132 uint8_t *buf; /* 动态增长缓冲起始 */
133 uint8_t *ptr; /* 当前写入位置（递减方向）*/
134 uint8_t *end; /* 缓冲结束 */
135 size_t cap; /* 当前容量（字节）*/
136 int flushed;
137 int oom; /* enc_grow 失败后禁止再写 */
138} rkvc_rans_enc_stream;
