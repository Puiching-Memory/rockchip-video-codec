24static int header_compatible(const rkvc_header *h, size_t minimum) {
25 uint32_t caller_major;
26 if (!h || h->struct_size < minimum)
27 return 0;
28 if (h->api_version == 0)
29 return 1;
30 caller_major = h->api_version >> 16;
31 return caller_major == RKVC_ABI_VERSION_MAJOR;
32}

47static rkvc_status copy_dir_list(const char **src, size_t count,
48 char ***owned_out, const char ***view_out) {
49 char **owned;
50 size_t i;
51 if (!count) {
52 *owned_out = NULL;
53 *view_out = NULL;
54 return RKVC_STATUS_OK;
55 }
56 if (!src)
57 return RKVC_STATUS_INVALID;
58 owned = rkvc_g_calloc(count, sizeof(*owned));
59 if (!owned)
60 return RKVC_STATUS_NOMEM;
61 for (i = 0; i < count; ++i) {
62 size_t len;
63 if (!src[i] || !src[i][0])
64 goto invalid;
65 len = strlen(src[i]);
66 owned[i] = rkvc_g_calloc(len + 1, 1);
67 if (!owned[i])
68 goto nomem;
69 memcpy(owned[i], src[i], len + 1);
70 }
71 *owned_out = owned;
72 *view_out = (const char **)owned;
73 return RKVC_STATUS_OK;
74invalid:
75 for (i = 0; i < count; ++i) rkvc_g_free(owned[i]);
76 rkvc_g_free(owned);
77 return RKVC_STATUS_INVALID;
78nomem:
79 for (i = 0; i < count; ++i) rkvc_g_free(owned[i]);
80 rkvc_g_free(owned);
81 return RKVC_STATUS_NOMEM;
82}

85static void probe_soc(char *out, size_t out_size) {
86 FILE *fp;
87 const char *path = "/proc/device-tree/compatible";
88 (void)out_size;
89 out[0] = '\0';
90 fp = fopen(path, "rb");
91 if (!fp)
92 return;
93 /* compatible 以 NUL 分隔多个字符串；取第一个 rockchip,<soc> */
94 {
95 char buf[1024];
96 size_t n = fread(buf, 1, sizeof(buf) - 1, fp);
97 size_t i;
98 fclose(fp);
99 buf[n] = '\0';
100 for (i = 0; i + 1 < n; ++i) {
101 if (buf[i] == '\0')
102 continue;
103 if (strncmp(&buf[i], "rockchip,", 9) == 0) {
104 size_t j = i + 9;
105 while (j < n && buf[j] != '\0' &&
106 (j - (i + 9)) < out_size - 1) {
107 out[j - (i + 9)] = buf[j];
108 ++j;
109 }
110 out[j - (i + 9)] = '\0';
111 break;
112 }
113 i += strlen(&buf[i]);
114 }
115 }
116}

128rkvc_status rkvc_context_create(const rkvc_context_options *opts,
129 rkvc_context **out) {
130 rkvc_context *c;
131 size_t opt_size = opts ? opts->header.struct_size : 0;
132 rkvc_status st;
133 if (!out)
134 return RKVC_STATUS_INVALID;
135 *out = NULL;
136 if (opts && !header_compatible(&opts->header,
137 offsetof(rkvc_context_options, thread_model) +
138 sizeof(opts->thread_model)))
139 return RKVC_STATUS_INVALID;
140 c = rkvc_g_calloc(1, sizeof(*c));
141 if (!c)
142 return RKVC_STATUS_NOMEM;
143 atomic_init(&c->refcount, 1);
144 c->header.struct_size = sizeof(*c);
145 c->header.api_version = RKVC_ABI_VERSION;
146 c->thread_model = opts ? opts->thread_model : RKVC_THREAD_MODEL_DEFAULT;
147 c->inspect_timeout_ms = 2000;
148 if (opts && FIELD_AVAILABLE(rkvc_context_options, opt_size, paths)) {
149 if (opts->paths.backend_dir_count > RKVC_MAX_BACKEND_DIRS ||
150 opts->paths.model_dir_count > RKVC_MAX_MODEL_DIRS) {
151 rkvc_g_free(c);
152 return RKVC_STATUS_INVALID;
153 }
154 st = copy_dir_list(opts->paths.backend_dirs,
155 opts->paths.backend_dir_count,
156 &c->owned_backend_dirs,
157 &c->paths.backend_dirs);
158 if (st != RKVC_STATUS_OK) { rkvc_g_free(c); return st; }
159 c->paths.backend_dir_count = opts->paths.backend_dir_count;
160 st = copy_dir_list(opts->paths.model_dirs,
161 opts->paths.model_dir_count,
162 &c->owned_model_dirs,
163 &c->paths.model_dirs);
164 if (st != RKVC_STATUS_OK) { free_paths(c); rkvc_g_free(c); return st; }
165 c->paths.model_dir_count = opts->paths.model_dir_count;
166 }
167 if (opts && FIELD_AVAILABLE(rkvc_context_options, opt_size,
168 model_dir_override) &&
169 opts->model_dir_override && opts->model_dir_override[0]) {
170 size_t len = strlen(opts->model_dir_override);
171 c->model_dir_override = rkvc_g_calloc(len + 1, 1);
172 if (!c->model_dir_override) { free_paths(c); rkvc_g_free(c); return RKVC_STATUS_NOMEM; }
173 memcpy(c->model_dir_override, opts->model_dir_override, len + 1);
174 }
175 if (opts && FIELD_AVAILABLE(rkvc_context_options, opt_size,
176 inspect_timeout_ms) && opts->inspect_timeout_ms)
177 c->inspect_timeout_ms = opts->inspect_timeout_ms;
178 if (opts && FIELD_AVAILABLE(rkvc_context_options, opt_size, log_level))
179 c->log_level = opts->log_level;
180 probe_device(&c->caps);
181 (void)rkvc_model_registry_scan(c); /* 候选失败只淘汰，不失败创建 */
182 (void)rkvc_backend_dso_scan(c);
183 *out = c;
184 return RKVC_STATUS_OK;
185}

231static rkvc_status validate_backend_factories(const rkvc_backend *be) {
232 const rkvc_node_factory *factories;
233 size_t count = 0, i;
234
235 if (!be->factories)
236 return RKVC_STATUS_OK;
237 factories = be->factories(be->probe_ctx, &count);
238 if (count > RKVC_MAX_FACTORIES_PER_BACKEND || (count && !factories))
239 return RKVC_STATUS_FORMAT;
240 for (i = 0; i < count; ++i) {
241 const rkvc_node_factory *f = &factories[i];
242 if (!f->id || !f->id[0] || !f->backend_id ||
243 strcmp(f->backend_id, be->id) != 0 ||
244 f->stage < RKVC_NODE_STAGE_SOURCE ||
245 f->stage > RKVC_NODE_STAGE_SINK ||
246 !f->matches || !f->create)
247 return RKVC_STATUS_FORMAT;
248 }
249 return RKVC_STATUS_OK;
250}

252rkvc_status rkvc_registry_add_backend(rkvc_context *ctx,
253 const rkvc_backend *be) {
254 rkvc_diag *diag = NULL;
255 rkvc_status st;
256 size_t i;
257 if (!ctx || !be || !be->id || !be->id[0])
258 return RKVC_STATUS_INVALID;
259 if (ctx->backend_count >= RKVC_MAX_BACKENDS)
260 return RKVC_STATUS_NOMEM;
261 /* 校验 ABI 握手 */
262 if (be->abi_version != RKVC_ABI_VERSION)
263 return RKVC_STATUS_FORMAT;
264 for (i = 0; i < ctx->backend_count; ++i)
265 if (strcmp(ctx->backends[i]->id, be->id) == 0)
266 return RKVC_STATUS_FORMAT;
267 if (be->probe && be->probe(&ctx->caps, be->probe_ctx, &diag) != 0) {
268 rkvc_diag_release(diag);
269 return RKVC_STATUS_HW;
270 }
271 rkvc_diag_release(diag);
272 st = validate_backend_factories(be);
273 if (st != RKVC_STATUS_OK)
274 return st;
275 ctx->backends[ctx->backend_count++] = be;
276 if (be->capability_flags & RKVC_BACKEND_CAP_MPP_DECODE)
277 ctx->caps.has_mpp_decoder = 1;
278 if (be->capability_flags & RKVC_BACKEND_CAP_MPP_ENCODE)
279 ctx->caps.has_mpp_encoder = 1;
280 if (be->capability_flags & RKVC_BACKEND_CAP_RGA)
281 ctx->caps.has_rga = 1;
282 if (be->capability_flags & RKVC_BACKEND_CAP_RKNN) {
283 ctx->caps.has_rknn = 1;
284 if (!ctx->caps.npu_cores)
285 ctx->caps.npu_cores = 1;
286 }
287 return RKVC_STATUS_OK;
288}

316struct ranked_factory {
317 const rkvc_node_factory *factory;

319};

322static int ranked_factory_cmp(const void *ap, const void *bp) {
323 const struct ranked_factory *a = ap;
324 const struct ranked_factory *b = bp;
325 int rc;
326 if (a->score != b->score)
327 return a->score > b->score ? -1 : 1;
328 rc = strcmp(a->factory->backend_id, b->factory->backend_id);
329 if (rc != 0)
330 return rc;
331 return strcmp(a->factory->id, b->factory->id);
332}

334int rkvc_plan_build(const rkvc_context *ctx, const rkvc_request *req,
335 const rkvc_device_caps *caps, rkvc_plan *plan,
336 rkvc_diag **diag) {
337 rkvc_node_stage required[5];
338 size_t required_count = 0, i;
339
340 if (!ctx || !req || !plan)
341 return (int)RKVC_STATUS_INVALID;
342
343 /* 文件端点（带 uri）由内建 source/sink 节点承载；流式端点走 push/pull。
344 * job 层已校验 FILE 端点必须携带 uri；此处对空 uri 宽容以便规划器
345 * 可独立单测。 */
346 if (req->input.kind == RKVC_ENDPOINT_FILE && req->input.uri)
347 required[required_count++] = RKVC_NODE_STAGE_SOURCE;
348
349 switch (req->operation) {
350 case RKVC_OPERATION_ENCODE:
351 required[required_count++] = RKVC_NODE_STAGE_ENCODE;
352 break;
353 case RKVC_OPERATION_DECODE:
354 required[required_count++] = RKVC_NODE_STAGE_DECODE;
355 break;
356 case RKVC_OPERATION_UPSCALE:
357 required[required_count++] = RKVC_NODE_STAGE_TRANSFORM;
358 break;
359 case RKVC_OPERATION_TRANSCODE:
360 required[required_count++] = RKVC_NODE_STAGE_DECODE;
361 /* transform 节点可选：仅当请求指定了输出尺寸，或 model_id 显式
362 * 强制匹配的 transform 工厂时才纳入计划。 */
363 if (req->width || req->height || req->model_id)
364 required[required_count++] = RKVC_NODE_STAGE_TRANSFORM;
365 required[required_count++] = RKVC_NODE_STAGE_ENCODE;
366 break;
367 default:
368 return (int)RKVC_STATUS_INVALID;
369 }
370
371 if (req->output.kind == RKVC_ENDPOINT_FILE && req->output.uri)
372 required[required_count++] = RKVC_NODE_STAGE_SINK;
373
374 plan->steps = rkvc_g_calloc(required_count, sizeof(*plan->steps));
375 if (!plan->steps)
376 return (int)RKVC_STATUS_NOMEM;
377 plan->step_count = 0;
378
379 for (i = 0; i < required_count; ++i) {
380 struct ranked_factory *ranked;
381 size_t ranked_count = 0, capacity = 0, b;
382
383 for (b = 0; b < ctx->backend_count; ++b) {
384 size_t count = 0;
385 const rkvc_backend *be = ctx->backends[b];
386 if (be->factories)
387 (void)be->factories(be->probe_ctx, &count);
388 if (count > RKVC_MAX_FACTORIES_PER_BACKEND ||
389 capacity > SIZE_MAX - count) {
390 rkvc_plan_release(plan);
391 return (int)RKVC_STATUS_FORMAT;
392 }
393 capacity += count;
394 }
395 ranked = rkvc_g_calloc(capacity ? capacity : 1, sizeof(*ranked));
396 if (!ranked) {
397 rkvc_plan_release(plan);
398 return (int)RKVC_STATUS_NOMEM;
399 }
400 for (b = 0; b < ctx->backend_count; ++b) {
401 size_t count = 0, f;
402 const rkvc_backend *be = ctx->backends[b];
403 const rkvc_node_factory *fs = be->factories
404 ? be->factories(be->probe_ctx, &count) : NULL;
405 if (count > RKVC_MAX_FACTORIES_PER_BACKEND || (count && !fs)) {
406 rkvc_g_free(ranked);
407 rkvc_plan_release(plan);
408 return (int)RKVC_STATUS_FORMAT;
409 }
410 for (f = 0; f < count; ++f) {
411 int score, bonus = 0;
412 int64_t total;
413 if (fs[f].stage != required[i] || !fs[f].matches ||
414 !fs[f].matches(req->operation, req->codec, caps))
415 continue;
416 score = fs[f].priority;
417 if (fs[f].score)
418 bonus = fs[f].score(req, caps, fs[f].create_ctx);
419 total = (int64_t)score + bonus;
420 score = total > INT_MAX ? INT_MAX :
421 total < INT_MIN ? INT_MIN : (int)total;
422 if (ranked_count == capacity) {
423 rkvc_g_free(ranked);
424 rkvc_plan_release(plan);
425 return (int)RKVC_STATUS_FORMAT;
426 }
427 ranked[ranked_count].factory = &fs[f];
428 ranked[ranked_count].score = score;
429 ranked_count++;
430 }
431 }
432 if (!ranked_count) {
433 rkvc_g_free(ranked);
434 if (diag)
435 rkvc_diag_push(diag, RKVC_STATUS_NOT_FOUND, 2,
436 "planner", "required stage has no candidate");
437 rkvc_plan_release(plan);
438 return (int)RKVC_STATUS_NOT_FOUND;
439 }
440 qsort(ranked, ranked_count, sizeof(*ranked), ranked_factory_cmp);
441 plan->steps[plan->step_count].candidates =
442 rkvc_g_calloc(ranked_count, sizeof(*plan->steps[0].candidates));
443 if (!plan->steps[plan->step_count].candidates) {
444 rkvc_g_free(ranked);
445 rkvc_plan_release(plan);
446 return (int)RKVC_STATUS_NOMEM;
447 }
448 for (b = 0; b < ranked_count; ++b)
449 plan->steps[plan->step_count].candidates[b] = ranked[b].factory;
450 plan->steps[plan->step_count].candidate_count = ranked_count;
451 plan->steps[plan->step_count].candidate_index = 0;
452 plan->steps[plan->step_count].factory = ranked[0].factory;
453 plan->steps[plan->step_count].request = *req;
454 plan->step_count++;
455 rkvc_g_free(ranked);
456 }
457 return 0;
458}

460int rkvc_plan_advance(rkvc_plan *plan, size_t failure_step) {
461 size_t i, j;
462 if (!plan || !plan->step_count)
463 return 0;
464 if (plan->fallback_count >= RKVC_MAX_PLAN_FALLBACKS)
465 return 0;
466
467 /* 首选只替换实际失败的阶段，保留其他已选候选。 */
468 if (failure_step < plan->step_count) {
469 rkvc_plan_step *step = &plan->steps[failure_step];
470 if (step->candidate_index + 1 < step->candidate_count) {
471 step->candidate_index++;
472 step->factory = step->candidates[step->candidate_index];
473 for (j = failure_step + 1; j < plan->step_count; ++j) {
474 plan->steps[j].candidate_index = 0;
475 plan->steps[j].factory = plan->steps[j].candidates[0];
476 }
477 plan->fallback_count++;
478 return 1;
479 }
480 }
481
482 /* 失败阶段已耗尽时按稳定的逆序进位寻找下一条组合。 */
483 for (i = plan->step_count; i > 0; --i) {
484 rkvc_plan_step *step = &plan->steps[i - 1];
485 if (i - 1 == failure_step)
486 continue;
487 if (step->candidate_index + 1 < step->candidate_count) {
488 step->candidate_index++;
489 step->factory = step->candidates[step->candidate_index];
490 for (j = i; j < plan->step_count; ++j) {
491 plan->steps[j].candidate_index = 0;
492 plan->steps[j].factory = plan->steps[j].candidates[0];
493 }
494 plan->fallback_count++;
495 return 1;
496 }
497 }
498 return 0;
499}
