34 * void *，并且仍能稳定定位承载核心实现的 ELF 映像。 */

38static int package_backend_dir(char *out, size_t cap) {
39 Dl_info info;
40 char *slash;
41
42 if (!dladdr(&rkvc_backend_dso_anchor, &info) || !info.dli_fname)
43 return -1;
44 if (strlen(info.dli_fname) >= cap)
45 return -1;
46 strcpy(out, info.dli_fname);
47 slash = strrchr(out, '/');
48 if (!slash)
49 return -1;
50 *slash = '\0';
51 if (strlen(out) + strlen("/rkvc/backends") + 1 > cap)
52 return -1;
53 strcat(out, "/rkvc/backends");
54 return 0;
55}

58static void collect_candidates(const char *dir, char **paths, size_t *npaths) {
59 DIR *d = opendir(dir);
60 struct dirent *de;
61
62 if (!d)
63 return;
64 while (*npaths < RKVC_MAX_BACKENDS && (de = readdir(d)) != NULL) {
65 size_t len = strlen(de->d_name);
66 char *full;
67 if (len < 4 || strcmp(de->d_name + len - 3, ".so") != 0)
68 continue;
69 full = rkvc_g_calloc(1, strlen(dir) + len + 2);
70 if (!full)
71 break;
72 sprintf(full, "%s/%s", dir, de->d_name);
73 paths[(*npaths)++] = full;
74 }
75 closedir(d);
76}

79static void try_load(rkvc_context *ctx, const char *path) {
80 void *handle;
81 void *symbol;
82 rkvc_backend_query_fn query;
83 const rkvc_backend *be;
84 rkvc_status st;
85
86 /* RTLD_LOCAL：后端符号不污染全局命名空间；RTLD_NOW：立即解析，失败即
87 * 淘汰（避免运行期惰性绑定的不可控失败点）。 */
88 handle = dlopen(path, RTLD_NOW | RTLD_LOCAL);
89 if (!handle) {
90 snprintf(ctx->backend_diag, sizeof(ctx->backend_diag),
91 "%s: dlopen: %s", path, dlerror());
92 ctx->backend_skipped++;
93 return;
94 }
95 symbol = dlsym(handle, RKVC_BACKEND_QUERY_SYMBOL);
96 if (!symbol) {
97 snprintf(ctx->backend_diag, sizeof(ctx->backend_diag),
98 "%s: missing %s()", path, RKVC_BACKEND_QUERY_SYMBOL);
99 ctx->backend_skipped++;
100 dlclose(handle);
101 return;
102 }
103 /* POSIX 规定 dlsym() 可返回函数地址，但 ISO C 不允许对象/函数指针
104 * 直接强转。目标平台二者同宽；memcpy 保留位模式且通过 -Wpedantic。 */
105 _Static_assert(sizeof(query) == sizeof(symbol),
106 "dlsym function pointer size mismatch");
107 memcpy(&query, &symbol, sizeof(query));
108 be = query();
109 st = rkvc_registry_add_backend(ctx, be); /* 含 ABI 握手校验 */
110 if (st != RKVC_STATUS_OK) {
111 snprintf(ctx->backend_diag, sizeof(ctx->backend_diag),
112 "%s: rejected: %s", path, rkvc_status_str(st));
113 ctx->backend_skipped++;
114 dlclose(handle);
115 return;
116 }
117 if (ctx->dso_count < RKVC_MAX_BACKENDS)
118 ctx->dso_handles[ctx->dso_count++] = handle;
119}

121rkvc_status rkvc_backend_dso_scan(rkvc_context *ctx) {
122 char *paths[RKVC_MAX_BACKENDS];
123 size_t npaths = 0;
124 char dirs[RKVC_MAX_BACKEND_DIRS][PATH_MAX];
125 size_t ndirs = 0;
126 size_t i;
127
128 if (!ctx)
129 return RKVC_STATUS_INVALID;
130
131 /* 内建后端先行（确定性顺序的第一段） */
132 rkvc_backend_register_builtins(ctx);
133
134 if (package_backend_dir(dirs[0], PATH_MAX) == 0)
135 ndirs = 1;
136 snprintf(dirs[ndirs++], PATH_MAX, "/usr/local/lib/rkvc/backends");
137 snprintf(dirs[ndirs++], PATH_MAX, "/usr/lib/rkvc/backends");
138 for (i = 0; i < ctx->paths.backend_dir_count &&
139 ndirs < RKVC_MAX_BACKEND_DIRS; ++i)
140 snprintf(dirs[ndirs++], PATH_MAX, "%s", ctx->paths.backend_dirs[i]);
141
142 for (i = 0; i < ndirs; ++i)
143 collect_candidates(dirs[i], paths, &npaths);
144 qsort(paths, npaths, sizeof(paths[0]), path_cmp);
145
146 for (i = 0; i < npaths; ++i) {
147 try_load(ctx, paths[i]);
148 rkvc_g_free(paths[i]);
149 }
150 return RKVC_STATUS_OK;
151}
