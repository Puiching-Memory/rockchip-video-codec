71static rkvc_err mock_fill_nv12(rkvc_buffer *frame, int w, int h, int64_t pts)
72{
73 uint8_t *planes[4] = {0};
74 int strides[4] = {0};
75 rkvc_err err = rkvc_buffer_get_video_planes(frame, planes, strides);
76 if (err != RKVC_OK)
77 return err;
78
79 /* 竖直灰阶条 + 随 pts 水平滚动，便于肉眼/哈希区分帧 */
80 int shift = (int)(pts % (int64_t)(w > 0 ? w : 1));
81 for (int y = 0; y < h; y++) {
82 uint8_t *row = planes[0] + y * strides[0];
83 for (int x = 0; x < w; x++)
84 row[x] = (uint8_t)(((x + shift) * 255) / (w > 1 ? w - 1 : 1));
85 }
86 for (int y = 0; y < h / 2; y++) {
87 uint8_t *row = planes[1] + y * strides[1];
88 for (int x = 0; x < w; x += 2) {
89 row[x] = 128;
90 if (x + 1 < w)
91 row[x + 1] = 128;
92 }
93 }
94 return RKVC_OK;
95}

97static rkvc_err mock_read_frame(rkvc_v4l2_cap *c, rkvc_buffer **out)
98{
99 rkvc_buffer *frame = NULL;
100 rkvc_err err =
101 rkvc_buffer_alloc_video_host(&frame, c->width, c->height,
102 RKVC_PIX_FMT_NV12);
103 if (err != RKVC_OK)
104 return err;
105
106 err = mock_fill_nv12(frame, c->width, c->height, c->next_pts);
107 if (err != RKVC_OK) {
108 rkvc_buffer_unref(frame);
109 return err;
110 }
111 rkvc_buffer_set_pts(frame, c->next_pts++);
112 *out = frame;
113 return RKVC_OK;
114}

116rkvc_err rkvc_v4l2_open(rkvc_v4l2_cap **out, const rkvc_v4l2_config *cfg)
117{
118 if (!out || !cfg || !cfg->device || !cfg->device[0])
119 return RKVC_ERR_INVALID;
120 if (cfg->width <= 0 || cfg->height <= 0)
121 return RKVC_ERR_INVALID;
122
123 *out = NULL;
124 rkvc_v4l2_cap *c = rkvc_calloc(1, sizeof(*c));
125 if (!c)
126 return RKVC_ERR_NOMEM;
127
128 snprintf(c->device, sizeof(c->device), "%s", cfg->device);
129 c->fd = -1;
130 c->width = cfg->width;
131 c->height = cfg->height;
132 c->stride = cfg->width;
133 c->fps_num = cfg->fps_num > 0 ? cfg->fps_num : 30;
134 c->fps_den = cfg->fps_den > 0 ? cfg->fps_den : 1;
135
136 if (is_mock_device(c->device)) {
137 c->mock = 1;
138 c->streaming = 1;
139 c->stride = c->width;
140 RKVC_LOG("v4l2 open mock %dx%d (synthetic NV12)", c->width, c->height);
141 *out = c;
142 return RKVC_OK;
143 }
144
145 c->fd = open(c->device, O_RDWR | O_NONBLOCK | O_CLOEXEC);
146 if (c->fd < 0) {
147 rkvc_free(c);
148 return (errno == EACCES || errno == EPERM) ? RKVC_ERR_PERMISSION
149 : RKVC_ERR_IO;
150 }
151
152 struct v4l2_capability cap;
153 memset(&cap, 0, sizeof(cap));
154 if (xioctl(c->fd, VIDIOC_QUERYCAP, &cap) < 0) {
155 rkvc_v4l2_close(c);
156 return RKVC_ERR_IO;
157 }
158
159 int mplane = (cap.capabilities & V4L2_CAP_VIDEO_CAPTURE_MPLANE) != 0;
160 int single = (cap.capabilities & V4L2_CAP_VIDEO_CAPTURE) != 0;
161 if (!mplane && !single) {
162 rkvc_v4l2_close(c);
163 return RKVC_ERR_FORMAT;
164 }
165 c->mplane = mplane ? 1 : 0;
166
167 enum v4l2_buf_type type =
168 mplane ? V4L2_BUF_TYPE_VIDEO_CAPTURE_MPLANE
169 : V4L2_BUF_TYPE_VIDEO_CAPTURE;
170
171 if (mplane) {
172 struct v4l2_format fmt;
173 memset(&fmt, 0, sizeof(fmt));
174 fmt.type = type;
175 fmt.fmt.pix_mp.width = (uint32_t)c->width;
176 fmt.fmt.pix_mp.height = (uint32_t)c->height;
177 fmt.fmt.pix_mp.pixelformat = V4L2_PIX_FMT_NV12;
178 fmt.fmt.pix_mp.field = V4L2_FIELD_NONE;
179 fmt.fmt.pix_mp.num_planes = 1;
180 if (xioctl(c->fd, VIDIOC_S_FMT, &fmt) < 0) {
181 rkvc_v4l2_close(c);
182 return RKVC_ERR_FORMAT;
183 }
184 c->width = (int)fmt.fmt.pix_mp.width;
185 c->height = (int)fmt.fmt.pix_mp.height;
186 c->stride = (int)fmt.fmt.pix_mp.plane_fmt[0].bytesperline;
187 if (c->stride < c->width)
188 c->stride = c->width;
189 } else {
190 struct v4l2_format fmt;
191 memset(&fmt, 0, sizeof(fmt));
192 fmt.type = type;
193 fmt.fmt.pix.width = (uint32_t)c->width;
194 fmt.fmt.pix.height = (uint32_t)c->height;
195 fmt.fmt.pix.pixelformat = V4L2_PIX_FMT_NV12;
196 fmt.fmt.pix.field = V4L2_FIELD_NONE;
197 if (xioctl(c->fd, VIDIOC_S_FMT, &fmt) < 0) {
198 rkvc_v4l2_close(c);
199 return RKVC_ERR_FORMAT;
200 }
201 c->width = (int)fmt.fmt.pix.width;
202 c->height = (int)fmt.fmt.pix.height;
203 c->stride = (int)fmt.fmt.pix.bytesperline;
204 if (c->stride < c->width)
205 c->stride = c->width;
206 }
207
208 struct v4l2_streamparm parm;
209 memset(&parm, 0, sizeof(parm));
210 parm.type = type;
211 parm.parm.capture.timeperframe.numerator = (uint32_t)c->fps_den;
212 parm.parm.capture.timeperframe.denominator = (uint32_t)c->fps_num;
213 (void)xioctl(c->fd, VIDIOC_S_PARM, &parm);
214
215 struct v4l2_requestbuffers req;
216 memset(&req, 0, sizeof(req));
217 req.count = RKVC_V4L2_NBUF;
218 req.type = type;
219 req.memory = V4L2_MEMORY_MMAP;
220 if (xioctl(c->fd, VIDIOC_REQBUFS, &req) < 0 || req.count < 1) {
221 rkvc_v4l2_close(c);
222 return RKVC_ERR_IO;
223 }
224 c->nbuf = (int)req.count;
225 if (c->nbuf > RKVC_V4L2_NBUF)
226 c->nbuf = RKVC_V4L2_NBUF;
227
228 for (int i = 0; i < c->nbuf; i++) {
229 struct v4l2_buffer buf;
230 struct v4l2_plane planes[1];
231 memset(&buf, 0, sizeof(buf));
232 memset(planes, 0, sizeof(planes));
233 buf.type = type;
234 buf.memory = V4L2_MEMORY_MMAP;
235 buf.index = (uint32_t)i;
236 if (mplane) {
237 buf.m.planes = planes;
238 buf.length = 1;
239 }
240 if (xioctl(c->fd, VIDIOC_QUERYBUF, &buf) < 0) {
241 rkvc_v4l2_close(c);
242 return RKVC_ERR_IO;
243 }
244 size_t length = mplane ? planes[0].length : buf.length;
245 off_t offset = mplane ? (off_t)planes[0].m.mem_offset
246 : (off_t)buf.m.offset;
247 c->bufs[i].length = length;
248 c->bufs[i].start =
249 mmap(NULL, length, PROT_READ | PROT_WRITE, MAP_SHARED, c->fd,
250 offset);
251 if (c->bufs[i].start == MAP_FAILED) {
252 rkvc_v4l2_close(c);
253 return RKVC_ERR_IO;
254 }
255 }
256
257 for (int i = 0; i < c->nbuf; i++) {
258 struct v4l2_buffer buf;
259 struct v4l2_plane planes[1];
260 memset(&buf, 0, sizeof(buf));
261 memset(planes, 0, sizeof(planes));
262 buf.type = type;
263 buf.memory = V4L2_MEMORY_MMAP;
264 buf.index = (uint32_t)i;
265 if (mplane) {
266 buf.m.planes = planes;
267 buf.length = 1;
268 }
269 if (xioctl(c->fd, VIDIOC_QBUF, &buf) < 0) {
270 rkvc_v4l2_close(c);
271 return RKVC_ERR_IO;
272 }
273 }
274
275 if (xioctl(c->fd, VIDIOC_STREAMON, &type) < 0) {
276 RKVC_LOG("v4l2 STREAMON failed on %s: errno=%d (%s)", c->device, errno,
277 strerror(errno));
278 rkvc_v4l2_close(c);
279 return RKVC_ERR_IO;
280 }
281 c->streaming = 1;
282
283 RKVC_LOG("v4l2 open %s %dx%d stride=%d NV12 nbuf=%d", c->device, c->width,
284 c->height, c->stride, c->nbuf);
285 *out = c;
286 return RKVC_OK;
287}

289void rkvc_v4l2_close(rkvc_v4l2_cap *c)
290{
291 if (!c)
292 return;
293 if (c->mock) {
294 rkvc_free(c);
295 return;
296 }
297 if (c->fd >= 0 && c->streaming) {
298 enum v4l2_buf_type type = c->mplane ? V4L2_BUF_TYPE_VIDEO_CAPTURE_MPLANE
299 : V4L2_BUF_TYPE_VIDEO_CAPTURE;
300 (void)xioctl(c->fd, VIDIOC_STREAMOFF, &type);
301 c->streaming = 0;
302 }
303 v4l2_unmap(c);
304 if (c->fd >= 0)
305 close(c->fd);
306 rkvc_free(c);
307}

317rkvc_err rkvc_v4l2_read_frame(rkvc_v4l2_cap *c, rkvc_buffer **out,
318 int timeout_ms)
319{
320 if (!c || !out)
321 return RKVC_ERR_INVALID;
322 *out = NULL;
323
324 if (c->mock)
325 return mock_read_frame(c, out);
326
327 if (c->fd < 0)
328 return RKVC_ERR_INVALID;
329
330 struct pollfd pfd = { .fd = c->fd, .events = POLLIN };
331 int pr = poll(&pfd, 1, timeout_ms);
332 if (pr == 0)
333 return RKVC_ERR_AGAIN;
334 if (pr < 0)
335 return RKVC_ERR_IO;
336
337 enum v4l2_buf_type type =
338 c->mplane ? V4L2_BUF_TYPE_VIDEO_CAPTURE_MPLANE
339 : V4L2_BUF_TYPE_VIDEO_CAPTURE;
340
341 struct v4l2_buffer buf;
342 struct v4l2_plane planes[1];
343 memset(&buf, 0, sizeof(buf));
344 memset(planes, 0, sizeof(planes));
345 buf.type = type;
346 buf.memory = V4L2_MEMORY_MMAP;
347 if (c->mplane) {
348 buf.m.planes = planes;
349 buf.length = 1;
350 }
351 if (xioctl(c->fd, VIDIOC_DQBUF, &buf) < 0) {
352 if (errno == EAGAIN)
353 return RKVC_ERR_AGAIN;
354 return RKVC_ERR_IO;
355 }
356
357 if (buf.index >= (uint32_t)c->nbuf) {
358 (void)xioctl(c->fd, VIDIOC_QBUF, &buf);
359 return RKVC_ERR_INTERNAL;
360 }
361
362 const int w = c->width;
363 const int h = c->height;
364 const int src_stride = c->stride > 0 ? c->stride : w;
365 size_t bytesused = c->mplane ? planes[0].bytesused : buf.bytesused;
366 /* 至少需要 stride*h + stride*(h/2) 的映射长度 */
367 size_t need_map = (size_t)src_stride * (size_t)h +
368 (size_t)src_stride * ((size_t)h / 2);
369 if (c->bufs[buf.index].length < need_map && bytesused < need_map) {
370 (void)xioctl(c->fd, VIDIOC_QBUF, &buf);
371 return RKVC_ERR_FORMAT;
372 }
373
374 rkvc_buffer *frame = NULL;
375 rkvc_err err =
376 rkvc_buffer_alloc_video_host(&frame, c->width, c->height,
377 RKVC_PIX_FMT_NV12);
378 if (err != RKVC_OK) {
379 (void)xioctl(c->fd, VIDIOC_QBUF, &buf);
380 return err;
381 }
382
383 uint8_t *planes_out[4] = {0};
384 int strides[4] = {0};
385 err = rkvc_buffer_get_video_planes(frame, planes_out, strides);
386 if (err != RKVC_OK) {
387 rkvc_buffer_unref(frame);
388 (void)xioctl(c->fd, VIDIOC_QBUF, &buf);
389 return err;
390 }
391
392 const uint8_t *src = (const uint8_t *)c->bufs[buf.index].start;
393 const uint8_t *uv = src + (size_t)src_stride * (size_t)h;
394
395 if (src_stride == w && strides[0] == w && strides[1] == w) {
396 memcpy(planes_out[0], src, (size_t)w * (size_t)h);
397 memcpy(planes_out[1], uv, (size_t)w * (size_t)h / 2);
398 } else {
399 for (int y = 0; y < h; y++)
400 memcpy(planes_out[0] + y * strides[0],
401 src + (size_t)y * (size_t)src_stride, (size_t)w);
402 for (int y = 0; y < h / 2; y++)
403 memcpy(planes_out[1] + y * strides[1],
404 uv + (size_t)y * (size_t)src_stride, (size_t)w);
405 }
406
407 rkvc_buffer_set_pts(frame, c->next_pts++);
408 if (xioctl(c->fd, VIDIOC_QBUF, &buf) < 0) {
409 rkvc_buffer_unref(frame);
410 return RKVC_ERR_IO;
411 }
412
413 *out = frame;
414 return RKVC_OK;
415}
