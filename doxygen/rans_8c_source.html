22/* RansByte: uint32_t state, uint8_t unit */
23#define RANS_BYTE_STATE_BITS 31
24#define RANS_BYTE_MAX_SCALE 30 /* min(STATE_BITS-1, 32) = min(30,32) */
25#define RANS_BYTE_LOWER_BOUND (1u << (RANS_BYTE_STATE_BITS - 8)) /* 1<<23 */
26#define RANS_BYTE_UNIT_SIZE 1 /* sizeof(uint8_t) */
27#define RANS_BYTE_UNITS_PER_STATE 4 /* sizeof(uint32_t)/sizeof(uint8_t) */
28
29/* Rans64: uint64_t state, uint32_t unit */
30#define RANS_64_STATE_BITS 63
31#define RANS_64_MAX_SCALE 32 /* min(STATE_BITS-1, 32) = min(62,32) */
32#define RANS_64_LOWER_BOUND (1ULL << (RANS_64_STATE_BITS - 32)) /* 1<<31 */
33#define RANS_64_UNIT_SIZE 4 /* sizeof(uint32_t) */
34#define RANS_64_UNITS_PER_STATE 2 /* sizeof(uint64_t)/sizeof(uint32_t) */

36/* ── 64-bit 乘法高位（Rans64 Quotient 用）────────────────────────── */
37
38static inline uint64_t mul64hi(uint64_t a, uint64_t b)
39{
40#if defined(__SIZEOF_INT128__)
41 __uint128_t prod = (__uint128_t)a * (__uint128_t)b;
42 return (uint64_t)(prod >> 64);
43#else
44 /* Fallback：拆分乘法（理论可用，但 aarch64 有 __uint128__）*/
45 uint64_t a_lo = (uint32_t)a, a_hi = a >> 32;
46 uint64_t b_lo = (uint32_t)b, b_hi = b >> 32;
47 uint64_t lo = a_lo * b_lo;
48 uint64_t mid1 = a_hi * b_lo;
49 uint64_t mid2 = a_lo * b_hi;
50 uint64_t hi = a_hi * b_hi;
51 uint64_t mid = mid1 + mid2 + (lo >> 32);
52 return hi + (mid >> 32);
53#endif
54}

60static void enc_sym_init_byte(rkvc_rans_enc_sym *s,
61 uint32_t start, uint32_t freq, uint32_t scale_bits)
62{
63 uint32_t scale = 1u << scale_bits;
64
65 s->x_max_hi = freq << (RANS_BYTE_STATE_BITS - scale_bits);
66
67 if (freq > 1) {
68 /* Alverson 倒数除法 */
69 uint32_t shift = 1;
70 while (freq > (1u << shift))
71 shift++;
72 /* bits = sizeof(uint32_t)*8 = 32 */
73 uint64_t nom = (1ULL << (shift + 32 - 1)) + (freq - 1);
74 s->freq_rcp = (uint32_t)(nom / freq);
75 s->freq_rcp_shift = shift - 1 + 32; /* 额外 +32（非 uint64_t state）*/
76 s->bias = start;
77 } else {
78 /* freq == 1：rcp = ~0, shift = 0 */
79 s->freq_rcp = 0xFFFFFFFFu;
80 s->freq_rcp_shift = 0 + 32;
81 s->bias = start + scale - 1;
82 }
83 s->freq_cmpl = scale - freq;
84}

86static void enc_sym_init_64(rkvc_rans_enc_sym *s,
87 uint32_t start, uint32_t freq, uint32_t scale_bits)
88{
89 uint32_t scale = 1u << scale_bits;
90
91 /* min(STATE_BITS, 31) = min(63, 31) = 31 */
92 s->x_max_hi = freq << (31 - scale_bits);
93
94 if (freq > 1) {
95 uint32_t shift = 1;
96 while (freq > (1u << shift))
97 shift++;
98 /* uint64_t state 倒数：32 位跳过 + 恢复 */
99 uint64_t x0 = (uint64_t)(freq - 1);
100 uint64_t x1 = 1ULL << (shift + 31);
101 uint64_t t1 = x1 / freq;
102 x0 += (x1 % freq) << 32;
103 uint64_t t0 = x0 / freq;
104 uint64_t rcp = t0 + (t1 << 32);
105 s->freq_rcp = (uint32_t)rcp;
106 s->freq_rcp_hi = (uint32_t)(rcp >> 32);
107 s->freq_rcp_shift = shift - 1; /* 无额外移位（uint64_t state）*/
108 s->bias = start;
109 } else {
110 s->freq_rcp = 0xFFFFFFFFu;
111 s->freq_rcp_hi = 0xFFFFFFFFu;
112 s->freq_rcp_shift = 0;
113 s->bias = start + scale - 1;
114 }
115 s->freq_cmpl = scale - freq;
116}

122static int init_dist_descs(rkvc_rans_dist *dists, size_t num_dists,
123 const int32_t *pmf_lengths,
124 const int32_t *pmf_offsets,
125 size_t table_size)
126{
127 size_t cursor = 0;
128 for (size_t i = 0; i < num_dists; i++) {
129 int32_t length = pmf_lengths[i];
130 if (length <= 1 || table_size - cursor < (size_t)length)
131 return RKVC_RANS_ERR_PMF;
132 dists[i].value_offset = pmf_offsets[i];
133 dists[i].bypass_sentinel = length - 1;
134 dists[i].enc_sym_offset = cursor;
135 cursor += (size_t)length;
136 }
137 if (cursor != table_size)
138 return RKVC_RANS_ERR_PMF;
139 return RKVC_RANS_OK;
140}

143/* 熵编解码器初始化（PMF → 编码符号表 + 解码 CDF 表） */
144/* ════════════════════════════════════════════════════════════════════ */

146int rkvc_rans_coder_init(rkvc_rans_coder *coder, rkvc_rans_variant variant,
147 const int32_t *pmf_lengths, size_t num_lengths,
148 const int32_t *pmf_offsets, size_t num_offsets,
149 const int32_t *pmf_table, size_t table_size,
150 int symbol_bits, int bypass_bits)
151{
152 if (!coder || !pmf_lengths || !pmf_offsets || !pmf_table)
153 return RKVC_RANS_ERR_PARAMS;
154 if (num_lengths != num_offsets)
155 return RKVC_RANS_ERR_PMF;
156
157 size_t max_scale = (variant == RKVC_RANS_BYTE)
158 ? RANS_BYTE_MAX_SCALE : RANS_64_MAX_SCALE;
159 if (symbol_bits < 2 || (size_t)symbol_bits > max_scale)
160 return RKVC_RANS_ERR_PARAMS;
161 if (bypass_bits < 2 || (size_t)bypass_bits > max_scale)
162 return RKVC_RANS_ERR_PARAMS;
163
164 memset(coder, 0, sizeof(*coder));
165 coder->variant = variant;
166 coder->symbol_bits = symbol_bits;
167 coder->bypass_bits = bypass_bits;
168 coder->bypass_max_value = (1u << bypass_bits) - 1;
169
170 /* 分布描述 */
171 coder->dists = (rkvc_rans_dist *)calloc(num_lengths, sizeof(rkvc_rans_dist));
172 if (!coder->dists)
173 return RKVC_RANS_ERR_STATE;
174 coder->num_dists = num_lengths;
175
176 int rc = init_dist_descs(coder->dists, num_lengths,
177 pmf_lengths, pmf_offsets, table_size);
178 if (rc != RKVC_RANS_OK)
179 goto fail;
180
181 /* ── 编码符号表（从 PMF 直接映射，无间隙）── */
182 coder->enc_syms = (rkvc_rans_enc_sym *)calloc(table_size, sizeof(rkvc_rans_enc_sym));
183 if (!coder->enc_syms)
184 goto fail;
185 coder->num_enc_syms = table_size;
186
187 uint32_t max_freq = 1u << symbol_bits;
188 {
189 size_t tbl = 0;
190 for (size_t d = 0; d < num_lengths; d++) {
191 int32_t start = 0;
192 for (int32_t i = 0; i <= coder->dists[d].bypass_sentinel; i++) {
193 int32_t freq = pmf_table[tbl];
194 if (!(freq > 0 && freq <= (int32_t)(max_freq - start)))
195 goto fail;
196 if (variant == RKVC_RANS_BYTE)
197 enc_sym_init_byte(&coder->enc_syms[tbl],
198 (uint32_t)start, (uint32_t)freq, symbol_bits);
199 else
200 enc_sym_init_64(&coder->enc_syms[tbl],
201 (uint32_t)start, (uint32_t)freq, symbol_bits);
202 start += freq;
203 tbl++;
204 }
205 }
206 }
207
208 /* ── 解码 CDF 表（含每分布一个尾项，有间隙）── */
209 /* CDF 表大小 = table_size + num_lengths（每个分布多一个终止频率）*/
210 coder->cdf_table_size = table_size + num_lengths;
211 coder->cdf_table = (uint32_t *)calloc(coder->cdf_table_size, sizeof(uint32_t));
212 if (!coder->cdf_table)
213 goto fail;
214
215 {
216 size_t cursor = 0;
217 for (size_t d = 0; d < num_lengths; d++) {
218 rkvc_rans_dist *desc = &coder->dists[d];
219 /* 更新解码 CDF 偏移（含累计间隙）*/
220 desc->dec_cdf_offset = cursor + d;
221 int32_t start = 0;
222 for (int32_t i = 0; i <= desc->bypass_sentinel; i++, cursor++) {
223 int32_t freq = pmf_table[cursor];
224 if (!(freq > 0 && freq <= (int32_t)(max_freq - start)))
225 goto fail;
226 coder->cdf_table[cursor + d] = (uint32_t)start;
227 start += freq;
228 }
229 /* 尾项 = 总频率（= 1 << symbol_bits）*/
230 coder->cdf_table[cursor + d] = (uint32_t)start;
231 }
232 }
233
234 coder->initialized = 1;
235 return RKVC_RANS_OK;
236
237fail:
238 rkvc_rans_coder_free(coder);
239 return rc != RKVC_RANS_OK ? rc : RKVC_RANS_ERR_PMF;
240}

259/* ── 向后写入缓冲：确保有空间写入一个 unit ── */
260
261static int enc_grow(rkvc_rans_enc_stream *s)
262{
263 if (s->oom || !s->buf)
264 return -1;
265 size_t new_cap = s->cap * 2;
266 if (new_cap < s->cap + 65536)
267 new_cap = s->cap + 65536;
268 if (new_cap <= s->cap) {
269 s->oom = 1;
270 return -1;
271 }
272 uint8_t *new_buf = (uint8_t *)malloc(new_cap);
273 if (!new_buf) {
274 s->oom = 1;
275 return -1;
276 }
277 size_t content = (size_t)(s->end - s->ptr);
278 memcpy(new_buf + new_cap - content, s->ptr, content);
279 free(s->buf);
280 s->buf = new_buf;
281 s->cap = new_cap;
282 s->end = new_buf + new_cap;
283 s->ptr = new_buf + new_cap - content;
284 return 0;
285}

287static inline void enc_write_unit_byte(rkvc_rans_enc_stream *s, uint8_t val)
288{
289 if (s->oom || !s->buf || !s->ptr)
290 return;
291 if (s->ptr <= s->buf && enc_grow(s) != 0)
292 return;
293 if (s->ptr <= s->buf)
294 return;
295 *(--s->ptr) = val;
296}

298static inline void enc_write_unit_64(rkvc_rans_enc_stream *s, uint32_t val)
299{
300 if (s->oom || !s->buf || !s->ptr)
301 return;
302 if ((size_t)(s->ptr - s->buf) < 4 && enc_grow(s) != 0)
303 return;
304 if ((size_t)(s->ptr - s->buf) < 4)
305 return;
306 s->ptr -= 4;
307 memcpy(s->ptr, &val, 4); /* 原生字节序（aarch64 = LE）*/
308}

312static inline void renorm_byte(rkvc_rans_enc_stream *s, uint32_t x_max)
313{
314 uint32_t x = (uint32_t)s->state;
315 while (x >= x_max) {
316 enc_write_unit_byte(s, (uint8_t)x);
317 x >>= 8;
318 }
319 s->state = x;
320}
321
322static inline void renorm_64(rkvc_rans_enc_stream *s, uint64_t x_max)
323{
324 uint64_t x = s->state;
325 /* MaxScaleBits(32) <= 32 → 最多一次 */
326 while (x >= x_max) {
327 enc_write_unit_64(s, (uint32_t)x);
328 x >>= 32;
329 break;
330 }
331 s->state = x;
332}

334/* ── Put（预计算符号）── */
335
336static inline void put_sym_byte(rkvc_rans_enc_stream *s, const rkvc_rans_enc_sym *sym)
337{
338 /* x_max = m_x_max_hi（StateBits=31 不大于 31，无额外移位）*/
339 renorm_byte(s, sym->x_max_hi);
340 uint32_t x = (uint32_t)s->state;
341 uint32_t q = (uint32_t)(((uint64_t)x * sym->freq_rcp) >> sym->freq_rcp_shift);
342 x += q * sym->freq_cmpl + sym->bias;
343 s->state = x;
344}
345
346static inline void put_sym_64(rkvc_rans_enc_stream *s, const rkvc_rans_enc_sym *sym)
347{
348 /* x_max = x_max_hi << 32 = freq << (63 - ScaleBits)：重归一后 x < x_max
349 * 保证 x' = (x/freq)*M + start < 2^63（StateBits=63 不变式）。
350 * 原移植误用 << 33（StateBits=64 约定）导致状态可越过 2^63
351 * 回绕，长码流解码提前耗尽字节（由新增 test_rans 回归暴露）。 */
352 uint64_t x_max = ((uint64_t)sym->x_max_hi) << 32;
353 renorm_64(s, x_max);
354 uint64_t x = s->state;
355 uint64_t rcp = ((uint64_t)sym->freq_rcp_hi << 32) | sym->freq_rcp;
356 uint64_t q = mul64hi(x, rcp) >> sym->freq_rcp_shift;
357 x += q * sym->freq_cmpl + sym->bias;
358 s->state = x;
359}

363static inline void put_raw_byte(rkvc_rans_enc_stream *s,
364 uint32_t start, uint32_t freq, uint32_t scale_bits)

424{
425 uint8_t u;
426 if (dec_read_byte(s, &u))
427 return -1;

467static inline int dec_advance_byte(rkvc_rans_dec_stream *s,
468 uint32_t start, uint32_t freq, uint32_t scale_bits)
469{
470 uint32_t scale = 1u << scale_bits;
471 uint32_t x = (uint32_t)s->state;
472 uint32_t value = x & (scale - 1);
473 x = freq * (x >> scale_bits) + value - start;
474 while (x < RANS_BYTE_LOWER_BOUND) {
475 uint8_t u;
476 if (dec_read_byte(s, &u))
477 return -1;
478 x = (x << 8) + u;
479 }
480 s->state = x;
481 return 0;
482}

515static void enc_bypass(rkvc_rans_enc_stream *s, const rkvc_rans_coder *c,
516 uint32_t bypass_value)
517{
518 /* s_MaxBypassParts = sizeof(freq_t)*8/2 = 16 */
519 uint32_t buf[16];
520 int n = 0;
521 uint32_t v = bypass_value;
522 while (v != 0) {
523 buf[n++] = v & c->bypass_max_value;
524 v >>= c->bypass_bits;
525 }
526 /* 保留总部分数（n 会被逆序写入循环消耗）*/
527 int total_parts = n;
528 /* 逆序写入部分 */
529 while (n > 0)
530 put_raw_byte_or_64(s, c->variant, buf[--n], 1, c->bypass_bits);
531
532 /* 前缀计数 */
533 int prefix = 0;
534 uint32_t count = (uint32_t)total_parts;
535 while (count >= c->bypass_max_value) {
536 count -= c->bypass_max_value;
537 prefix++;
538 }
539 put_raw_byte_or_64(s, c->variant, count, 1, c->bypass_bits);
540 while (prefix > 0) {
541 put_raw_byte_or_64(s, c->variant, c->bypass_max_value, 1, c->bypass_bits);
542 prefix--;
543 }
544}

558/* ── bypass 解码 ── */
559
560static int dec_bypass(rkvc_rans_dec_stream *s, const rkvc_rans_coder *c,
561 uint32_t *out_value)
562{
563 uint32_t value;
564 uint32_t bypass_count;
565
566 /* Step 1: 读取 bypass count */
567 value = (c->variant == RKVC_RANS_BYTE)
568 ? dec_get_byte(s, c->bypass_bits) : dec_get_64(s, c->bypass_bits);
569 if (dec_advance_byte_or_64(s, c->variant, value, 1, c->bypass_bits))
570 return -1;
571 bypass_count = value;
572 while (value == c->bypass_max_value) {
573 value = (c->variant == RKVC_RANS_BYTE)
574 ? dec_get_byte(s, c->bypass_bits) : dec_get_64(s, c->bypass_bits);
575 if (dec_advance_byte_or_64(s, c->variant, value, 1, c->bypass_bits))
576 return -1;
577 bypass_count += value;
578 if (bypass_count > 32) /* sizeof(freq_t)*CHAR_BIT */
579 return -1;
580 }
581
582 /* Step 2: 读取 bypass value */
583 uint32_t encoded = 0;
584 uint32_t total_bits = bypass_count * c->bypass_bits;
585 for (uint32_t shift = 0; shift < total_bits; shift += c->bypass_bits) {
586 value = (c->variant == RKVC_RANS_BYTE)
587 ? dec_get_byte(s, c->bypass_bits) : dec_get_64(s, c->bypass_bits);
588 if (dec_advance_byte_or_64(s, c->variant, value, 1, c->bypass_bits))
589 return -1;
590 encoded |= value << shift;
591 }
592 *out_value = encoded;
593 return 0;
594}
595
596static inline int dec_advance_byte_or_64(rkvc_rans_dec_stream *s,
597 rkvc_rans_variant variant,
598 uint32_t start, uint32_t freq, uint32_t sb)
599{
600 if (variant == RKVC_RANS_BYTE)
601 return dec_advance_byte(s, start, freq, sb);
602 else
603 return dec_advance_64(s, start, freq, sb);
604}

610void rkvc_rans_enc_stream_init(rkvc_rans_enc_stream *s,
611 rkvc_rans_variant variant,
612 size_t initial_capacity)
613{
614 memset(s, 0, sizeof(*s));
615 s->variant = variant;
616 if (initial_capacity < 512)
617 initial_capacity = 512;
618 s->buf = (uint8_t *)malloc(initial_capacity);
619 if (!s->buf) {
620 s->oom = 1;
621 s->cap = 0;
622 s->end = NULL;
623 s->ptr = NULL;
624 s->state = (variant == RKVC_RANS_BYTE)
625 ? RANS_BYTE_LOWER_BOUND : RANS_64_LOWER_BOUND;
626 return;
627 }
628 s->cap = initial_capacity;
629 s->end = s->buf + initial_capacity;
630 s->ptr = s->end; /* 从末尾向前写 */
631 s->state = (variant == RKVC_RANS_BYTE)
632 ? RANS_BYTE_LOWER_BOUND : RANS_64_LOWER_BOUND;

635/* 注：曾尝试按变体（RANS_BYTE vs RANS_64）分裂热循环以消除逐符号分派，
636 * RK3576 交替 A/B 实测（200 万符号×4 轮）无收益且解码回退约 8%
637 * （分支预测命中、双份循环体增大 i-cache 压力），故保留单循环结构。 */
638
639int rkvc_rans_enc_stream_encode(rkvc_rans_enc_stream *s,
640 const rkvc_rans_coder *coder,
641 const int32_t *indices,
642 const int32_t *values, size_t count)
643{
644 if (!s || !coder || !indices || !values)
645 return RKVC_RANS_ERR_PARAMS;
646 if (!coder->initialized)
647 return RKVC_RANS_ERR_STATE;
648 if (s->flushed || s->oom || !s->buf)
649 return RKVC_RANS_ERR_STATE;
650
651 /* 数据逆序编码（与 C++ 一致）*/
652 for (size_t i = count; i > 0; i--) {
653 int32_t index = indices[i - 1];
654 if (index < 0)
655 continue; /* 跳过：解码时返回 0 */
656
657 /* clamp 到有效范围 */
658 if ((size_t)index >= coder->num_dists)
659 index = (int32_t)coder->num_dists - 1;
660 const rkvc_rans_dist *desc = &coder->dists[index];
661
662 int32_t value = values[i - 1] + desc->value_offset;
663 if (value < 0 || value >= desc->bypass_sentinel) {
664 /* 超出符号范围 → bypass 编码 */
665 uint32_t bv;
666 if (value < 0)
667 bv = 2u * (uint32_t)(-value) - 1;
668 else
669 bv = 2u * (uint32_t)(value - desc->bypass_sentinel);
670 enc_bypass(s, coder, bv);
671 value = desc->bypass_sentinel;
672 }
673
674 size_t sym = desc->enc_sym_offset + (size_t)value;
675 if (s->variant == RKVC_RANS_BYTE)
676 put_sym_byte(s, &coder->enc_syms[sym]);
677 else
678 put_sym_64(s, &coder->enc_syms[sym]);
679 if (s->oom)
680 return RKVC_RANS_ERR_STATE;
681 }
682 return RKVC_RANS_OK;
683}

685const uint8_t *rkvc_rans_enc_stream_flush(rkvc_rans_enc_stream *s,
686 size_t *out_size)
687{
688 if (!s || s->flushed || s->oom || !s->buf) {
689 if (out_size) *out_size = 0;
690 return NULL;
691 }
692 if (s->variant == RKVC_RANS_BYTE)
693 flush_byte(s);
694 else
695 flush_64(s);
696 if (s->oom) {
697 if (out_size) *out_size = 0;
698 return NULL;
699 }
700
701 s->flushed = 1;
702 if (out_size)
703 *out_size = (size_t)(s->end - s->ptr);
704 return s->ptr;
705}

758int rkvc_rans_dec_stream_decode(rkvc_rans_dec_stream *s,
759 const rkvc_rans_coder *coder,
760 int32_t *values,
761 const int32_t *indices, size_t count)
762{
763 if (!s || !coder || !values || !indices)
764 return RKVC_RANS_ERR_PARAMS;
765 if (!coder->initialized || !s->opened)
766 return RKVC_RANS_ERR_STATE;
767
768 for (size_t i = 0; i < count; i++) {
769 int32_t index = indices[i];
770 if (index < 0) {
771 values[i] = 0;
772 continue;
773 }
774 if ((size_t)index >= coder->num_dists)
775 index = (int32_t)coder->num_dists - 1;
776 const rkvc_rans_dist *desc = &coder->dists[index];
777
778 uint32_t cum_freq = (s->variant == RKVC_RANS_BYTE)
779 ? dec_get_byte(s, coder->symbol_bits)
780 : dec_get_64(s, coder->symbol_bits);
781
782 /* CDF 二分查找 */
783 const uint32_t *base = coder->cdf_table + desc->dec_cdf_offset;
784 int32_t lo = 0;
785 int32_t hi = desc->bypass_sentinel + 1; /* upper_bound 范围 */
786 while (lo < hi) {
787 int32_t mid = lo + (hi - lo) / 2;
788 if (base[mid + 1] <= cum_freq)
789 lo = mid + 1;
790 else
791 hi = mid;
792 }
793 int32_t symbol = lo;
794 uint32_t start = base[symbol];
795 uint32_t freq = base[symbol + 1] - base[symbol];
796
797 if (dec_advance_byte_or_64(s, s->variant, start, freq, coder->symbol_bits))
798 return RKVC_RANS_ERR_STREAM;
799
800 if (symbol == desc->bypass_sentinel) {
801 uint32_t bv;
802 if (dec_bypass(s, coder, &bv))
803 return RKVC_RANS_ERR_STREAM;
804 if (bv & 1)
805 symbol = -(int32_t)(bv >> 1) - 1;
806 else
807 symbol = (int32_t)(bv >> 1) + desc->bypass_sentinel;
808 }
809 values[i] = symbol - desc->value_offset;
810 }
811 return RKVC_RANS_OK;
812}
