23{

44void rkvc_port_queue_destroy(rkvc_port_queue *q)
45{
46 if (!q)
47 return;
48
49 pthread_mutex_lock(&q->lock);
50 while (av_fifo_can_read(q->fifo) > 0) {

53 rkvc_buffer_unref(b);

57 pthread_mutex_destroy(&q->lock);
58 pthread_cond_destroy(&q->not_full);

64rkvc_err rkvc_port_queue_push(rkvc_port_queue *q, rkvc_buffer *buf)
65{
66 rkvc_buffer *ref;
67
68 if (!q || !buf)
69 return RKVC_ERR_INVALID;
70
71 pthread_mutex_lock(&q->lock);
72 if (av_fifo_can_write(q->fifo) == 0) {
73 pthread_mutex_unlock(&q->lock);
74 return RKVC_ERR_AGAIN;
75 }
76
77 ref = rkvc_buffer_ref(buf);
78 if (!ref) {
79 pthread_mutex_unlock(&q->lock);
80 return RKVC_ERR_NOMEM;
81 }
82
83 if (av_fifo_write(q->fifo, &ref, 1) < 0) {
84 rkvc_buffer_unref(ref);
85 pthread_mutex_unlock(&q->lock);
86 return RKVC_ERR_INTERNAL;
87 }
88
89 pthread_cond_signal(&q->not_empty);
90 pthread_mutex_unlock(&q->lock);
91 return RKVC_OK;
92}

105rkvc_err rkvc_port_queue_pull(rkvc_port_queue *q, rkvc_buffer **buf,
106 int timeout_ms)
107{
108 if (!q || !buf)
109 return RKVC_ERR_INVALID;
110
111 *buf = NULL;
112 pthread_mutex_lock(&q->lock);
113
114 while (av_fifo_can_read(q->fifo) == 0) {
115 if (timeout_ms == 0) {
116 pthread_mutex_unlock(&q->lock);
117 return RKVC_ERR_AGAIN;
118 }
119
120 if (timeout_ms < 0) {
121 pthread_cond_wait(&q->not_empty, &q->lock);
122 continue;
123 }
124
125 struct timespec deadline;
126 clock_gettime(CLOCK_REALTIME, &deadline);
127 timespec_add_ms(&deadline, timeout_ms);
128 int rc = pthread_cond_timedwait(&q->not_empty, &q->lock, &deadline);
129 if (rc == ETIMEDOUT) {
130 pthread_mutex_unlock(&q->lock);
131 return RKVC_ERR_AGAIN;
132 }
133 }
134
135 if (av_fifo_read(q->fifo, buf, 1) < 0) {
136 pthread_mutex_unlock(&q->lock);
137 return RKVC_ERR_INTERNAL;
138 }
139
140 pthread_cond_signal(&q->not_full);
141 pthread_mutex_unlock(&q->lock);
142 return RKVC_OK;
143}
