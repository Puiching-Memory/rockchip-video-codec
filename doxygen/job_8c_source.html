62static int graph_build_with_fallback(const rkvc_context *ctx,
63 rkvc_plan *plan, rkvc_diag **diag,
64 rkvc_graph **out) {
65 int rc;
66 if (!plan || !out)
67 return (int)RKVC_STATUS_INVALID;
68 *out = NULL;
69 for (;;) {
70 size_t failed_step;
71 rkvc_graph *graph = rkvc_graph_new();
72 if (!graph)
73 return (int)RKVC_STATUS_NOMEM;
74 rkvc_graph_set_context(graph, ctx); /* bind_model 节点需注册表 */
75 rc = rkvc_graph_build(graph, plan, diag);
76 if (rc == 0) {
77 graph->plan = *plan;
78 memset(plan, 0, sizeof(*plan));
79 *out = graph;
80 return 0;
81 }
82 failed_step = graph->failure_step;
83 rkvc_graph_free(graph);
84 if ((rc != (int)RKVC_STATUS_NEGOTIATE &&
85 rc != (int)RKVC_STATUS_INTERNAL) ||
86 !rkvc_plan_advance(plan, failed_step))
87 return rc;
88 if (diag)
89 rkvc_diag_push(diag, (rkvc_status)rc, 2, "planner",
90 "candidate rejected; trying fallback");
91 }
92}

95static int job_open_with_fallback(struct rkvc_job *job, rkvc_diag **diag) {
96 for (;;) {
97 size_t failed_step;
98 int rc = rkvc_graph_open(job->graph, diag);
99 rkvc_plan plan;
100 if (rc == 0)
101 return 0;
102 if (rc != (int)RKVC_STATUS_HW)
103 return rc;
104
105 failed_step = job->graph->failure_step;
106 plan = job->graph->plan;
107 memset(&job->graph->plan, 0, sizeof(job->graph->plan));
108 rkvc_graph_free(job->graph);
109 job->graph = NULL;
110 if (!rkvc_plan_advance(&plan, failed_step)) {
111 rkvc_plan_release(&plan);
112 return rc;
113 }
114 if (diag)
115 rkvc_diag_push(diag, RKVC_STATUS_HW, 2, "planner",
116 "open failed; trying fallback candidate");
117 rc = graph_build_with_fallback(job->ctx, &plan, diag, &job->graph);
118 if (rc != 0) {
119 rkvc_plan_release(&plan);
120 return rc;
121 }
122 }

126static void *job_thread(void *argp) {
127 struct rkvc_job *j = argp;
128 j->run_status = rkvc_exec_run(j->exec, &j->diag);
129 pthread_mutex_lock(&j->mutex);
130 j->finished = 1;
131 pthread_cond_broadcast(&j->finished_cond);
132 pthread_mutex_unlock(&j->mutex);
133 return NULL;
134}

136rkvc_status rkvc_job_create(const rkvc_context *ctx,
137 const rkvc_request *req,
138 rkvc_diag **diag,
139 rkvc_job **out) {
140 struct rkvc_job *j;
141 rkvc_plan plan = {0};
142 int rc;
143
144 if (!ctx || !req || !out)
145 return RKVC_STATUS_INVALID;
146 *out = NULL;
147 if (req->header.struct_size <
148 offsetof(rkvc_request, output) + sizeof(req->output) ||
149 req->header.struct_size > (1u << 20) ||
150 (req->header.api_version &&
151 (req->header.api_version >> 16) != RKVC_ABI_VERSION_MAJOR))
152 return RKVC_STATUS_INVALID;
153 /* 文件端点必须携带 uri（规划器据此注入内建 source/sink 节点）。 */
154 if ((req->input.kind == RKVC_ENDPOINT_FILE && !req->input.uri) ||
155 (req->output.kind == RKVC_ENDPOINT_FILE && !req->output.uri))
156 return RKVC_STATUS_INVALID;
157
158 j = rkvc_g_calloc(1, sizeof(*j));
159 if (!j)
160 return RKVC_STATUS_NOMEM;
161 j->ctx = ctx;
162 memcpy(&j->req, req, req->header.struct_size < sizeof(j->req)
163 ? req->header.struct_size : sizeof(j->req));
164 if (j->req.input.uri) {
165 j->input_uri = job_strdup(j->req.input.uri);
166 if (!j->input_uri) goto nomem;
167 j->req.input.uri = j->input_uri;
168 }
169 if (j->req.output.uri) {
170 j->output_uri = job_strdup(j->req.output.uri);
171 if (!j->output_uri) goto nomem;
172 j->req.output.uri = j->output_uri;
173 }
174 if (req->header.struct_size >=
175 offsetof(rkvc_request, model_id) + sizeof(req->model_id) &&
176 j->req.model_id) {
177 j->model_id = job_strdup(j->req.model_id);
178 if (!j->model_id) goto nomem;
179 j->req.model_id = j->model_id;
180 if (!rkvc_model_registry_select(ctx, &j->req, diag)) {
181 job_free_strings(j);
182 rkvc_g_free(j);
183 return RKVC_STATUS_NOT_FOUND;
184 }
185 }
186 pthread_mutex_init(&j->mutex, NULL);
187 pthread_cond_init(&j->finished_cond, NULL);
188 j->state = 0;
189
190 rc = rkvc_plan_build(ctx, &j->req, &ctx->caps, &plan, diag);
191 if (rc != 0) {
192 pthread_cond_destroy(&j->finished_cond);
193 pthread_mutex_destroy(&j->mutex);
194 job_free_strings(j);
195 rkvc_g_free(j);
196 return (rkvc_status)rc;
197 }
198
199 rc = graph_build_with_fallback(ctx, &plan, diag, &j->graph);
200 if (rc != 0) {
201 rkvc_plan_release(&plan);
202 pthread_cond_destroy(&j->finished_cond);
203 pthread_mutex_destroy(&j->mutex);
204 job_free_strings(j);
205 rkvc_g_free(j);
206 return (rkvc_status)rc;
207 }
208 /* 规划已由 graph_build_with_fallback 移交图持有。 */
209 rkvc_context_retain(ctx);
210
211 *out = j;
212 return RKVC_STATUS_OK;
213nomem:
214 job_free_strings(j);
215 rkvc_g_free(j);
216 return RKVC_STATUS_NOMEM;
217}

219rkvc_status rkvc_job_start(rkvc_job *job, rkvc_diag **diag) {
220 if (!job || !job->graph)
221 return RKVC_STATUS_INVALID;
222 pthread_mutex_lock(&job->mutex);
223 if (job->state != 0) {
224 pthread_mutex_unlock(&job->mutex);
225 return RKVC_STATUS_INVALID;
226 }
227
228 {
229 int rc = job_open_with_fallback(job, diag);
230 if (rc != 0) {
231 job->state = 2;
232 job->run_status = rc;
233 job->joined = 1; /* 无执行线程；wait 直接返回启动失败状态 */
234 pthread_mutex_unlock(&job->mutex);
235 return (rkvc_status)rc;
236 }
237 }
238
239 job->exec = rkvc_exec_create(job->graph, job->graph->node_count);
240 if (!job->exec) {
241 if (diag)
242 rkvc_diag_push(diag, RKVC_STATUS_NOMEM, 3, "job", "exec alloc");
243 pthread_mutex_unlock(&job->mutex);
244 return RKVC_STATUS_NOMEM;
245 }
246 job->graph->exec = job->exec;
247
248 if (pthread_create(&job->thread, NULL, job_thread, job) != 0) {
249 if (diag)
250 rkvc_diag_push(diag, RKVC_STATUS_INTERNAL, 3, "job",
251 "thread create failed");
252 rkvc_exec_destroy(job->exec);
253 job->exec = NULL;
254 job->graph->exec = NULL;
255 pthread_mutex_unlock(&job->mutex);
256 return RKVC_STATUS_INTERNAL;
257 }
258 job->thread_started = 1;
259 job->state = 1;
260
261 /* 文件端点：源节点不消费外部输入，直接 EOS 触发 flush 产出 */
262 if (job->req.input.kind == RKVC_ENDPOINT_FILE)
263 rkvc_exec_eos(job->exec);
264
265 pthread_mutex_unlock(&job->mutex);
266 return RKVC_STATUS_OK;
267}

269rkvc_status rkvc_job_wait(rkvc_job *job) {
270 pthread_t thread;
271 rkvc_status result;
272 if (!job)
273 return RKVC_STATUS_INVALID;
274 pthread_mutex_lock(&job->mutex);
275 if (job->state == 0) {
276 pthread_mutex_unlock(&job->mutex);
277 return RKVC_STATUS_INVALID;
278 }
279 while (job->joining && !job->joined)
280 pthread_cond_wait(&job->finished_cond, &job->mutex);
281 if (job->joined) {
282 result = (rkvc_status)job->run_status;
283 pthread_mutex_unlock(&job->mutex);
284 return result;
285 }
286 job->joining = 1;
287 thread = job->thread;
288 pthread_mutex_unlock(&job->mutex);
289 pthread_join(thread, NULL);
290 pthread_mutex_lock(&job->mutex);
291 job->thread_started = 0;
292 job->joined = 1;
293 job->joining = 0;
294 job->state = 2;
295 result = (rkvc_status)job->run_status;
296 pthread_cond_broadcast(&job->finished_cond);
297 pthread_mutex_unlock(&job->mutex);
298 return result;
299}

324rkvc_status rkvc_job_push(rkvc_job *job, rkvc_frame *frame) {
325 rkvc_exec *exec;

341 pthread_mutex_unlock(&job->mutex);
