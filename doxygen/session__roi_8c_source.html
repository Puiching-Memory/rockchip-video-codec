11rkvc_err rkvc_session_set_roi(rkvc_session *session,
12 const rkvc_roi_rect *rects, int count)
13{
14 if (!session)
15 return RKVC_ERR_INVALID;
16 if (count < 0 || count > RKVC_ROI_MAX)
17 return RKVC_ERR_INVALID;
18 if (count > 0 && !rects)
19 return RKVC_ERR_INVALID;
20
21 int denom = session->desc.enc_scale_denom;
22 int ew = session->desc.width;
23 int eh = session->desc.height;
24 if (denom > 1) {
25 ew = ew > 0 ? (ew / denom) & ~1 : 0;
26 eh = eh > 0 ? (eh / denom) & ~1 : 0;
27 }
28
29 for (int i = 0; i < count; i++) {
30 if (rects[i].x < 0 || rects[i].y < 0 ||
31 rects[i].w <= 0 || rects[i].h <= 0)
32 return RKVC_ERR_INVALID;
33 /* 已知编码分辨率时拒绝越界；宽高仍为 0 则只做正矩形检查。 */
34 if (ew > 0 && eh > 0) {
35 if (rects[i].x > ew - rects[i].w ||
36 rects[i].y > eh - rects[i].h)
37 return RKVC_ERR_INVALID;
38 }
39 }
40
41 pthread_mutex_lock(&session->lock);
42 session->roi_count = count;
43 if (count > 0)
44 memcpy(session->rois, rects, (size_t)count * sizeof(rkvc_roi_rect));
45 pthread_mutex_unlock(&session->lock);
46 return RKVC_OK;
47}
