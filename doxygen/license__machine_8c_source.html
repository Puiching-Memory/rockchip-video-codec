23static int read_file_text(const char *path, char *buf, size_t size)
24{
25 FILE *f = fopen(path, "rb");
26 if (!f)
27 return -1;
28 size_t n = fread(buf, 1, size - 1, f);
29 fclose(f);
30 buf[n] = '\0';
31 /* 去尾部空白/NUL（设备树属性以 NUL 结尾） */
32 while (n > 0 && (buf[n - 1] == '\0' || isspace((unsigned char)buf[n - 1])))
33 buf[--n] = '\0';
34 return n > 0 ? (int)n : -1;
35}

37static void note_open_fail(char *note, size_t note_sz, const char *path)
38{
39 int e = errno;
40 if (e == ENOENT)
41 snprintf(note, note_sz, "missing (%s)", path);
42 else if (e == EACCES || e == EPERM)
43 snprintf(note, note_sz, "permission denied (%s)", path);
44 else
45 snprintf(note, note_sz, "open failed (%s): %s", path, strerror(e));
46}

48static int try_dt_serial(char *raw, size_t raw_sz, char *path_out, size_t path_sz,
49 char *note, size_t note_sz)
50{
51 static const char *path = "/proc/device-tree/serial-number";
52 snprintf(path_out, path_sz, "%s", path);
53 errno = 0;
54 int n = read_file_text(path, raw, raw_sz);
55 if (n > 0) {
56 snprintf(note, note_sz, "ok");
57 return 0;
58 }
59 if (errno != 0)
60 note_open_fail(note, note_sz, path);
61 else
62 snprintf(note, note_sz, "empty or unreadable (%s)", path);
63 return -1;
64}

66static int try_otp(char *raw, size_t raw_sz, char *path_out, size_t path_sz,
67 char *note, size_t note_sz)
68{
69 static const char *path = "/sys/bus/nvmem/devices/rockchip-otp0/nvmem";
70 snprintf(path_out, path_sz, "%s", path);
71 errno = 0;
72 FILE *f = fopen(path, "rb");
73 if (!f) {
74 note_open_fail(note, note_sz, path);
75 return -1;
76 }
77 uint8_t bin[32];
78 size_t n = fread(bin, 1, sizeof(bin), f);
79 int err = ferror(f);
80 fclose(f);
81 if (err || n == 0) {
82 snprintf(note, note_sz, "empty or read error (%s)", path);
83 return -1;
84 }
85 if (n * 2 >= raw_sz) {
86 snprintf(note, note_sz, "buffer too small for %zu OTP bytes", n);
87 return -1;
88 }
89 for (size_t i = 0; i < n; i++)
90 snprintf(raw + i * 2, raw_sz - i * 2, "%02x", bin[i]);
91 snprintf(note, note_sz, "ok (%zu bytes)", n);
92 return 0;
93}

95static int try_mac(char *raw, size_t raw_sz, char *path_out, size_t path_sz,
96 char *note, size_t note_sz)
97{
98 DIR *dir = opendir("/sys/class/net");
99 if (!dir) {
100 note_open_fail(note, note_sz, "/sys/class/net");
101 return -1;
102 }
103 int found = 0;
104 struct dirent *ent;
105 while (!found && (ent = readdir(dir)) != NULL) {
106 if (ent->d_name[0] == '.')
107 continue;
108 char vpath[320];
109 snprintf(vpath, sizeof(vpath), "/sys/class/net/%s/virtual", ent->d_name);
110 /* 跳过虚拟接口（lo/docker/br-…） */
111 if (access(vpath, F_OK) == 0)
112 continue;
113 char apath[320];
114 snprintf(apath, sizeof(apath), "/sys/class/net/%s/address", ent->d_name);
115 if (read_file_text(apath, raw, raw_sz) > 0 &&
116 strcmp(raw, "00:00:00:00:00:00") != 0) {
117 snprintf(path_out, path_sz, "%s", apath);
118 snprintf(note, note_sz, "ok (%.64s)", ent->d_name);
119 found = 1;
120 }
121 }
122 closedir(dir);
123 if (!found) {
124 snprintf(note, note_sz, "no usable physical NIC MAC");
125 path_out[0] = '\0';
126 return -1;
127 }
128 return 0;
129}

131/* 容器识别：标记文件或 PID 1 cgroup 特征。容器内 MAC 随实例重建而变，
132 * 且同宿主机多容器可能同 MAC，不能作为 1机1码 的稳定指纹。 */
133static int in_container(void)
134{
135 if (access("/.dockerenv", F_OK) == 0 ||
136 access("/run/.containerenv", F_OK) == 0)
137 return 1;
138 char buf[2048];
139 if (read_file_text("/proc/1/cgroup", buf, sizeof(buf)) > 0 &&
140 (strstr(buf, "docker") || strstr(buf, "kubepods") ||
141 strstr(buf, "containerd") || strstr(buf, "libpod") ||
142 strstr(buf, "lxc")))
143 return 1;
144 return 0;
145}

153static void sha256_hex(const char *input, size_t input_len, char *out_hex)
154{
155 uint8_t digest[crypto_hash_sha256_BYTES];
156 crypto_hash_sha256(digest, (const unsigned char *)input,
157 (unsigned long)input_len);
158 for (size_t i = 0; i < sizeof(digest); i++)
159 snprintf(out_hex + i * 2, 3, "%02x", digest[i]);
160}

162int lic_machine_id_collect(lic_fp_info *info)
163{
164 if (!info)
165 return -1;
166 memset(info, 0, sizeof(*info));
167
168 char raw[LIC_FP_RAW_MAX];
169 char path[LIC_FP_PATH_MAX];
170 const char *tag = NULL;
171
172 if (try_dt_serial(raw, sizeof(raw), path, sizeof(path),
173 info->note_dt, sizeof(info->note_dt)) == 0) {
174 tag = "dt-serial";
175 } else if (try_otp(raw, sizeof(raw), path, sizeof(path),
176 info->note_otp, sizeof(info->note_otp)) == 0) {
177 tag = "otp";
178 } else if (try_mac(raw, sizeof(raw), path, sizeof(path),
179 info->note_mac, sizeof(info->note_mac)) == 0) {
180 tag = "mac";
181 }
182
183 /* 未选中的层级补全 note（成功短路时尚未探测的层级） */
184 if (tag && strcmp(tag, "dt-serial") == 0) {
185 snprintf(info->note_otp, sizeof(info->note_otp),
186 "skipped (higher priority selected)");
187 snprintf(info->note_mac, sizeof(info->note_mac),
188 "skipped (higher priority selected)");
189 } else if (tag && strcmp(tag, "otp") == 0) {
190 snprintf(info->note_mac, sizeof(info->note_mac),
191 "skipped (higher priority selected)");
192 } else if (!tag) {
193 /* 三级均失败：补探测 otp/mac 若前面短路未跑到 */
194 if (info->note_otp[0] == '\0')
195 (void)try_otp(raw, sizeof(raw), path, sizeof(path),
196 info->note_otp, sizeof(info->note_otp));
197 if (info->note_mac[0] == '\0')
198 (void)try_mac(raw, sizeof(raw), path, sizeof(path),
199 info->note_mac, sizeof(info->note_mac));
200 return -1;
201 }
202
203 /* 容器内拒绝 MAC 兜底，除非显式放行 */
204 if (strcmp(tag, "mac") == 0 && in_container() && !container_mac_allowed()) {
205 snprintf(info->note_mac, sizeof(info->note_mac),
206 "rejected: MAC fallback inside container (set "
207 "RKVC_LICENSE_ALLOW_CONTAINER_MAC=1 to override)");
208 return -1;
209 }
210
211 /* 组合：tag + ':' + 原始值 → SHA-256 */
212 char concat[512];
213 int n = snprintf(concat, sizeof(concat), "%s:%s", tag, raw);
214 if (n < 0 || (size_t)n >= sizeof(concat))
215 return -1;
216
217 snprintf(info->tag, sizeof(info->tag), "%s", tag);
218 snprintf(info->path, sizeof(info->path), "%s", path);
219 snprintf(info->raw, sizeof(info->raw), "%s", raw);
220 sha256_hex(concat, (size_t)n, info->machine_id);
221 return 0;
222}

236int lic_machine_id_grouped(const char *hex64, char *out, size_t out_size)
237{
238 if (!hex64 || strlen(hex64) != 64 ||
239 !out || out_size < LIC_MACHINE_ID_GROUPED_LEN)
240 return -1;
241
242 size_t o = 0;
243 for (size_t i = 0; i < 64; i++) {
244 if (i > 0 && (i % 4) == 0)
245 out[o++] = '-';
246 out[o++] = hex64[i];
247 }
248 out[o] = '\0';
249 return 0;
250}
