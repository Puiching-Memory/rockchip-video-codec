53rkvc_err rkvc_init(void)
54{
55 pthread_mutex_lock(&s_init_mutex);
56 if (!s_initialized) {
57 rkvc_err e = rkvc_init_impl();
58 if (e != RKVC_OK) {
59 pthread_mutex_unlock(&s_init_mutex);
60 return e;
61 }
62 }
63 pthread_mutex_unlock(&s_init_mutex);
64 return RKVC_OK;
65}

110static void rkvc_dev_path(const char *path, char *buf, size_t size)
111{
112#ifdef RKVC_ENABLE_FAULT_INJECTION
113 const char *root = getenv("RKVC_TEST_DEV_ROOT");
114 if (root && root[0] != '\0') {
115 snprintf(buf, size, "%s%s", root, path);
116 return;
117 }
118#endif
119 snprintf(buf, size, "%s", path);
120}

122static int rkvc_test_dev_path_denied(const char *path)
123{
124#ifdef RKVC_ENABLE_FAULT_INJECTION
125 const char *deny = getenv("RKVC_TEST_DENY_DEV_PATH");
126 if (deny && strcmp(deny, path) == 0)
127 return 1;
128#endif
129 (void)path;
130 return 0;
131}

133static int rkvc_dev_access(const char *path, int mode)
134{
135 if (rkvc_test_dev_path_denied(path)) {
136 errno = EACCES;
137 return -1;
138 }
139
140 char resolved[PATH_MAX];
141 rkvc_dev_path(path, resolved, sizeof(resolved));
142 return access(resolved, mode);
143}

146{

172static int path_openable(const char *path, int flags)
173{
174 if (rkvc_test_dev_path_denied(path)) {
175 errno = EACCES;
176 return 0;
177 }
178
179 char resolved[PATH_MAX];
180 rkvc_dev_path(path, resolved, sizeof(resolved));
181
182 int fd = open(resolved, flags | O_CLOEXEC);
183 if (fd < 0)
184 return 0;
185
186 close(fd);
187 return 1;
188}

200int rkvc_npu_accessible(void)
201{
202 if (rkvc_dev_access("/sys/kernel/debug/rknpu/version", R_OK) == 0)
203 return 1;
204
205 DIR *dir = opendir("/dev/dri/by-path");
206 if (!dir)
207 return 0;
208
209 int found = 0;
210 struct dirent *ent;
211 while ((ent = readdir(dir)) != NULL) {
212 if (!strstr(ent->d_name, "npu-render"))
213 continue;
214 char path[PATH_MAX];
215 snprintf(path, sizeof(path), "/dev/dri/by-path/%s", ent->d_name);
216 if (rkvc_dev_access(path, R_OK | W_OK) == 0) {
217 found = 1;
218 break;
219 }
220 }
221 closedir(dir);
222 return found;
223}

248static int mpp_default_dma_heap_accessible(void)
249{
250 /*
251 * rockchip-mpp opens type 0 as /dev/dma_heap/system-uncached. If that
252 * fails, it only remaps type 0 by flipping cacheable/dma32 flags, so cma*
253 * heaps are not valid fallbacks for the default encoder/decoder path.
254 */
255 static const char *const names[] = {
256 "system-uncached",
257 "system",
258 "system-uncached-dma32",
259 "system-dma32",
260 NULL
261 };
262
263 for (int i = 0; names[i]; i++) {
264 char path[PATH_MAX];
265
266 int written = snprintf(path, sizeof(path), "/dev/dma_heap/%s",
267 names[i]);
268 if (written < 0 || (size_t)written >= sizeof(path))
269 continue;
270
271 if (path_openable(path, O_RDONLY))
272 return 1;
273 }
274
275 return 0;
276}

278rkvc_err rkvc_check_hw_permissions(void)
279{
280 if (!mpp_codec_device_accessible()) {
281 RKVC_LOG("MPP codec device permission denied: /dev/mpp_service or legacy codec nodes");
282 return RKVC_ERR_PERMISSION;
283 }
284
285 if (dma_heap_runtime_selected()) {
286 if (!mpp_default_dma_heap_accessible()) {
287 RKVC_LOG("MPP default DMA heap permission denied: /dev/dma_heap/system-uncached");
288 return RKVC_ERR_PERMISSION;
289 }
290 return RKVC_OK;
291 }
292
293 if (!drm_allocator_accessible() && !ion_allocator_accessible()) {
294 RKVC_LOG("DMA-BUF allocator permission denied: /dev/dma_heap/* or /dev/dri/*");
295 return RKVC_ERR_PERMISSION;
296 }
297
298 return RKVC_OK;
299}

301rkvc_err rkvc_query_caps(rkvc_caps *caps)
302{
303 if (!caps)
304 return RKVC_ERR_INVALID;
305
306 memset(caps, 0, sizeof(*caps));
307
308 /*
309 * 平台信息全部来自运行时探测（lib/platform.c），无任何预设板卡表。
310 * 硬件编解码能力 = 运行时探测（ffmpeg 注册 + 设备权限）∩ MPP 内核
311 * 能力位上报的 VPU 硬件支持；软件编码器（SVT-AV1）与平台无关，不做门控。
312 */
313 const rkvc_platform_info *pi = rkvc_platform_probe();
314 memcpy(caps->soc, pi->soc, sizeof(caps->soc));
315
316 rkvc_err hw_perm = rkvc_check_hw_permissions();
317 int hw_usable = (hw_perm == RKVC_OK);
318
319 caps->has_h264_enc = (avcodec_find_encoder_by_name("h264_rkmpp") != NULL
320 && hw_usable && pi->vpu_h264_enc_hw);
321 caps->has_hevc_enc = (avcodec_find_encoder_by_name("hevc_rkmpp") != NULL
322 && hw_usable && pi->vpu_hevc_enc_hw);
323 caps->has_av1_enc = (avcodec_find_encoder_by_name("libsvtav1") != NULL);
324
325 caps->has_h264_dec = (avcodec_find_decoder_by_name("h264_rkmpp") != NULL
326 && hw_usable && pi->vpu_h264_dec_hw);
327 caps->has_hevc_dec = (avcodec_find_decoder_by_name("hevc_rkmpp") != NULL
328 && hw_usable && pi->vpu_hevc_dec_hw);
329 caps->has_av1_dec = (avcodec_find_decoder_by_name("av1_rkmpp") != NULL
330 && hw_usable && pi->vpu_av1_dec_hw);
331
332 caps->has_dma_heap = mpp_default_dma_heap_accessible();
333 caps->has_rga = pi->has_rga && (rkvc_dev_access("/dev/rga", R_OK | W_OK) == 0);
334 caps->has_rknn = rkvc_rknn_sr_available() ? 1 : 0;
335
336 return RKVC_OK;
337}
