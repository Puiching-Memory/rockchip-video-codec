87void rkvc_diag_push(rkvc_diag **diag, rkvc_status status, int stage,
88 const char *subject, const char *reason) {
89 /* 字符串随节点拷贝：后端 DSO 可能在诊断消费前被 dlclose，
90 * 静态串指针会随映射卸载失效（板上实测崩溃）。 */
91 struct rkvc_diag_owned {
92 rkvc_diag node;
93 char subject_text[RKVC_DIAG_TEXT_MAX];
94 char reason_text[RKVC_DIAG_TEXT_MAX];
95 };
96 struct rkvc_diag_owned *n =
97 rkvc_g_calloc(1, sizeof(*n));
98 if (!n)
99 return; /* 内存不足：链保持原样，调用方仍可见更上层的错误 */
100 n->node.status = status;
101 n->node.stage = stage;
102 if (subject)
103 snprintf(n->subject_text, RKVC_DIAG_TEXT_MAX, "%s", subject);
104 if (reason)
105 snprintf(n->reason_text, RKVC_DIAG_TEXT_MAX, "%s", reason);
106 n->node.subject = n->subject_text;
107 n->node.reason = n->reason_text;
108 n->node.next = *diag;
109 *diag = &n->node;
110}

112void rkvc_diag_release(rkvc_diag *diag) {
113 while (diag) {
114 rkvc_diag *next = (rkvc_diag *)diag->next;
115 rkvc_g_free(diag); /* push 时按内联串结构整体分配 */
116 diag = next;
117 }
118}

120void rkvc_diag_fmt_text(const rkvc_diag *diag, char *buf, size_t size) {
121 size_t off = 0;
122 while (diag && off < size - 1) {
123 int n = snprintf(buf + off, size - off, "%s(%s): %s; ",
124 diag->subject ? diag->subject : "?",
125 diag->reason ? diag->reason : "?",
126 rkvc_status_str(diag->status));
127 if (n < 0)
128 break;
129 if ((size_t)n >= size - off)
130 break;
131 off += (size_t)n;
132 diag = diag->next;
133 }
134 buf[size - 1] = '\0';
135}

137void rkvc_diag_fmt_json(const rkvc_diag *diag, char *buf, size_t size) {
138 size_t off = 0;
139 int first = 1;
140 buf[0] = '\0';
141 off += (size_t)snprintf(buf + off, size - off, "[");
142 while (diag && off < size - 8) {
143 if (!first)
144 off += (size_t)snprintf(buf + off, size - off, ",");
145 first = 0;
146 off += (size_t)snprintf(buf + off, size - off,
147 "{\"status\":\"%s\",\"sub\":\"%s\",\"why\":\"%s\"}",
148 rkvc_status_str(diag->status),
149 diag->subject ? diag->subject : "",
150 diag->reason ? diag->reason : "");
151 diag = diag->next;
152 }
153 off += (size_t)snprintf(buf + off, size - off, "]");
154}
