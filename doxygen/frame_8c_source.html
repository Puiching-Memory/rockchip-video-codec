20void rkvc_frame_desc_init(rkvc_frame_desc *desc, size_t size) {
21 if (!desc || size < sizeof(rkvc_header))
22 return;
23 memset(desc, 0, size);
24 desc->header.struct_size = size;
25 desc->header.api_version = RKVC_ABI_VERSION;
26 if (size >= offsetof(rkvc_frame_desc, fd) + sizeof(desc->fd))
27 desc->fd = -1;
28 if (size >= offsetof(rkvc_frame_desc, pts) + sizeof(desc->pts))
29 desc->pts = RKVC_FRAME_TS_UNKNOWN;
30 if (size >= offsetof(rkvc_frame_desc, dts) + sizeof(desc->dts))
31 desc->dts = RKVC_FRAME_TS_UNKNOWN;
32}

35static rkvc_status validate_desc(const rkvc_frame_desc *desc) {
36 size_t i;
37
38 if (!desc || desc->header.struct_size <
39 offsetof(rkvc_frame_desc, fd) + sizeof(desc->fd) ||
40 desc->header.struct_size > (1u << 20) ||
41 (desc->header.api_version &&
42 (desc->header.api_version >> 16) != RKVC_ABI_VERSION_MAJOR))
43 return RKVC_STATUS_INVALID;
44 if (desc->spec.fmt == RKVC_FRAME_FMT_UNKNOWN)
45 return RKVC_STATUS_FORMAT;
46 if (desc->spec.domain == RKVC_MEM_DOMAIN_HOST) {
47 if (desc->fd >= 0)
48 return RKVC_STATUS_FORMAT;
49 } else if (desc->spec.domain == RKVC_MEM_DOMAIN_DMABUF) {
50 if (desc->fd < 0)
51 return RKVC_STATUS_FORMAT;
52 } else {
53 return RKVC_STATUS_FORMAT;
54 }
55 if (desc->header.struct_size >=
56 offsetof(rkvc_frame_desc, roi_region_count) +
57 sizeof(desc->roi_region_count)) {
58 if (desc->roi_region_count > RKVC_ROI_MAX_REGIONS ||
59 (desc->roi_region_count && !desc->roi_regions) ||
60 (desc->roi_region_count &&
61 desc->spec.fmt == RKVC_FRAME_FMT_BITSTREAM))
62 return RKVC_STATUS_INVALID;
63 for (i = 0; i < desc->roi_region_count; ++i) {
64 const rkvc_roi_region *r = &desc->roi_regions[i];
65 if (!r->width || !r->height || r->qp_delta < -51 ||
66 r->qp_delta > 51 ||
67 (desc->spec.width &&
68 (r->x >= desc->spec.width ||
69 r->width > desc->spec.width - r->x)) ||
70 (desc->spec.height &&
71 (r->y >= desc->spec.height ||
72 r->height > desc->spec.height - r->y)))
73 return RKVC_STATUS_INVALID;
74 }
75 }
76 if (desc->header.struct_size >=
77 offsetof(rkvc_frame_desc, encode) + sizeof(desc->encode) &&
78 (desc->encode.bitrate_bps < 0 ||
79 desc->encode.gop_size > INT32_MAX))
80 return RKVC_STATUS_INVALID;
81 return RKVC_STATUS_OK;
82}

85static rkvc_frame *frame_from_desc(const rkvc_frame_desc *desc,
86 void (*free_fn)(void *), void *free_ctx) {
87 rkvc_frame *frame = rkvc_g_calloc(1, sizeof(*frame));
88 if (!frame)
89 return NULL;
90 frame->refcount = 1;
91 frame->spec = desc->spec;
92 frame->data = desc->data;
93 frame->size = desc->size;
94 frame->fd = desc->fd;
95 frame->pts = desc->header.struct_size >=
96 offsetof(rkvc_frame_desc, pts) + sizeof(desc->pts)
97 ? desc->pts : RKVC_FRAME_TS_UNKNOWN;
98 frame->dts = desc->header.struct_size >=
99 offsetof(rkvc_frame_desc, dts) + sizeof(desc->dts)
100 ? desc->dts : RKVC_FRAME_TS_UNKNOWN;
101 frame->flags = desc->header.struct_size >=
102 offsetof(rkvc_frame_desc, flags) + sizeof(desc->flags)
103 ? desc->flags : 0;
104 if (desc->header.struct_size >=
105 offsetof(rkvc_frame_desc, roi_region_count) +
106 sizeof(desc->roi_region_count) &&
107 desc->roi_region_count) {
108 frame->roi_regions = rkvc_g_calloc(desc->roi_region_count,
109 sizeof(*frame->roi_regions));
110 if (!frame->roi_regions) {
111 rkvc_g_free(frame);
112 return NULL;
113 }
114 memcpy(frame->roi_regions, desc->roi_regions,
115 desc->roi_region_count * sizeof(*frame->roi_regions));
116 frame->roi_region_count = desc->roi_region_count;
117 }
118 if (desc->header.struct_size >=
119 offsetof(rkvc_frame_desc, encode) + sizeof(desc->encode))
120 frame->encode = desc->encode;
121 frame->free_fn = free_fn;
122 frame->free_ctx = free_ctx;
123 return frame;
124}

127static rkvc_status minimum_host_size(const rkvc_frame_spec *spec,
128 size_t *required) {
129 size_t row, base;
130 *required = 0;
131 if (spec->fmt == RKVC_FRAME_FMT_BITSTREAM ||
132 !spec->width || !spec->height)
133 return RKVC_STATUS_OK;
134 row = spec->stride;
135 if (!row) {
136 size_t bytes_per_pixel =
137 spec->fmt == RKVC_FRAME_FMT_P010 ? 2u :
138 spec->fmt == RKVC_FRAME_FMT_RGB24 ? 3u : 1u;
139 if ((size_t)spec->width > SIZE_MAX / bytes_per_pixel)
140 return RKVC_STATUS_INVALID;
141 row = (size_t)spec->width * bytes_per_pixel;
142 }
143 if (row > SIZE_MAX / spec->height)
144 return RKVC_STATUS_INVALID;
145 base = row * spec->height;
146 switch (spec->fmt) {
147 case RKVC_FRAME_FMT_NV12:
148 case RKVC_FRAME_FMT_NV21:
149 case RKVC_FRAME_FMT_YUV420P:
150 case RKVC_FRAME_FMT_P010:
151 if (base > SIZE_MAX - base / 2)
152 return RKVC_STATUS_INVALID;
153 *required = base + base / 2;
154 break;
155 case RKVC_FRAME_FMT_NV16:
156 if (base > SIZE_MAX / 2)
157 return RKVC_STATUS_INVALID;
158 *required = base * 2;
159 break;
160 case RKVC_FRAME_FMT_RGB24:
161 *required = base;
162 break;
163 default:
164 return RKVC_STATUS_FORMAT;
165 }
166 return RKVC_STATUS_OK;
167}

169rkvc_status rkvc_frame_wrap(const rkvc_frame_desc *desc, rkvc_frame **out) {
170 rkvc_status st;
171 if (!out)
172 return RKVC_STATUS_INVALID;
173 *out = NULL;
174 st = validate_desc(desc);
175 if (st != RKVC_STATUS_OK)
176 return st;
177 *out = frame_from_desc(desc, NULL, NULL);
178 return *out ? RKVC_STATUS_OK : RKVC_STATUS_NOMEM;

205rkvc_status rkvc_backend_frame_create_dmabuf(
206 const rkvc_frame_desc *desc, rkvc_frame **out) {
207 rkvc_frame_desc owned;
208 rkvc_frame *frame;
209 rkvc_status st;
210 int *fd;
211
212 if (!out)
213 return RKVC_STATUS_INVALID;
214 *out = NULL;
215 st = validate_desc(desc);
216 if (st != RKVC_STATUS_OK)
217 return st;
218 if (desc->spec.domain != RKVC_MEM_DOMAIN_DMABUF)
219 return RKVC_STATUS_FORMAT;
220 fd = rkvc_g_calloc(1, sizeof(*fd));
221 if (!fd)
222 return RKVC_STATUS_NOMEM;
223 *fd = dup(desc->fd);
224 if (*fd < 0) {
225 rkvc_g_free(fd);
226 return RKVC_STATUS_IO;
227 }
228 owned = *desc;
229 owned.fd = *fd;
230 frame = frame_from_desc(&owned, close_owned_fd, fd);
231 if (!frame) {
232 close_owned_fd(fd);
233 return RKVC_STATUS_NOMEM;
234 }
235 *out = frame;
236 return RKVC_STATUS_OK;
237}

239rkvc_status rkvc_frame_wrap_host(const rkvc_frame_spec *spec, void *data,
240 size_t size, rkvc_frame **out) {
241 rkvc_frame_desc desc;
242 size_t required;
243 rkvc_status st;
244
245 if (!spec || !out)
246 return RKVC_STATUS_INVALID;
247 *out = NULL;
248 if (spec->fmt == RKVC_FRAME_FMT_UNKNOWN ||
249 spec->domain != RKVC_MEM_DOMAIN_HOST) {
250 return RKVC_STATUS_FORMAT;
251 }
252 st = minimum_host_size(spec, &required);
253 if (st != RKVC_STATUS_OK)
254 return st;
255 if (data && size != SIZE_MAX && size < required)
256 return RKVC_STATUS_INVALID;
257
258 rkvc_frame_desc_init(&desc, sizeof(desc));
259 desc.spec = *spec;
260 desc.data = data;
261 desc.size = size == SIZE_MAX ? 0 : size;
262 return rkvc_frame_wrap(&desc, out);
263}

271void rkvc_frame_release(rkvc_frame *frame) {
272 if (!frame)
273 return;
274 if (__atomic_sub_fetch(&frame->refcount, 1, __ATOMIC_ACQ_REL) == 0) {
275 if (frame->free_fn)
276 frame->free_fn(frame->free_ctx);
277 rkvc_g_free(frame->roi_regions);
278 rkvc_g_free(frame);
279 }
280}
