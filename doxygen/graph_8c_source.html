22 return NULL;

25 if (n->in_ports[i].name && strcmp(n->in_ports[i].name, name) == 0)
26 return &n->in_ports[i];
27 } else {
28 for (i = 0; i < n->out_count; ++i)
29 if (n->out_ports[i].name && strcmp(n->out_ports[i].name, name) == 0)
30 return &n->out_ports[i];
31 }
32 return NULL;

52static int negotiate_edge(rkvc_port *out, rkvc_port *in) {
53 rkvc_frame_spec resolved;
54
55 if (out->fmt.fmt == RKVC_FRAME_FMT_UNKNOWN &&
56 in->fmt.fmt == RKVC_FRAME_FMT_UNKNOWN)
57 return 0;
58 if (out->fmt.fmt == RKVC_FRAME_FMT_UNKNOWN)
59 resolved = in->fmt;
60 else if (in->fmt.fmt == RKVC_FRAME_FMT_UNKNOWN)
61 resolved = out->fmt;
62 else {
63 if (out->fmt.fmt != in->fmt.fmt ||
64 out->fmt.domain != in->fmt.domain ||
65 out->fmt.modifier != in->fmt.modifier)
66 return -1;
67 resolved = out->fmt;
68 if (merge_dimension(out->fmt.width, in->fmt.width,
69 &resolved.width) != 0 ||
70 merge_dimension(out->fmt.height, in->fmt.height,
71 &resolved.height) != 0 ||
72 merge_dimension(out->fmt.stride, in->fmt.stride,
73 &resolved.stride) != 0 ||
74 merge_dimension(out->fmt.ver_stride, in->fmt.ver_stride,
75 &resolved.ver_stride) != 0)
76 return -1;
77 }
78 out->fmt = resolved;
79 in->fmt = resolved;
80 return 0;
81}

83int rkvc_node_connect(rkvc_node *a, const char *out_name, rkvc_node *b,
84 const char *in_name) {
85 rkvc_port *out = a ? rkvc_node_get_port(a, out_name, 0) : NULL;
86 rkvc_port *in = b ? rkvc_node_get_port(b, in_name, 1) : NULL;
87 rkvc_graph *g = a ? a->graph : NULL;
88 rkvc_queue *q;
89 if (!out || !in || !g)
90 return -2; /* 无效端口或未绑定图 */
91 q = rkvc_queue_create(g->queue_capacity ? g->queue_capacity : 4);
92 if (!q)
93 return -1;
94 out->queue = q;
95 in->queue = q;
96 return 0;
97}

152static int bind_step_model(rkvc_graph *g, rkvc_node *n,
153 const rkvc_request *req, rkvc_diag **diag) {
154 const rkvc_rkmodel *m;
155 rkvc_model_binding binding;
156 uint32_t default_kind = RKMODEL_PAYLOAD_RKNN;
157 rkvc_status st;
158
159 if (!g->ctx) {
160 if (diag)
161 rkvc_diag_push(diag, RKVC_STATUS_INTERNAL, 1, n->ops->id,
162 "model binding requires a context");
163 return (int)RKVC_STATUS_INTERNAL;
164 }
165 m = rkvc_model_registry_select(g->ctx, req, diag);
166 if (!m) {
167 if (diag)
168 rkvc_diag_push(diag, RKVC_STATUS_NOT_FOUND, 1, n->ops->id,
169 "transform node requires a model but the "
170 "registry has no compatible candidate");
171 return (int)RKVC_STATUS_NEGOTIATE;
172 }
173 /* 缺省交付 RKNN 载荷；容器未携带时回退到首个载荷（PMF 等角色）。 */
174 if (!(m->info.payload_mask & (1u << RKMODEL_PAYLOAD_RKNN)) &&
175 m->payload_count)
176 default_kind = m->payloads[0].kind;
177
178 release_model_payload(g);
179 if (m->payload_count) {
180 g->payload_views = rkvc_g_calloc(m->payload_count,
181 sizeof(*g->payload_views));
182 g->payload_buffers = rkvc_g_calloc(m->payload_count,
183 sizeof(*g->payload_buffers));
184 if (!g->payload_views || !g->payload_buffers) {
185 release_model_payload(g);
186 if (diag)
187 rkvc_diag_push(diag, RKVC_STATUS_NOMEM, 1, n->ops->id,
188 "model payload view alloc failed");
189 return (int)RKVC_STATUS_NOMEM;
190 }
191 for (uint32_t i = 0; i < m->payload_count; ++i) {
192 st = rkvc_rkmodel_load_payload(m, m->payloads[i].kind,
193 &g->payload_buffers[i],
194 &binding.payload_size);
195 if (st != RKVC_STATUS_OK) {
196 release_model_payload(g);
197 if (diag)
198 rkvc_diag_push(diag, st, 1, n->ops->id,
199 "model payload load failed");
200 return (int)st;
201 }
202 g->payload_views[i].kind = m->payloads[i].kind;
203 g->payload_views[i].data = g->payload_buffers[i];
204 g->payload_views[i].size = binding.payload_size;
205 }
206 g->payload_count = m->payload_count;
207 }
208
209 binding.info = &m->info;
210 binding.payloads = g->payload_views;
211 binding.payload_count = g->payload_count;
212 for (size_t i = 0; i < g->payload_count; ++i) {
213 if (g->payload_views[i].kind == default_kind) {
214 binding.payload = g->payload_views[i].data;
215 binding.payload_size = g->payload_views[i].size;
216 break;
217 }
218 }
219 if (!binding.payloads) {
220 binding.payload = NULL;
221 binding.payload_size = 0;
222 }
223 if (n->ops->bind_model(n, &binding, diag) != 0) {
224 release_model_payload(g);
225 return (int)RKVC_STATUS_NEGOTIATE;
226 }
227 return 0;
228}

231static void reverse_release(rkvc_graph *g, size_t created) {
232 size_t i;
233 /* 逆序关闭已打开节点 */
234 for (i = created; i > 0; --i) {
235 rkvc_node *n = g->nodes[i - 1];
236 if (n && n->state == RKVC_NODE_OPEN && n->ops->close)
237 n->ops->close(n);
238 if (n)
239 n->state = RKVC_NODE_CLOSED;
240 }
241 /* 逆序销毁已创建节点 */
242 for (i = created; i > 0; --i) {
243 rkvc_node *n = g->nodes[i - 1];
244 if (n && n->ops->destroy)
245 n->ops->destroy(n);
246 g->nodes[i - 1] = NULL;
247 }
248 /* 释放连接队列 */
249 for (i = 0; i < g->queue_count; ++i)
250 rkvc_queue_destroy(g->queues[i]);
251 g->queue_count = 0;
252}

254int rkvc_graph_build(rkvc_graph *g, const rkvc_plan *plan, rkvc_diag **diag) {
255 size_t i, created = 0;
256
257 if (!g || !plan)
258 return -2;
259 g->failure_step = SIZE_MAX;
260 g->node_count = plan->step_count;
261 g->nodes = rkvc_g_calloc(g->node_count ? g->node_count : 1, sizeof(*g->nodes));
262 g->queues = rkvc_g_calloc(g->node_count ? g->node_count : 1, sizeof(*g->queues));
263 if (!g->nodes || !g->queues) {
264 rkvc_g_free(g->nodes); g->nodes = NULL;
265 rkvc_g_free(g->queues); g->queues = NULL;
266 if (diag) rkvc_diag_push(diag, RKVC_STATUS_NOMEM, 1, "graph", "alloc");
267 return (int)RKVC_STATUS_NOMEM;
268 }
269
270 for (i = 0; i < g->node_count; ++i) {
271 rkvc_node *n = plan->steps[i].factory
272 ->create(plan->steps[i].factory, &plan->steps[i].request,
273 plan->steps[i].factory->create_ctx);
274 if (!n) {
275 g->failure_step = i;
276 if (diag) rkvc_diag_push(diag, RKVC_STATUS_INTERNAL, 1,
277 plan->steps[i].factory->id, "create failed");
278 reverse_release(g, created);
279 g->node_count = 0;
280 return (int)RKVC_STATUS_INTERNAL;
281 }
282 n->graph = g;
283 n->idx = i;
284 n->state = RKVC_NODE_CREATED;
285 g->nodes[i] = n;
286 /* 声明 bind_model 的节点在 configure 前拿到已选模型（含 I/O 契约
287 * 所需的载荷字节）；失败淘汰该候选并交由计划回退。 */
288 if (n->ops->bind_model) {
289 int brc = bind_step_model(g, n, &plan->steps[i].request, diag);
290 if (brc != 0) {
291 g->failure_step = i;
292 reverse_release(g, i + 1); /* 节点 i 已创建，须一并销毁 */
293 g->node_count = 0;
294 return brc;
295 }
296 }
297 created = i + 1;
298 }
299
300 /* 连接线性链：节点 i 的最后一个输出 → 节点 i+1 的第一个输入 */
301 for (i = 0; i + 1 < g->node_count; ++i) {
302 rkvc_node *src = g->nodes[i];
303 rkvc_node *dst = g->nodes[i + 1];
304 rkvc_port *out = src->out_count ? &src->out_ports[src->out_count - 1] : NULL;
305 rkvc_port *in = dst->in_count ? &dst->in_ports[0] : NULL;
306 rkvc_queue *q;
307 if (!out || !in) {
308 g->failure_step = i + 1;
309 if (diag) rkvc_diag_push(diag, RKVC_STATUS_NEGOTIATE, 1,
310 "graph", "linear link missing port");
311 reverse_release(g, created);
312 g->node_count = 0;
313 return (int)RKVC_STATUS_NEGOTIATE;
314 }
315 q = rkvc_queue_create(g->queue_capacity ? g->queue_capacity : 4);
316 if (!q) {
317 if (diag) rkvc_diag_push(diag, RKVC_STATUS_NOMEM, 1, "graph", "queue alloc");
318 reverse_release(g, created);
319 g->node_count = 0;
320 return (int)RKVC_STATUS_NOMEM;
321 }
322 out->queue = q;
323 in->queue = q;
324 g->queues[g->queue_count++] = q;
325 }
326
327 /* 协商（configure）：只定格式，不打开设备 */
328 for (i = 0; i < g->node_count; ++i) {
329 rkvc_node *n = g->nodes[i];
330 if (n->ops->configure && n->ops->configure(n, diag) != 0) {
331 g->failure_step = i;
332 reverse_release(g, created);
333 g->node_count = 0;
334 return (int)RKVC_STATUS_NEGOTIATE;
335 }
336 n->state = RKVC_NODE_CONFIGURED;
337 }
338
339 /* configure 完成后统一解析每条边；不兼容格式必须在 open 设备前失败。 */
340 for (i = 0; i + 1 < g->node_count; ++i) {
341 rkvc_node *src = g->nodes[i];
342 rkvc_node *dst = g->nodes[i + 1];
343 rkvc_port *out = &src->out_ports[src->out_count - 1];
344 rkvc_port *in = &dst->in_ports[0];
345 if (negotiate_edge(out, in) != 0) {
346 g->failure_step = i + 1;
347 if (diag)
348 rkvc_diag_push(diag, RKVC_STATUS_NEGOTIATE, 1,
349 out->name ? out->name : "graph",
350 "incompatible connected port specs");
351 reverse_release(g, created);
352 g->node_count = 0;
353 return (int)RKVC_STATUS_NEGOTIATE;
354 }
355 }
356
357 /* FRAME_SINK 端点无 source 节点：单节点图中编码节点输入端口无人
358 * 填充格式（节点间边协商不覆盖它），open 阶段会读到 UNKNOWN 而失败。
359 * 把请求端点声明的格式/几何注入首节点输入端口；字段级合并，冲突即
360 * 协商失败。末节点输出端口同理注入输出端点声明。 */
361 if (g->node_count) {
362 const rkvc_request *req = &plan->steps[0].request;
363 rkvc_node *first = g->nodes[0];
364 rkvc_node *last = g->nodes[g->node_count - 1];
365 if (req->input.kind == RKVC_ENDPOINT_FRAME_SINK &&
366 first->in_count && req->input.fmt != RKVC_FRAME_FMT_UNKNOWN) {
367 rkvc_frame_spec resolved = first->in_ports[0].fmt;
368 if (resolved.fmt == RKVC_FRAME_FMT_UNKNOWN) {
369 resolved.fmt = req->input.fmt;
370 resolved.domain = RKVC_MEM_DOMAIN_HOST;
371 } else if (resolved.fmt != req->input.fmt) {
372 g->failure_step = 0;
373 if (diag)
374 rkvc_diag_push(diag, RKVC_STATUS_NEGOTIATE, 1,
375 first->in_ports[0].name
376 ? first->in_ports[0].name : "graph",
377 "input endpoint fmt conflicts with node");
378 reverse_release(g, created);
379 g->node_count = 0;
380 return (int)RKVC_STATUS_NEGOTIATE;
381 }
382 if (merge_dimension(resolved.width, req->width, &resolved.width) != 0 ||
383 merge_dimension(resolved.height, req->height, &resolved.height) != 0) {
384 reverse_release(g, created);
385 g->node_count = 0;
386 return (int)RKVC_STATUS_NEGOTIATE;
387 }
388 first->in_ports[0].fmt = resolved;
389 }
390 if (req->output.kind == RKVC_ENDPOINT_FRAME_SINK &&
391 last->out_count && req->output.fmt != RKVC_FRAME_FMT_UNKNOWN) {
392 rkvc_frame_spec resolved = last->out_ports[last->out_count - 1].fmt;
393 if (resolved.fmt == RKVC_FRAME_FMT_UNKNOWN) {
394 resolved.fmt = req->output.fmt;
395 resolved.domain = RKVC_MEM_DOMAIN_HOST;
396 } else if (resolved.fmt != req->output.fmt) {
397 g->failure_step = g->node_count - 1;
398 if (diag)
399 rkvc_diag_push(diag, RKVC_STATUS_NEGOTIATE, 1,
400 last->out_ports[last->out_count - 1].name
401 ? last->out_ports[last->out_count - 1].name
402 : "graph",
403 "output endpoint fmt conflicts with node");
404 reverse_release(g, created);
405 g->node_count = 0;
406 return (int)RKVC_STATUS_NEGOTIATE;
407 }
408 last->out_ports[last->out_count - 1].fmt = resolved;
409 }
410 }
411
412 g->state = 1;
413 return 0;
414}

416int rkvc_graph_open(rkvc_graph *g, rkvc_diag **diag) {
417 size_t i;
418
419 if (!g)
420 return (int)RKVC_STATUS_INVALID;
421 if (g->state == 2)
422 return 0;
423 if (g->state != 1)
424 return (int)RKVC_STATUS_INVALID;
425 g->failure_step = SIZE_MAX;
426
427 /* 实例化（open）：只在 job_start 阶段打开设备、分配大块内存。 */
428 for (i = 0; i < g->node_count; ++i) {
429 rkvc_node *n = g->nodes[i];
430 if (n->ops->open && n->ops->open(n, diag) != 0) {
431 g->failure_step = i;
432 reverse_release(g, g->node_count);
433 g->node_count = 0;
434 g->state = 3;
435 return (int)RKVC_STATUS_HW;
436 }
437 n->state = RKVC_NODE_OPEN;
438 }
439
440 g->state = 2;
441 return 0;
442}

444int rkvc_graph_run(rkvc_graph *g, rkvc_diag **diag) {
445 rkvc_exec *e;
446 int rc;
447 if (!g)
448 return -2;
449 if (g->state == 1) {
450 rc = rkvc_graph_open(g, diag);
451 if (rc != 0)
452 return rc;
453 }
454 if (g->state != 2)
455 return (int)RKVC_STATUS_INVALID;
456 e = rkvc_exec_create(g, g->node_count);
457 if (!e)
458 return (int)RKVC_STATUS_NOMEM;
459 g->exec = e;
460 rc = rkvc_exec_run(e, diag);
461 return rc;
462}
