80int rkvc_queue_push(rkvc_queue *q, rkvc_frame *f, void *execp) {
81 rkvc_exec *e = execp;
82 pthread_mutex_lock(&q->m);
83 while (q->count == q->capacity && !q->eos &&
84 !(e && atomic_load(&e->canceled))) {
85 pthread_cond_wait(&q->not_full, &q->m);
86 }
87 if (e && atomic_load(&e->canceled)) {
88 pthread_mutex_unlock(&q->m);
89 return (int)RKVC_STATUS_CANCELED;
90 }
91 if (q->eos) {
92 pthread_mutex_unlock(&q->m);
93 return (int)RKVC_STATUS_EOF;
94 }
95 q->slots[(q->head + q->count) % q->capacity] = f;
96 q->count++;
97 pthread_cond_signal(&q->not_empty);
98 pthread_mutex_unlock(&q->m);
99 return 0;
100}

102int rkvc_queue_try_push(rkvc_queue *q, rkvc_frame *f, void *execp) {
103 rkvc_exec *e = execp;
104 int rc = 0;
105 if (!q || !f)
106 return (int)RKVC_STATUS_INVALID;
107 pthread_mutex_lock(&q->m);
108 if (e && atomic_load(&e->canceled))
109 rc = (int)RKVC_STATUS_CANCELED;
110 else if (q->eos)
111 rc = (int)RKVC_STATUS_EOF;
112 else if (q->count == q->capacity)
113 rc = (int)RKVC_STATUS_AGAIN;
114 else {
115 q->slots[(q->head + q->count) % q->capacity] = f;
116 q->count++;
117 pthread_cond_signal(&q->not_empty);
118 }
119 pthread_mutex_unlock(&q->m);
120 return rc;
121}

123int rkvc_queue_pop(rkvc_queue *q, rkvc_frame **out, void *execp) {
124 rkvc_exec *e = execp;
125 pthread_mutex_lock(&q->m);
126 while (q->count == 0 && !q->eos &&
127 !(e && atomic_load(&e->canceled))) {
128 pthread_cond_wait(&q->not_empty, &q->m);
129 }
130 if (q->count == 0) {
131 int canceled = e && atomic_load(&e->canceled);
132 pthread_mutex_unlock(&q->m);
133 if (canceled)
134 return (int)RKVC_STATUS_CANCELED;
135 return 0; /* EOS */
136 }
137 *out = q->slots[q->head];
138 q->head = (q->head + 1) % q->capacity;
139 q->count--;
140 pthread_cond_signal(&q->not_full);
141 pthread_mutex_unlock(&q->m);
142 return 1;
143}

145int rkvc_queue_try_pop(rkvc_queue *q, rkvc_frame **out, void *execp) {
146 rkvc_exec *e = execp;
147 int rc = 0;
148 if (!q || !out)
149 return (int)RKVC_STATUS_INVALID;
150 pthread_mutex_lock(&q->m);
151 if (q->count) {
152 *out = q->slots[q->head];
153 q->head = (q->head + 1) % q->capacity;
154 q->count--;
155 pthread_cond_signal(&q->not_full);
156 rc = 1;
157 } else if (e && atomic_load(&e->canceled))
158 rc = (int)RKVC_STATUS_CANCELED;
159 else
160 rc = q->eos ? 0 : (int)RKVC_STATUS_AGAIN;
161 pthread_mutex_unlock(&q->m);
162 return rc;
163}

182static void exec_broadcast(rkvc_exec *e) {
183 size_t i;

196static void *exec_worker(void *argp) {
197 struct worker_arg *w = argp;
198 rkvc_exec *e = w->e;
199 rkvc_node *node = e->g->nodes[w->idx];
200 rkvc_queue *inq = (w->idx == 0) ? e->input_queue : e->g->queues[w->idx - 1];
201 rkvc_queue *outq = (w->idx + 1 == e->g->node_count)
202 ? e->output_queue : e->g->queues[w->idx];
203
204 for (;;) {
205 rkvc_frame *f = NULL;
206 rkvc_diag *d = NULL;
207 int r = rkvc_queue_pop(inq, &f, e);
208 if (r < 0)
209 break; /* 取消 */
210 if (r == 0) { /* EOS */
211 if (node->ops->flush) {
212 int rc = node->ops->flush(node, &d);
213 if (rc != 0) {
214 atomic_store(&e->error_code,
215 rc < 0 ? rc : (int)RKVC_STATUS_INTERNAL);
216 atomic_store(&e->failed, 1);
217 }
218 }
219 rkvc_diag_release(d);
220 rkvc_queue_set_eos(outq);
221 break;
222 }
223 if (node->ops->process) {
224 int rc = node->ops->process(node, f, &d);
225 if (rc != 0) {
226 atomic_store(&e->error_code,
227 rc < 0 ? rc : (int)RKVC_STATUS_INTERNAL);
228 rkvc_diag_release(d);
229 rkvc_frame_release(f);
230 atomic_store(&e->failed, 1);
231 atomic_store(&e->canceled, 1);
232 exec_broadcast(e);
233 break;
234 }
235 }
236 rkvc_diag_release(d);
237 rkvc_frame_release(f);
238 }
239 rkvc_g_free(w); /* 每个线程自持有其 worker_arg */
240 return NULL;
241}

243struct rkvc_exec *rkvc_exec_create(rkvc_graph *g, size_t worker_threads) {
244 rkvc_exec *e = rkvc_g_calloc(1, sizeof(*e));
245 if (!e)
246 return NULL;
247 e->g = g;
248 e->nthreads = worker_threads;
249 e->qcap = g->queue_count + 2;
250 e->qlist = rkvc_g_calloc(e->qcap ? e->qcap : 1, sizeof(*e->qlist));
251 if (!e->qlist) {
252 rkvc_g_free(e);
253 return NULL;
254 }
255 e->input_queue = rkvc_queue_create(g->queue_capacity ? g->queue_capacity : 4);
256 e->output_queue = rkvc_queue_create(g->queue_capacity ? g->queue_capacity : 4);
257 if (!e->input_queue || !e->output_queue) {
258 rkvc_queue_destroy(e->input_queue);
259 rkvc_queue_destroy(e->output_queue);
260 rkvc_g_free(e->qlist);
261 rkvc_g_free(e);
262 return NULL;
263 }
264 exec_addq(e, e->input_queue);
265 exec_addq(e, e->output_queue);
266 for (size_t i = 0; i < g->queue_count; ++i)
267 exec_addq(e, g->queues[i]);
268
269 /* 链接首/末节点端口到执行器边界队列（流式 push/pull 出入口） */
270 if (g->node_count) {
271 rkvc_node *first = g->nodes[0];
272 rkvc_node *last = g->nodes[g->node_count - 1];
273 if (first->in_count)
274 first->in_ports[0].queue = e->input_queue;
275 if (last->out_count)
276 last->out_ports[last->out_count - 1].queue = e->output_queue;
277 }
278 return e;
279}

281int rkvc_exec_run(rkvc_exec *e, rkvc_diag **diag) {
282 size_t i;
283 int launch_error = 0;
284 if (!e || e->nthreads == 0)
285 return 0;
286 e->threads = rkvc_g_calloc(e->nthreads, sizeof(*e->threads));
287 if (!e->threads)
288 return (int)RKVC_STATUS_NOMEM;
289 for (i = 0; i < e->nthreads; ++i) {
290 struct worker_arg *wa = rkvc_g_calloc(1, sizeof(*wa));
291 if (!wa) {
292 launch_error = (int)RKVC_STATUS_NOMEM;
293 atomic_store(&e->canceled, 1);
294 exec_broadcast(e);
295 break;
296 }
297 wa->e = e;
298 wa->idx = i;
299 if (pthread_create(&e->threads[i], NULL, exec_worker, wa) != 0) {
300 rkvc_g_free(wa);
301 launch_error = (int)RKVC_STATUS_INTERNAL;
302 atomic_store(&e->canceled, 1);
303 exec_broadcast(e);
304 break;
305 }
306 e->started_threads++;
307 }
308 for (i = 0; i < e->started_threads; ++i)
309 pthread_join(e->threads[i], NULL);
310 e->joined = 1;
311
312 if (launch_error)
313 return launch_error;
314
315 if (atomic_load(&e->failed)) {
316 int code = atomic_load(&e->error_code);
317 if (diag)
318 rkvc_diag_push(diag, (rkvc_status)code, 3,
319 "executor", "node failed");
320 return code;
321 }
322 if (atomic_load(&e->canceled))
323 return (int)RKVC_STATUS_CANCELED;
324 return 0;
325}

358int rkvc_exec_pull(rkvc_exec *e, rkvc_frame **frame) {
359 int r;
360 if (!e || !frame)
361 return (int)RKVC_STATUS_INVALID;
362 r = rkvc_queue_pop(e->output_queue, frame, e);
363 if (r == 1)
364 return 0;
365 if (r == 0) {
366 if (atomic_load(&e->failed))
367 return atomic_load(&e->error_code);
368 return (int)RKVC_STATUS_EOF;
369 }
370 if (r < 0 && atomic_load(&e->failed))
371 return atomic_load(&e->error_code);
372 return r;
373}
