24void mlvc_px_extract_scales(const int32_t *z_r, int32_t *s0, int32_t *s1,
25 int YC, int YH, int YW, int ZH, int ZW)
26{
27 for (int c = 0; c < YC; c++) {
28 const int32_t *z0 = z_r +
29 (size_t)(c / MLVC_PX_CHANNEL_REPEAT) * ZH * ZW;
30 const int32_t *z1 = z_r +
31 (size_t)((c + YC) / MLVC_PX_CHANNEL_REPEAT) * ZH * ZW;
32 int32_t *p0 = s0 + (size_t)c * YH * YW;
33 int32_t *p1 = s1 + (size_t)c * YH * YW;
34
35 for (int zy = 0; zy * MLVC_PX_SPATIAL_REPEAT < YH; zy++) {
36 int y0 = zy * MLVC_PX_SPATIAL_REPEAT;
37 int ye = y0 + MLVC_PX_SPATIAL_REPEAT;
38 if (ye > YH) ye = YH;
39 const int32_t *z0r = z0 + (size_t)zy * ZW;
40 const int32_t *z1r = z1 + (size_t)zy * ZW;
41
42 for (int zx = 0; zx * MLVC_PX_SPATIAL_REPEAT < YW; zx++) {
43 int x0 = zx * MLVC_PX_SPATIAL_REPEAT;
44 int xe = x0 + MLVC_PX_SPATIAL_REPEAT;
45 if (xe > YW) xe = YW;
46 /* v0/v1 在 4×4 块内不变：每块算一次（原实现逐元素除法）*/
47 int v0 = abs(z0r[zx]);
48 int v1 = abs(z1r[zx]);
49 if (v0 > MLVC_PX_SCALE_MAX_IDX) v0 = MLVC_PX_SCALE_MAX_IDX;
50 if (v1 > MLVC_PX_SCALE_MAX_IDX) v1 = MLVC_PX_SCALE_MAX_IDX;
51
52 /* 满宽块（4 元素整行）：棋盘模式固定，向量化整行写入 */
53#ifdef MLVC_PX_NEON
54 if (xe - x0 == MLVC_PX_SPATIAL_REPEAT) {
55 int32x2_t h01 = vset_lane_s32(v1, vdup_n_s32(v0), 1);
56 int32x2_t h10 = vset_lane_s32(v0, vdup_n_s32(v1), 1);
57 int32x4_t pat01 = vcombine_s32(h01, h01);
58 int32x4_t pat10 = vcombine_s32(h10, h10);
59 for (int y = y0; y < ye; y++) {
60 int32_t *r0 = p0 + (size_t)y * YW + x0;
61 int32_t *r1 = p1 + (size_t)y * YW + x0;
62 if ((y & 1) == 0) {
63 vst1q_s32(r0, pat01);
64 vst1q_s32(r1, pat10);
65 } else {
66 vst1q_s32(r0, pat10);
67 vst1q_s32(r1, pat01);
68 }
69 }
70 continue;
71 }
72#endif
73 for (int y = y0; y < ye; y++) {
74 int32_t *r0 = p0 + (size_t)y * YW;
75 int32_t *r1 = p1 + (size_t)y * YW;
76 int par = y & 1;
77 for (int x = x0; x < xe; x++) {
78 int chk = (par == (x & 1));
79 r0[x] = chk ? v0 : v1;
80 r1[x] = chk ? v1 : v0;
81 }
82 }
83 }
84 }
85 }
86}

94/* 8×8 uint16 矩阵转置（8 个像素 × 8 通道 ↔ 8 个通道 × 8 像素）。
95 * 三级 trn：16-bit → 32-bit → 64-bit。输入 r[i] = 像素 i 的 8 个通道，
96 * 输出 r[k] = 通道 k 的 8 个像素。 */
97static inline void transpose_8x8_u16(uint16x8_t r[8])
98{
99 uint16x8_t t0 = vtrn1q_u16(r[0], r[1]), t1 = vtrn2q_u16(r[0], r[1]);
100 uint16x8_t t2 = vtrn1q_u16(r[2], r[3]), t3 = vtrn2q_u16(r[2], r[3]);
101 uint16x8_t t4 = vtrn1q_u16(r[4], r[5]), t5 = vtrn2q_u16(r[4], r[5]);
102 uint16x8_t t6 = vtrn1q_u16(r[6], r[7]), t7 = vtrn2q_u16(r[6], r[7]);
103
104 uint32x4_t v0 = vtrn1q_u32(vreinterpretq_u32_u16(t0), vreinterpretq_u32_u16(t2));
105 uint32x4_t v1 = vtrn2q_u32(vreinterpretq_u32_u16(t0), vreinterpretq_u32_u16(t2));
106 uint32x4_t v2 = vtrn1q_u32(vreinterpretq_u32_u16(t1), vreinterpretq_u32_u16(t3));
107 uint32x4_t v3 = vtrn2q_u32(vreinterpretq_u32_u16(t1), vreinterpretq_u32_u16(t3));
108 uint32x4_t v4 = vtrn1q_u32(vreinterpretq_u32_u16(t4), vreinterpretq_u32_u16(t6));
109 uint32x4_t v5 = vtrn2q_u32(vreinterpretq_u32_u16(t4), vreinterpretq_u32_u16(t6));
110 uint32x4_t v6 = vtrn1q_u32(vreinterpretq_u32_u16(t5), vreinterpretq_u32_u16(t7));
111 uint32x4_t v7 = vtrn2q_u32(vreinterpretq_u32_u16(t5), vreinterpretq_u32_u16(t7));
112
113 /* 实测映射：trn1/2(v0,v4)=ch0/4，trn1/2(v1,v5)=ch2/6，
114 * trn1/2(v2,v6)=ch1/5，trn1/2(v3,v7)=ch3/7 */
115 r[0] = vreinterpretq_u16_u64(vtrn1q_u64(vreinterpretq_u64_u32(v0), vreinterpretq_u64_u32(v4)));
116 r[1] = vreinterpretq_u16_u64(vtrn1q_u64(vreinterpretq_u64_u32(v2), vreinterpretq_u64_u32(v6)));
117 r[2] = vreinterpretq_u16_u64(vtrn1q_u64(vreinterpretq_u64_u32(v1), vreinterpretq_u64_u32(v5)));
118 r[3] = vreinterpretq_u16_u64(vtrn1q_u64(vreinterpretq_u64_u32(v3), vreinterpretq_u64_u32(v7)));
119 r[4] = vreinterpretq_u16_u64(vtrn2q_u64(vreinterpretq_u64_u32(v0), vreinterpretq_u64_u32(v4)));
120 r[5] = vreinterpretq_u16_u64(vtrn2q_u64(vreinterpretq_u64_u32(v2), vreinterpretq_u64_u32(v6)));
121 r[6] = vreinterpretq_u16_u64(vtrn2q_u64(vreinterpretq_u64_u32(v1), vreinterpretq_u64_u32(v5)));
122 r[7] = vreinterpretq_u16_u64(vtrn2q_u64(vreinterpretq_u64_u32(v3), vreinterpretq_u64_u32(v7)));
123}

125void mlvc_px_nc1hwc2_to_nchw(const uint16_t *src, int32_t *dst,
126 int C, int H, int W)
127{
128 int C1 = C >> 3;
129 for (int c1 = 0; c1 < C1; c1++) {
130 for (int y = 0; y < H; y++) {
131 const uint16_t *sp = src + (((size_t)c1 * H + y) * W) * 8;
132 int32_t *dp[8];
133 for (int k = 0; k < 8; k++)
134 dp[k] = dst + (((size_t)(c1 * 8 + k) * H + y) * W);
135
136 int x = 0;
137 for (; x + 8 <= W; x += 8) {
138 uint16x8_t r[8];
139 for (int k = 0; k < 8; k++)
140 r[k] = vld1q_u16(sp + k * 8);
141 transpose_8x8_u16(r);
142 for (int k = 0; k < 8; k++) {
143 float16x8_t h = vreinterpretq_f16_u16(r[k]);
144 /* vcvtnq：round-to-nearest-even，等价 lrintf 默认舍入 */
145 vst1q_s32(dp[k] + x,
146 vcvtnq_s32_f32(vcvt_f32_f16(vget_low_f16(h))));
147 vst1q_s32(dp[k] + x + 4,
148 vcvtnq_s32_f32(vcvt_high_f32_f16(h)));
149 }
150 sp += 64;
151 }
152 int xt = x; /* 尾部像素相对 sp 的偏移 */
153 for (; x < W; x++)
154 for (int k = 0; k < 8; k++)
155 dp[k][x] = lrintf(mlvc_px_f16_to_f32(sp[(size_t)(x - xt) * 8 + k]));
156 }
157 }
158}

190/* ════════════════════════════════════════════════════════════════════ */
191/* NC1HWC2 → DepthToSpace(DCR) 融合 */
192/* ════════════════════════════════════════════════════════════════════ */
193
194void mlvc_px_nc1hwc2_d2s_dcr_f16(const uint16_t *src, uint16_t *out,
195 int C1, int H, int W, int C2, int bs)
196{
197 int C = C1 * C2;
198 int oc = C / (bs * bs);
199 int oh = H * bs, ow = W * bs;
200
201 for (int c = 0; c < oc; c++) {
202 for (int dy = 0; dy < bs; dy++) {
203 /* ic = dy*(bs*oc) + dx*oc + c（ONNX DCR），与 w 无关：
204 * 每个 dx 的源通道偏移在 h/w 循环外预计算一次 */
205 size_t choff[16];
206 for (int dx = 0; dx < bs; dx++) {
207 int ic = dy * (bs * oc) + dx * oc + c;
208 choff[dx] = (size_t)(ic / C2) * H * W * C2 + (size_t)(ic % C2);
209 }
210 for (int h = 0; h < H; h++) {
211 uint16_t *orow = out + ((size_t)c * oh + h * bs + dy) * ow;
212 const uint16_t *srow = src + (size_t)h * W * C2;
213 for (int w = 0; w < W; w++) {
214 /* 输出行连续写入（原实现输出步长 bs 的散写）*/
215 for (int dx = 0; dx < bs; dx++)
216 orow[w * bs + dx] = srow[(size_t)w * C2 + choff[dx]];
217 }
218 }
219 }
220 }
221}

227void mlvc_px_yuv_to_nhwc_fp16(const uint8_t *yp, int y_stride,
228 const uint8_t *up, const uint8_t *vp,
229 int uv_stride, int nv12,
230 int W, int H, uint16_t *nhwc)
231{
232 /* 256 项 LUT：构建表达式与优化前逐像素 (v * (1/255)) 完全一致 */
233 uint16_t lut[256];
234 for (int i = 0; i < 256; i++)
235 lut[i] = mlvc_px_f32_to_f16((float)i * (1.0f / 255.0f));
236
237 int y = 0;
238 /* 两行一组：共享一行 UV，UV 索引计算减半 */
239 for (; y + 2 <= H; y += 2) {
240 const uint8_t *yr0 = yp + (size_t)y * y_stride;
241 const uint8_t *yr1 = yr0 + y_stride;
242 const uint8_t *uvr = up + (size_t)(y / 2) * uv_stride;
243 const uint8_t *vvr = vp + (size_t)(y / 2) * uv_stride;
244 uint16_t *d0 = nhwc + (size_t)y * W * 3;
245 uint16_t *d1 = d0 + (size_t)W * 3;
246 for (int x = 0; x < W; x++) {
247 size_t uo = (size_t)(x >> 1) * (nv12 ? 2 : 1);
248 uint16_t u = lut[uvr[uo]];
249 uint16_t v = lut[vvr[uo]];
250 d0[x * 3 + 0] = lut[yr0[x]];
251 d0[x * 3 + 1] = u;
252 d0[x * 3 + 2] = v;
253 d1[x * 3 + 0] = lut[yr1[x]];
254 d1[x * 3 + 1] = u;
255 d1[x * 3 + 2] = v;
256 }
257 }
258 if (y < H) { /* 奇数高度尾行 */
259 const uint8_t *yr0 = yp + (size_t)y * y_stride;
260 const uint8_t *uvr = up + (size_t)(y / 2) * uv_stride;
261 const uint8_t *vvr = vp + (size_t)(y / 2) * uv_stride;
262 uint16_t *d0 = nhwc + (size_t)y * W * 3;
263 for (int x = 0; x < W; x++) {
264 size_t uo = (size_t)(x >> 1) * (nv12 ? 2 : 1);
265 d0[x * 3 + 0] = lut[yr0[x]];
266 d0[x * 3 + 1] = lut[uvr[uo]];
267 d0[x * 3 + 2] = lut[vvr[uo]];
268 }
269 }
270}

278/* f16x8 → u8x8：×255 + round-to-nearest-even + 饱和到 0..255。
279 * 语义等价 clip(0,1) 后 lrintf(f*255) 再截断饱和。 */
280static inline uint8x8_t f16x8_to_u8x8_sat255(uint16x8_t h)
281{
282 float16x8_t f = vreinterpretq_f16_u16(h);
283 float32x4_t lo = vmulq_n_f32(vcvt_f32_f16(vget_low_f16(f)), 255.0f);
284 float32x4_t hi = vmulq_n_f32(vcvt_high_f32_f16(f), 255.0f);
285 uint16x4_t nlo = vqmovun_s32(vcvtnq_s32_f32(lo));
286 uint16x4_t nhi = vqmovun_s32(vcvtnq_s32_f32(hi));
287 return vqmovn_u16(vcombine_u16(nlo, nhi));
288}

290void mlvc_px_nchw_yuv_fp16_to_nv12_planes(const uint16_t *src, int W, int H,
291 uint8_t *yp, int y_stride,
292 uint8_t *uv, int uv_stride)

295
296 for (int y = 0; y < H; y++) {
297 const uint16_t *sp = src + (size_t)y * W;
298 uint8_t *dp = yp + (size_t)y * y_stride;
299 int x = 0;
300 for (; x + 8 <= W; x += 8)
301 vst1_u8(dp + x, f16x8_to_u8x8_sat255(vld1q_u16(sp + x)));
302 for (; x < W; x++) {
303 int v = (int)lrintf(mlvc_px_f16_to_f32(sp[x]) * 255.0f);
304 dp[x] = (uint8_t)(v < 0 ? 0 : (v > 255 ? 255 : v));
305 }
306 }
307 for (int y = 0; y < H / 2; y++) {
308 const uint16_t *ur = src + plane + (size_t)(y * 2) * W;
309 const uint16_t *vr = src + 2 * plane + (size_t)(y * 2) * W;
310 uint8_t *dp = uv + (size_t)y * uv_stride;
311 int x = 0;
312 for (; x + 8 <= W / 2; x += 8) {
313 /* vld2 解交织：偶数列即 2x 采样点 */
314 uint16x8x2_t uu = vld2q_u16(ur + x * 2);
315 uint16x8x2_t vv = vld2q_u16(vr + x * 2);
316 uint8x8x2_t o = { { f16x8_to_u8x8_sat255(uu.val[0]),
317 f16x8_to_u8x8_sat255(vv.val[0]) } };
318 vst2_u8(dp + x * 2, o);
319 }
320 for (; x < W / 2; x++) {
321 int u = (int)lrintf(mlvc_px_f16_to_f32(ur[x * 2]) * 255.0f);
322 int v = (int)lrintf(mlvc_px_f16_to_f32(vr[x * 2]) * 255.0f);
