19rkvc_err rkvc_mpp_enc_open(rkvc_mpp_enc **out, const rkvc_mpp_enc_config *cfg)
20{
21 if (!out || !cfg || !cfg->route)
22 return RKVC_ERR_INVALID;
23
24 *out = NULL;
25 rkvc_err init_err = rkvc_init();
26 if (init_err != RKVC_OK)
27 return init_err;
28
29 const char *name = cfg->route->enc_name;
30 const AVCodec *codec = avcodec_find_encoder_by_name(name);
31 if (!codec) {
32 if (cfg->route->codec == RKVC_CODEC_HEVC)
33 codec = avcodec_find_encoder(AV_CODEC_ID_HEVC);
34 else
35 codec = avcodec_find_encoder(AV_CODEC_ID_H264);
36 }
37 if (!codec)
38 return RKVC_ERR_NOT_FOUND;
39
40 rkvc_err perm = rkvc_check_hw_permissions();
41 if (perm != RKVC_OK)
42 return perm;
43
44 rkvc_mpp_enc *enc = rkvc_calloc(1, sizeof(*enc));
45 if (!enc)
46 return RKVC_ERR_NOMEM;
47
48 enc->route = cfg->route;
49 enc->ctx = avcodec_alloc_context3(codec);
50 if (!enc->ctx) {
51 rkvc_mpp_enc_close(enc);
52 return RKVC_ERR_NOMEM;
53 }
54
55 enc->ctx->width = cfg->width;
56 enc->ctx->height = cfg->height;
57 enc->ctx->time_base = (AVRational){cfg->fps_den, cfg->fps_num};
58 enc->ctx->framerate = (AVRational){cfg->fps_num, cfg->fps_den};
59 enc->ctx->bit_rate = cfg->bitrate;
60 enc->ctx->gop_size = cfg->gop_size;
61 enc->ctx->max_b_frames = 0;
62 enc->ctx->thread_count = 1;
63
64 if (cfg->low_latency)
65 enc->ctx->flags |= AV_CODEC_FLAG_LOW_DELAY;
66
67 enc->ctx->pix_fmt = rkvc_to_av_pix_fmt(cfg->input_format);
68
69 if (!rkvc_is_valid_rc_mode(cfg->rc_mode)) {
70 rkvc_mpp_enc_close(enc);
71 return RKVC_ERR_INVALID;
72 }
73
74 AVDictionary *enc_opts = NULL;
75 rkvc_err perr = rkvc_dict_parse_opts(&enc_opts, cfg->codec_opts);
76 if (perr != RKVC_OK) {
77 rkvc_mpp_enc_close(enc);
78 return perr;
79 }
80
81 perr = rkvc_dict_set_int(&enc_opts, "rc_mode", cfg->rc_mode);
82 if (perr != RKVC_OK) {
83 rkvc_dict_free(&enc_opts);
84 rkvc_mpp_enc_close(enc);
85 return perr;
86 }
87 if (cfg->qp_init >= 0) {
88 perr = rkvc_dict_set_int(&enc_opts, "qp_init", cfg->qp_init);
89 if (perr != RKVC_OK) {
90 rkvc_dict_free(&enc_opts);
91 rkvc_mpp_enc_close(enc);
92 return perr;
93 }
94 }
95
96 perr = rkvc_opt_set_dict(enc->ctx->priv_data, &enc_opts);
97 if (perr != RKVC_OK) {
98 rkvc_dict_free(&enc_opts);
99 rkvc_mpp_enc_close(enc);
100 return perr;
101 }
102
103 AVBufferRef *hw = NULL;
104 rkvc_err herr = rkvc_get_hw_device_ctx(&hw);
105 if (herr != RKVC_OK)
106 rkvc_log_print(AV_LOG_WARNING,
107 "mpp_enc: RKMPP hw device init failed (err=%d), "
108 "encoder open may fail next\n", herr);
109 if (hw)
110 av_buffer_unref(&hw);
111
112 rkvc_err err = rkvc_codec_open2(enc->ctx, codec, &enc_opts, "mpp_enc");
113 rkvc_dict_free(&enc_opts);
114 if (err != RKVC_OK) {
115 rkvc_mpp_enc_close(enc);
116 return err;
117 }
118
119 enc->pkt = av_packet_alloc();
120 if (!enc->pkt) {
121 rkvc_mpp_enc_close(enc);
122 return RKVC_ERR_NOMEM;
123 }
124
125 *out = enc;
126 return RKVC_OK;
127}

157rkvc_err rkvc_mpp_enc_send_frame_roi_ex(rkvc_mpp_enc *enc, rkvc_buffer *frame,
158 const rkvc_roi_rect *rois, int roi_count,
159 int force_idr)
160{
161 if (!enc || !enc->ctx)
162 return RKVC_ERR_INVALID;
163 if (enc->flushed)
164 return RKVC_ERR_EOF;
165 if (!frame)
166 return rkvc_mpp_enc_drain(enc);
167
168 AVFrame *avf = NULL;
169 rkvc_err err = rkvc_buffer_to_host_frame(frame, &avf);
170 if (err != RKVC_OK)
171 return err;
172
173 /* 解复用 PTS 与编码器 time_base 不同；统一用单调帧序号，避免 MP4 帧率错乱。 */
174 avf->pts = enc->next_pts;
175
176 if (force_idr)
177 avf->pict_type = AV_PICTURE_TYPE_I;
178
179 if (rois && roi_count > 0) {
180 int n = roi_count > RKVC_ROI_MAX ? RKVC_ROI_MAX : roi_count;
181 size_t bytes = (size_t)n * sizeof(AVRegionOfInterest);
182 AVFrameSideData *sd =
183 av_frame_new_side_data(avf, AV_FRAME_DATA_REGIONS_OF_INTEREST,
184 (int)bytes);
185 if (!sd) {
186 av_frame_free(&avf);
187 return RKVC_ERR_NOMEM;
188 }
189 AVRegionOfInterest *dst = (AVRegionOfInterest *)sd->data;
190 /* force_intra 不在 AVRegionOfInterest 内：用帧 metadata 传给 rkmppenc */
191 char intra_buf[RKVC_ROI_MAX * 2 + 1];
192 size_t ip = 0;
193 for (int i = 0; i < n; i++) {
194 int qp = rois[i].qp_offset;
195 if (qp > 51)
196 qp = 51;
197 if (qp < -51)
198 qp = -51;
199 dst[i].self_size = sizeof(AVRegionOfInterest);
200 dst[i].top = rois[i].y;
201 dst[i].left = rois[i].x;
202 dst[i].bottom = rois[i].y + rois[i].h;
203 dst[i].right = rois[i].x + rois[i].w;
204 /* 唯一换算点：整数 qp_offset → FFmpeg qoffset = qp/51 */
205 dst[i].qoffset = (AVRational){ qp, 51 };
206
207 if (ip + 2 < sizeof(intra_buf)) {
208 if (i)
209 intra_buf[ip++] = ',';
210 intra_buf[ip++] = rois[i].force_intra ? '1' : '0';
211 }
212 }
213 intra_buf[ip] = '\0';
214 if (av_dict_set(&avf->metadata, "rkvc_roi_force_intra", intra_buf, 0) < 0) {
215 av_frame_free(&avf);
216 return RKVC_ERR_NOMEM;
217 }
218 }
219
220 int ret = avcodec_send_frame(enc->ctx, avf);
221 av_frame_free(&avf);
222 if (ret == AVERROR(EAGAIN))
223 return RKVC_ERR_AGAIN;
224 if (ret >= 0)
225 enc->next_pts++;
226 return rkvc_from_averror(ret);
227}

229rkvc_err rkvc_mpp_enc_receive_packet(rkvc_mpp_enc *enc, rkvc_buffer **out)
230{
231 if (!enc || !enc->ctx || !out)
232 return RKVC_ERR_INVALID;
233
234 *out = NULL;
235 av_packet_unref(enc->pkt);
236 int ret = avcodec_receive_packet(enc->ctx, enc->pkt);
237 if (ret < 0) {
238 if (ret == AVERROR(EAGAIN))
239 return RKVC_ERR_AGAIN;
240 if (ret == AVERROR_EOF)
241 return RKVC_ERR_EOF;
242 return rkvc_from_averror(ret);
243 }
244
245 rkvc_buffer *b = rkvc_buffer_from_avpacket(enc->pkt);
246 if (!b)
247 return RKVC_ERR_NOMEM;
248
249 *out = b;
250 return RKVC_OK;
251}

253rkvc_err rkvc_mpp_enc_drain(rkvc_mpp_enc *enc)
254{
255 if (!enc || !enc->ctx)
256 return RKVC_ERR_INVALID;
257 if (enc->flushed)
258 return RKVC_OK;
259 int ret = avcodec_send_frame(enc->ctx, NULL);
260 if (ret == 0 || ret == AVERROR_EOF) {
261 enc->flushed = 1;
262 return RKVC_OK;
263 }
264 return rkvc_from_averror(ret);
265}
