17rkvc_err session_enc_size(const rkvc_session *s, int *ew, int *eh)
18{
19 int denom = s->desc.enc_scale_denom;
20 int w = s->desc.width;
21 int h = s->desc.height;
22 if (denom <= 1) {
23 *ew = w;
24 *eh = h;
25 return RKVC_OK;
26 }
27 *ew = session_align2(w / denom);
28 *eh = session_align2(h / denom);
29 /* 正分辨率再除 denom 后不得变成 0×0（align2 会把 1 收成 0）。 */
30 if (w > 0 && h > 0 && (*ew < 2 || *eh < 2))
31 return RKVC_ERR_INVALID;
32 return RKVC_OK;
33}

41rkvc_err session_downscale_for_encode(rkvc_session *s,
42 rkvc_buffer *frame,
43 rkvc_buffer **out)
44{
45 int ew = 0, eh = 0;
46 rkvc_err szerr = session_enc_size(s, &ew, &eh);
47 if (szerr != RKVC_OK)
48 return szerr;
49 *out = frame;
50
51 if ((int)frame->width == ew && (int)frame->height == eh)
52 return RKVC_OK;
53
54 rkvc_buffer *scaled = NULL;
55 rkvc_err err = rkvc_rga_scale_buffer(frame, &scaled, ew, eh,
56 s->desc.pixel_format,
57 RKVC_UPSCALE_BILINEAR);
58 if (err != RKVC_OK)
59 return err;
60 scaled->pts = frame->pts;
61 *out = scaled;
62 return RKVC_OK;
63}

153static int session_nv12_contiguous(const AVFrame *f)
154{
155 if (!f || !f->data[0] || f->format != AV_PIX_FMT_NV12 || !f->data[1])
156 return 0;
157 return (f->data[1] == f->data[0] + (ptrdiff_t)f->linesize[0] * f->height)
158 && (f->linesize[1] == f->linesize[0]);
159}

161rkvc_err session_write_nv12_frame(FILE *fp, const AVFrame *f)
162{
163 const int h = f->height;
164 const int ls = f->linesize[0];
165 const int us = f->linesize[1];
166
167 if (session_nv12_contiguous(f)) {
168 const size_t nbytes = (size_t)ls * (size_t)h +
169 (size_t)us * (size_t)(h / 2);
170 if (fwrite(f->data[0], 1, nbytes, fp) != nbytes)
171 return RKVC_ERR_IO;
172 return RKVC_OK;
173 }
174
175 for (int y = 0; y < h; y++)
176 if (fwrite(f->data[0] + y * ls, 1, (size_t)ls, fp) != (size_t)ls)
177 return RKVC_ERR_IO;
178 for (int y = 0; y < h / 2; y++)
179 if (fwrite(f->data[1] + y * us, 1, (size_t)us, fp) != (size_t)us)
180 return RKVC_ERR_IO;
181 return RKVC_OK;
182}

184rkvc_err session_write_nv12_buffer(FILE *fp, const rkvc_buffer *buf)
185{
186 if (!buf || !buf->av_frame)
187 return RKVC_ERR_INVALID;
188
189 const AVFrame *f = buf->av_frame;
190 rkvc_err err = rkvc_buffer_dmabuf_begin_cpu_read(buf);
191 if (err != RKVC_OK)
192 return err;
193
194 err = session_write_nv12_frame(fp, f);
195 rkvc_buffer_dmabuf_end_cpu_read(buf);
196 return err;
197}
