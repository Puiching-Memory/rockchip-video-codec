17static rkvc_err session_push_output(rkvc_session *s, rkvc_buffer *buf)
18{
19 rkvc_err perr = rkvc_port_push(&s->port_output, buf);
20 if (perr == RKVC_OK) {
21 pthread_mutex_lock(&s->lock);
22 s->stats.bytes_out += buf ? buf->size : 0;
23 pthread_mutex_unlock(&s->lock);
24 return RKVC_OK;
25 }
26 if (perr == RKVC_ERR_AGAIN) {
27 rkvc_session_stats_drop(s);
28 return RKVC_OK; /* 队列满：丢包记账，不中断管线 */
29 }
30 RKVC_LOG("output port push failed: %s", rkvc_err_str(perr));
31 return perr;
32}

93rkvc_err rkvc_decode_pump_next(rkvc_decode_pump *pump,
94 const rkvc_decode_pump_ops *ops,
95 void *opaque,
96 rkvc_buffer **frame)
97{
98 if (!pump || !ops || !ops->read_packet || !ops->send_packet ||
99 !ops->receive_frame || !ops->drain || !frame)
100 return RKVC_ERR_INVALID;
101
102 *frame = NULL;
103
104 if (!pump->pending_pkt && !pump->input_eof) {
105 rkvc_err err = ops->read_packet(opaque, &pump->pending_pkt);
106 if (err == RKVC_ERR_EOF) {
107 pump->input_eof = 1;
108 } else if (err != RKVC_OK) {
109 return err;
110 } else if (!pump->pending_pkt) {
111 return RKVC_ERR_INTERNAL;
112 }
113 }
114
115 int send_again = 0;
116 if (pump->pending_pkt) {
117 rkvc_err err = ops->send_packet(opaque, pump->pending_pkt);
118 if (err == RKVC_OK) {
119 rkvc_buffer_unref(pump->pending_pkt);
120 pump->pending_pkt = NULL;
121 } else if (err == RKVC_ERR_AGAIN) {
122 send_again = 1;
123 } else {
124 return err;
125 }
126 }
127
128 int drain_again = 0;
129 if (pump->input_eof && !pump->pending_pkt && !pump->drain_sent) {
130 rkvc_err err = ops->drain(opaque);
131 if (err == RKVC_OK) {
132 pump->drain_sent = 1;
133 } else if (err == RKVC_ERR_AGAIN) {
134 drain_again = 1;
135 } else {
136 return err;
137 }
138 }
139
140 rkvc_err err = ops->receive_frame(opaque, frame);
141 if (err == RKVC_OK)
142 return *frame ? RKVC_OK : RKVC_ERR_INTERNAL;
143 if (err == RKVC_ERR_EOF) {
144 if (pump->drain_sent)
145 return RKVC_ERR_EOF;
146 rkvc_log_print(AV_LOG_ERROR,
147 "decoder returned EOF before drain was accepted\n");
148 return RKVC_ERR_INTERNAL;
149 }
150 if (err != RKVC_ERR_AGAIN)
151 return err;
152
153 /* rkmpp is asynchronous and may transiently return EAGAIN from both ends.
154 Keep the pending input/drain state and avoid a tight retry loop. */
155 if (send_again || drain_again || pump->drain_sent)
156 av_usleep(100);
157 return RKVC_ERR_AGAIN;
158}

189static rkvc_err session_flush_encoder(rkvc_session *s)
190{
191 for (;;) {
192 rkvc_err err = session_drain_encoder(s);
193 if (err == RKVC_OK)
194 break;
195 if (err != RKVC_ERR_AGAIN)
196 return err;
197 err = drain_encoder_packets(s);
198 if (err != RKVC_OK)
199 return err;
200 av_usleep(100);
201 }
202
203 /* Flush 协议（FFmpeg avcodec / libsvtav1）：send_frame(NULL) 后须持续
204 receive_packet 直至真正 EOF。libsvtav1 的 eb_receive_packet 在收尾阶段对空
205 SVT 输出队列返回 EAGAIN（EB_NoErrorEmptyQueue）——此时 EOS 包尚未产出，若遇
206 EAGAIN 即 break 会丢尾包（preset 11 实测 90 帧入出帧不稳，39~54 抖动）。
207 故 EAGAIN 时退让重试，收到有效包后继续，直至 EOF。 */
208 for (;;) {
209 rkvc_buffer *pkt = NULL;
210 rkvc_err err = session_receive_packet(s, &pkt);
211 if (err == RKVC_ERR_EOF)
212 break;
213 if (err == RKVC_ERR_AGAIN) {
214 av_usleep(100);
215 continue;
216 }
217 if (err != RKVC_OK)
218 return err;
219 rkvc_err werr = session_write_packet(s, pkt);
220 rkvc_err perr = session_push_output(s, pkt);
221 rkvc_session_stats_tick(s, 1);
222 rkvc_buffer_unref(pkt);
223 if (werr != RKVC_OK)
224 return werr;
225 if (perr != RKVC_OK)
226 return perr;
227 }
228 return RKVC_OK;
229}

255static rkvc_err encode_one_frame(rkvc_session *s, rkvc_buffer *frame)
256{
257 /* MPP 硬 ROI：session ROI → AVFrame side data → rkmppenc → KEY_ROI_DATA。
258 * SVT 无硬 ROI，忽略 set_roi（不做像素 fallback）。 */
259 rkvc_roi_rect rois[RKVC_ROI_MAX];
260 int roi_n = 0;
261 int force_idr = 0;
262
263 rkvc_err rerr = rkvc_session_apply_reconfig(s, &force_idr);
264 if (rerr != RKVC_OK)
265 return rerr;
266
267 pthread_mutex_lock(&s->lock);
268 roi_n = s->roi_count;
269 if (roi_n > 0)
270 memcpy(rois, s->rois, (size_t)roi_n * sizeof(rois[0]));
271 pthread_mutex_unlock(&s->lock);
272
273 for (;;) {
274 rkvc_err err;
275#ifdef RKVC_ENABLE_MLVC
276 if (s->mlvc_enc) {
277 err = rkvc_mlvc_enc_send_frame(s->mlvc_enc, frame);
278 } else
279#endif
280 if (s->enc) {
281 err = rkvc_mpp_enc_send_frame_roi_ex(
282 s->enc, frame, roi_n > 0 ? rois : NULL, roi_n, force_idr);
283 } else {
284 err = rkvc_svt_enc_send_frame(s->svt, frame);
285 }
286 if (err == RKVC_OK)
287 return drain_encoder_packets(s);
288 if (err == RKVC_ERR_AGAIN) {
289 err = drain_encoder_packets(s);
290 if (err != RKVC_OK)
291 return err;
292 av_usleep(100);
293 continue;
294 }
295 return err;
296 }
297}

299static rkvc_err transcode_loop(rkvc_session *s)
300{
301 rkvc_err err = RKVC_OK;
302 rkvc_decode_pump pump = {0};
303
304 while (!s->stop_requested) {
305 rkvc_buffer *frame = NULL;
306 err = rkvc_decode_pump_next(&pump, &session_decode_pump_ops, s, &frame);
307 if (err == RKVC_ERR_AGAIN) {
308 if (s->svt) {
309 err = drain_encoder_packets(s);
310 if (err != RKVC_OK)
311 break;
312 }
313 continue;
314 }
315 if (err == RKVC_ERR_EOF) {
316 err = RKVC_OK;
317 break;
318 }
319 if (err != RKVC_OK)
320 break;
321
322 rkvc_session_stats_frame_in(s);
323
324 rkvc_buffer *enc_frame = NULL;
325 err = session_downscale_for_encode(s, frame, &enc_frame);
326 if (err != RKVC_OK) {
327 rkvc_buffer_unref(frame);
328 break;
329 }
330
331 err = encode_one_frame(s, enc_frame);
332 if (enc_frame != frame)
333 rkvc_buffer_unref(enc_frame);
334 rkvc_buffer_unref(frame);
335 if (err != RKVC_OK)
336 break;
337 }
338
339 rkvc_decode_pump_cleanup(&pump);
340 if (err != RKVC_OK && err != RKVC_ERR_EOF)
341 return err;
342 return session_flush_encoder(s);
343}

345static rkvc_err session_write_display(rkvc_session *s, FILE *fp,
346 rkvc_buffer *display)
347{
348 if (!display || !display->av_frame)
349 return RKVC_OK;
350
351 rkvc_buffer *write_buf = display;
352 rkvc_buffer *cpu = NULL;
353 rkvc_err err = RKVC_OK;
354
355 if (display->mem_type == RKVC_MEM_DMABUF &&
356 (!display->av_frame->data[0] ||
357 display->av_frame->format == AV_PIX_FMT_DRM_PRIME)) {
358 err = rkvc_dma_to_host(display, &cpu);
359 if (err == RKVC_OK)
360 write_buf = cpu;
361 }
362 if (err == RKVC_OK && write_buf->av_frame) {
363 if (write_buf->mem_type == RKVC_MEM_DMABUF)
364 err = session_write_nv12_buffer(fp, write_buf);
365 else
366 err = session_write_nv12_frame(fp, write_buf->av_frame);
367 }
368 rkvc_buffer_unref(cpu);
369 if (err == RKVC_OK) {
370 err = session_push_output(s, display);
371 if (err == RKVC_OK)
372 rkvc_session_stats_tick(s, 1);
373 }
374 return err;
375}

377static rkvc_err session_decode_one_frame(rkvc_session *s,
378 rkvc_decode_pump *pump,
379 rkvc_buffer **frame_out,
380 double *decode_sec)
381{
382 const double t0 = session_now_sec();
383 rkvc_err err = RKVC_OK;
384
385 rkvc_buffer *frame = NULL;
386 err = rkvc_decode_pump_next(pump, &session_decode_pump_ops, s, &frame);
387 if (err == RKVC_ERR_AGAIN) {
388 *decode_sec = session_now_sec() - t0;
389 return RKVC_ERR_AGAIN;
390 }
391 if (err == RKVC_ERR_EOF || err != RKVC_OK) {
392 *decode_sec = session_now_sec() - t0;
393 return err;
394 }
395
396 rkvc_buffer *host = NULL;
397 err = session_frame_to_host(s, frame, &host);
398 *decode_sec = session_now_sec() - t0;
399 rkvc_buffer_unref(frame);
400 if (err != RKVC_OK) {
401 rkvc_buffer_unref(host);
402 return err;
403 }
404
405 *frame_out = host;
406 return RKVC_OK;
407}

409static rkvc_err decode_loop_ai_sr(rkvc_session *s, FILE *fp)
410{
411 rkvc_err err = session_ensure_rknn_sr(s);
412 if (err != RKVC_OK)
413 return err;
414
415 rkvc_decode_pump pump = {0};
416 int has_pending = 0;
417
418 while (!s->stop_requested) {
419 rkvc_buffer *host = NULL;
420 double decode_sec = 0.0;
421 err = session_decode_one_frame(s, &pump, &host, &decode_sec);
422 if (err == RKVC_ERR_AGAIN) {
423 continue;
424 }
425 if (err == RKVC_ERR_EOF) {
426 err = RKVC_OK;
427 break;
428 }
429 if (err != RKVC_OK)
430 goto out;
431
432 rkvc_session_stats_add_timing(s, decode_sec, 0, 0, 0);
433
434 if (has_pending) {
435 rkvc_buffer *display = NULL;
436 const double t_rga0 = session_now_sec();
437 err = rkvc_rknn_sr_ctx_collect(s->rknn_sr, &display, 1);
438 const double t_rga1 = session_now_sec();
439 double wr_delta = 0.0;
440 double pp_delta = 0.0;
441 if (err == RKVC_OK) {
442 const double t_wr0 = session_now_sec();
443 err = session_write_display(s, fp, display);
444 const double t_wr1 = session_now_sec();
445 wr_delta = t_wr1 - t_wr0;
446 pp_delta = t_wr1 - t_rga0;
447 }
448 rkvc_session_stats_add_timing(s, 0, t_rga1 - t_rga0, wr_delta, pp_delta);
449 rkvc_buffer_unref(display);
450 if (err != RKVC_OK) {
451 rkvc_buffer_unref(host);
452 goto out;
453 }
454 has_pending = 0;
455 }
456
457 const double t_rga0 = session_now_sec();
458 err = rkvc_rknn_sr_ctx_submit(s->rknn_sr, host);
459 const double t_rga1 = session_now_sec();
460 rkvc_session_stats_add_timing(s, 0, t_rga1 - t_rga0, 0, 0);
461 rkvc_buffer_unref(host);
462 if (err != RKVC_OK)
463 goto out;
464
465 has_pending = 1;
466 }
467
468 if (has_pending) {
469 rkvc_buffer *display = NULL;
470 const double t_rga0 = session_now_sec();
471 err = rkvc_rknn_sr_ctx_collect(s->rknn_sr, &display, 1);
472 const double t_rga1 = session_now_sec();
473 double wr_delta = 0.0;
474 double pp_delta = 0.0;
475 if (err == RKVC_OK) {
476 const double t_wr0 = session_now_sec();
477 err = session_write_display(s, fp, display);
478 const double t_wr1 = session_now_sec();
479 wr_delta = t_wr1 - t_wr0;
480 pp_delta = t_wr1 - t_rga0;
481 }
482 rkvc_session_stats_add_timing(s, 0, t_rga1 - t_rga0, wr_delta, pp_delta);
483 rkvc_buffer_unref(display);
484 }
485
486out:
487 rkvc_decode_pump_cleanup(&pump);
488 return err;
489}

491static rkvc_err decode_loop(rkvc_session *s)
492{
493 if (!s->desc.output_path)
494 return RKVC_ERR_INVALID;
495
496 FILE *fp = fopen(s->desc.output_path, "wb");
497 if (!fp)
498 return RKVC_ERR_IO;
499
500 rkvc_err err = RKVC_OK;
501
502 rkvc_session_stats_reset_timing(s);
503
504 if (session_needs_post_upscale(s) && session_uses_ai_sr(s)) {
505 err = decode_loop_ai_sr(s, fp);
506 fclose(fp);
507 return err;
508 }
509
510 rkvc_decode_pump pump = {0};
511 while (!s->stop_requested) {
512 const double t_dec0 = session_now_sec();
513 rkvc_buffer *frame = NULL;
514 err = rkvc_decode_pump_next(&pump, &session_decode_pump_ops, s,
515 &frame);
516 if (err == RKVC_ERR_AGAIN)
517 continue;
518 if (err == RKVC_ERR_EOF) {
519 err = RKVC_OK;
520 break;
521 }
522 if (err != RKVC_OK)
523 break;
524
525 rkvc_buffer *host = NULL;
526 err = session_frame_to_host(s, frame, &host);
527 const double t_dec1 = session_now_sec();
528
529 rkvc_buffer *display = NULL;
530 const double t_rga0 = session_now_sec();
531 if (err == RKVC_OK && host)
532 err = session_apply_post_upscale(s, host, &display);
533 const double t_rga1 = session_now_sec();
534
535 double t_wr1 = t_rga1;
536 double wr_delta = 0.0;
537 if (err == RKVC_OK && display && display->av_frame) {
538 const double t_wr0 = session_now_sec();
539 err = session_write_display(s, fp, display);
540 t_wr1 = session_now_sec();
541 wr_delta = t_wr1 - t_wr0;
542 }
543
544 rkvc_session_stats_add_timing(s, t_dec1 - t_dec0,
545 t_rga1 - t_rga0, wr_delta,
546 t_wr1 - t_rga0);
547
548 if (display != host)
549 rkvc_buffer_unref(display);
550 rkvc_buffer_unref(host);
551 rkvc_buffer_unref(frame);
552
553 if (err != RKVC_OK)
554 break;
555 }
556
557 rkvc_decode_pump_cleanup(&pump);
558 fclose(fp);
559 return err;
560}

562static rkvc_err load_raw_frame(FILE *fp, rkvc_buffer *frame, int w, int h,
563 rkvc_pix_fmt fmt)
564{
565 const size_t y_size = (size_t)w * (size_t)h;
566 size_t frame_bytes = y_size + y_size / 2;
567 uint8_t *raw = rkvc_malloc(frame_bytes);
568 if (!raw)
569 return RKVC_ERR_NOMEM;
570
571 if (fread(raw, 1, frame_bytes, fp) != frame_bytes) {
572 rkvc_free(raw);
573 return RKVC_ERR_EOF;
574 }
575
576 uint8_t *planes[4] = {0};
577 int strides[4] = {0};
578 rkvc_buffer_get_video_planes(frame, planes, strides);
579
580 for (int y = 0; y < h; y++)
581 memcpy(planes[0] + y * strides[0],
582 raw + (size_t)y * (size_t)w, (size_t)w);
583
584 if (fmt == RKVC_PIX_FMT_YUV420P) {
585 const size_t c_w = (size_t)w / 2;
586 const size_t c_h = (size_t)h / 2;
587 const uint8_t *u_src = raw + y_size;
588 const uint8_t *v_src = u_src + c_w * c_h;
589 for (int y = 0; y < (int)c_h; y++) {
590 memcpy(planes[1] + y * strides[1], u_src + y * c_w, c_w);
591 memcpy(planes[2] + y * strides[2], v_src + y * c_w, c_w);
592 }
593 } else {
594 for (int y = 0; y < h / 2; y++)
595 memcpy(planes[1] + y * strides[1],
596 raw + y_size + (size_t)y * (size_t)w, (size_t)w);
597 }
598
599 rkvc_free(raw);
600 return RKVC_OK;
601}

603static rkvc_err live_capture_loop(rkvc_session *s)
604{
605 if (!s->v4l2)
606 return RKVC_ERR_INVALID;
607
608 rkvc_err err = RKVC_OK;
609 int frames = 0;
610 const int max_frames = s->desc.capture_max_frames;
611 const int timeout_ms = s->desc.capture_timeout_ms > 0
612 ? s->desc.capture_timeout_ms
613 : 1000;
614
615 while (!s->stop_requested) {
616 if (max_frames > 0 && frames >= max_frames)
617 break;
618
619 rkvc_buffer *frame = NULL;
620 err = rkvc_v4l2_read_frame(s->v4l2, &frame, timeout_ms);
621 if (err == RKVC_ERR_AGAIN)
622 continue;
623 if (err != RKVC_OK)
624 return err;
625
626 if (rkvc_port_push(&s->port_capture, frame) == RKVC_ERR_AGAIN) {
627 rkvc_buffer *drop = NULL;
628 if (rkvc_port_pull(&s->port_capture, &drop, 0) == RKVC_OK)
629 rkvc_buffer_unref(drop);
630 (void)rkvc_port_push(&s->port_capture, frame);
631 }
632 /* preview：同帧侧抽，满则丢最旧（低延迟预览） */
633 if (rkvc_port_push(&s->port_preview, frame) == RKVC_ERR_AGAIN) {
634 rkvc_buffer *drop = NULL;
635 if (rkvc_port_pull(&s->port_preview, &drop, 0) == RKVC_OK)
636 rkvc_buffer_unref(drop);
637 (void)rkvc_port_push(&s->port_preview, frame);
638 }

641
642 rkvc_buffer *enc_frame = NULL;

647 }
648

651 rkvc_buffer_unref(enc_frame);
652 rkvc_buffer_unref(frame);
653 if (err != RKVC_OK)
654 return err;
655 }

660static rkvc_err encode_file_loop(rkvc_session *s)
661{
662 if (!s->desc.input_path)
663 return RKVC_ERR_INVALID;
664
665 FILE *fp = fopen(s->desc.input_path, "rb");
666 if (!fp)
667 return RKVC_ERR_IO;
668
669 const int w = s->desc.width;
670 const int h = s->desc.height;
671 rkvc_err err = RKVC_OK;
672 rkvc_err loop_err = RKVC_OK;
673 int64_t pts = 0;
674
675 while (!s->stop_requested) {
676 rkvc_buffer *frame = NULL;
677 err = rkvc_buffer_alloc_video_host(&frame, w, h, s->desc.pixel_format);
678 if (err != RKVC_OK)
679 break;
680
681 err = load_raw_frame(fp, frame, w, h, s->desc.pixel_format);
682 if (err == RKVC_ERR_EOF) {
683 rkvc_buffer_unref(frame);
684 err = RKVC_OK;
685 break;
686 }
687 if (err != RKVC_OK) {
688 rkvc_buffer_unref(frame);
689 break;
690 }
691
692 rkvc_buffer_set_pts(frame, pts++);
693 rkvc_session_stats_frame_in(s);
694
695 rkvc_buffer *enc_frame = NULL;
696 err = session_downscale_for_encode(s, frame, &enc_frame);
697 if (err != RKVC_OK) {
698 rkvc_buffer_unref(frame);
699 break;
700 }
701
702 err = encode_one_frame(s, enc_frame);
703 if (enc_frame != frame)
704 rkvc_buffer_unref(enc_frame);
705 rkvc_buffer_unref(frame);
706 if (err != RKVC_OK) {
707 loop_err = err;
708 break;
709 }
710 }
711
712 fclose(fp);
713
714 if (loop_err != RKVC_OK)
715 return loop_err;
716
717 return session_flush_encoder(s);
718}

720rkvc_err rkvc_session_run_file(rkvc_session *session)
721{
722 if (!session)
723 return RKVC_ERR_INVALID;
724
725 rkvc_err err = rkvc_session_start(session);
726 if (err != RKVC_OK)
727 return err;
728
729 switch (session->desc.template_id) {
730 case RKVC_TEMPLATE_FILE_TRANSCODE:
731 case RKVC_TEMPLATE_AV1_STORAGE:
732 case RKVC_TEMPLATE_MLVC_STORAGE:
733 err = transcode_loop(session);
734 break;
735 case RKVC_TEMPLATE_LIVE_CAPTURE:
736 err = live_capture_loop(session);
737 break;
738 case RKVC_TEMPLATE_FILE_DECODE:
739 err = decode_loop(session);
740 break;
741 case RKVC_TEMPLATE_FILE_ENCODE:
742 err = encode_file_loop(session);
743 break;
744 default:
745 err = RKVC_ERR_INVALID;
746 }
747
748 rkvc_session_stop(session);
749 return err;
750}
