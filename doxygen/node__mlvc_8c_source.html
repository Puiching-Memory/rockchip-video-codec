66static rkvc_err load_pmf(mlvc_pmf *p, const char *path)
67{
68 FILE *f = NULL;
69 rkvc_err err = RKVC_ERR_FORMAT;
70
71 memset(p, 0, sizeof(*p));
72 f = fopen(path, "rb");
73 if (!f)
74 return RKVC_ERR_IO;
75
76 char magic[4];
77 if (fread(magic, 1, 4, f) != 4 || memcmp(magic, "PMF1", 4) != 0)
78 goto fail;
79
80 uint32_t nL, nO, nT;
81 if (fread(&nL, 4, 1, f) != 1 || fread(&nO, 4, 1, f) != 1 ||
82 fread(&nT, 4, 1, f) != 1)
83 goto fail;
84 if (nL == 0 || nL > MLVC_PMF_MAX_LEN ||
85 nO == 0 || nO > MLVC_PMF_MAX_LEN ||
86 nT == 0 || nT > MLVC_PMF_MAX_TAB)
87 goto fail;
88
89 p->num_lengths = nL;
90 p->num_offsets = nO;
91 p->num_table = nT;
92 p->lengths = rkvc_malloc((size_t)nL * 4u);
93 p->offsets = rkvc_malloc((size_t)nO * 4u);
94 p->table = rkvc_malloc((size_t)nT * 4u);
95 if (!p->lengths || !p->offsets || !p->table) {
96 err = RKVC_ERR_NOMEM;
97 goto fail;
98 }
99 if (fread(p->lengths, 4, nL, f) != nL ||
100 fread(p->offsets, 4, nO, f) != nO ||
101 fread(p->table, 4, nT, f) != nT)
102 goto fail;
103
104 uint32_t tag = 0;
105 if (fread(&tag, 4, 1, f) != 1)
106 goto fail;
107 if (tag == 1) {
108 if (fread(&p->scale_min, 8, 1, f) != 1 ||
109 fread(&p->scale_max, 8, 1, f) != 1 ||
110 fread(&p->scale_levels, 4, 1, f) != 1 ||
111 fread(&p->index_space, 4, 1, f) != 1)
112 goto fail;
113 } else if (tag == 2) {
114 if (fread(&p->qp_num, 4, 1, f) != 1 ||
115 fread(&p->channels, 4, 1, f) != 1)
116 goto fail;
117 } else {
118 goto fail;
119 }
120 fclose(f);
121 return RKVC_OK;
122
123fail:
124 fclose(f);
125 free_pmf(p);
126 return err;
127}

155static void rknn_model_cleanup(mlvc_rknn_model *m)
156{
157 if (!m)
158 return;
159 uint32_t ni = m->io_num.n_input;
160 uint32_t no = m->io_num.n_output;
161 if (ni > MLVC_MAX_IO)
162 ni = MLVC_MAX_IO;
163 if (no > MLVC_MAX_IO)
164 no = MLVC_MAX_IO;
165 for (uint32_t i = 0; i < ni; i++)
166 if (m->in_mem[i]) rknn_destroy_mem(m->ctx, m->in_mem[i]);
167 for (uint32_t i = 0; i < no; i++)
168 if (m->out_mem[i]) rknn_destroy_mem(m->ctx, m->out_mem[i]);
169 if (m->ctx) rknn_destroy(m->ctx);
170 rkvc_free(m->model_buf);
171 memset(m, 0, sizeof(*m));
172}

174static rkvc_err rknn_model_init(mlvc_rknn_model *m, const char *path,
175 const char *patch_path, int expected_qp)
176{
177 memset(m, 0, sizeof(*m));
178
179 FILE *fp = fopen(path, "rb");
180 if (!fp)
181 return RKVC_ERR_IO;
182 if (fseek(fp, 0, SEEK_END) != 0) {
183 fclose(fp);
184 return RKVC_ERR_IO;
185 }
186 long sz = ftell(fp);
187 if (sz <= 0 || (size_t)sz > MLVC_MAX_MODEL_BYTES) {
188 fclose(fp);
189 return RKVC_ERR_IO;
190 }
191 if (fseek(fp, 0, SEEK_SET) != 0) {
192 fclose(fp);
193 return RKVC_ERR_IO;
194 }
195 m->model_buf = rkvc_malloc((size_t)sz);
196 if (!m->model_buf) {
197 fclose(fp);
198 return RKVC_ERR_NOMEM;
199 }
200 if (fread(m->model_buf, 1, (size_t)sz, fp) != (size_t)sz) {
201 fclose(fp);
202 rknn_model_cleanup(m);
203 return RKVC_ERR_IO;
204 }
205 fclose(fp);
206 m->model_size = (size_t)sz;
207
208 if (patch_path && patch_path[0]) {
209 rkvc_err perr = rkvc_qppatch_apply_file((uint8_t *)m->model_buf,
210 m->model_size, patch_path,
211 expected_qp);
212 if (perr) {
213 RKVC_LOG("MLVC qp patch apply failed: %s", patch_path);
214 rknn_model_cleanup(m);
215 return perr;
216 }
217 RKVC_LOG("MLVC applied qp patch %s", patch_path);
218 }
219
220 /* 分发构建：在 rknn_init 前强制 librknnrt 静默，避免 RKNN_LOG_LEVEL
221 * 泄露已解密模型的网络结构（见 rkvc_rknn_quiet_runtime 说明）。 */
222 rkvc_rknn_quiet_runtime();
223 int rc = rknn_init(&m->ctx, m->model_buf, (uint32_t)sz, 0, NULL);
224 if (rc != RKNN_SUCC) {
225 RKVC_LOG("RKNN error %d at %s:%d", rc, __FILE__, __LINE__);
226 rknn_model_cleanup(m);
227 return RKVC_ERR_HW;
228 }
229 /*
230 * 多核 NPU（如 RK3588 三核）显式启用全部核心：默认 AUTO 仅单核调度，
231 * 显式 core_mask 让运行时在核间分配算子，显著降低编码推理延迟
232 *（RK3588 受控实测编码 +24%）。核心数来自平台探测，并由驱动自适应
233 * 校正（rkvc_rknn_apply_npu_cores 向下尝试）；单核平台（如 RV1126B）
234 * 保持默认单核行为，不调用 set_core_mask。
235 */
236 {
237 const rkvc_platform_info *pi = rkvc_platform_probe();
238 if (pi->has_npu && pi->npu_cores > 1)
239 rkvc_rknn_apply_npu_cores(m->ctx, pi->npu_cores);
240 }
241 rc = rknn_query(m->ctx, RKNN_QUERY_IN_OUT_NUM, &m->io_num, sizeof(m->io_num));
242 if (rc != RKNN_SUCC) {
243 RKVC_LOG("RKNN error %d at %s:%d", rc, __FILE__, __LINE__);
244 rknn_model_cleanup(m);
245 return RKVC_ERR_HW;
246 }
247 if (m->io_num.n_input > MLVC_MAX_IO || m->io_num.n_output > MLVC_MAX_IO) {
248 RKVC_LOG("RKNN I/O count %u/%u exceeds %d",
249 m->io_num.n_input, m->io_num.n_output, MLVC_MAX_IO);
250 rknn_model_cleanup(m);
251 return RKVC_ERR_FORMAT;
252 }
253
254 for (uint32_t i = 0; i < m->io_num.n_input; i++) {
255 m->in_attr[i].index = i;
256 rc = rknn_query(m->ctx, RKNN_QUERY_INPUT_ATTR, &m->in_attr[i],
257 sizeof(m->in_attr[i]));
258 if (rc != RKNN_SUCC) {
259 RKVC_LOG("RKNN error %d at %s:%d", rc, __FILE__, __LINE__);
260 rknn_model_cleanup(m);
261 return RKVC_ERR_HW;
262 }
263 }
264 for (uint32_t i = 0; i < m->io_num.n_output; i++) {
265 m->out_attr[i].index = i;
266 rc = rknn_query(m->ctx, RKNN_QUERY_OUTPUT_ATTR, &m->out_attr[i],
267 sizeof(m->out_attr[i]));
268 if (rc != RKNN_SUCC) {
269 RKVC_LOG("RKNN error %d at %s:%d", rc, __FILE__, __LINE__);
270 rknn_model_cleanup(m);
271 return RKVC_ERR_HW;
272 }
273 }
274 return RKVC_OK;
275}

291static rkvc_err rknn_alloc_input_mems(mlvc_rknn_model *m)
292{
293 for (uint32_t i = 0; i < m->io_num.n_input; i++) {
294 rknn_tensor_attr a = m->in_attr[i];
295 a.index = i;
296 a.type = RKNN_TENSOR_FLOAT16;
297 a.fmt = RKNN_TENSOR_NHWC;
298 a.pass_through = 1;
299 a.size = a.size_with_stride;
300 m->in_mem[i] = rknn_create_mem(m->ctx, a.size_with_stride);
301 if (!m->in_mem[i]) return RKVC_ERR_HW;
302 MLVC_RKNN_CHECK(rknn_set_io_mem(m->ctx, m->in_mem[i], &a));
303 }
304 return RKVC_OK;
305}

307static rkvc_err rknn_alloc_output_mems(mlvc_rknn_model *m)
308{
309 for (uint32_t i = 0; i < m->io_num.n_output; i++) {
310 rknn_tensor_attr a;
311 memset(&a, 0, sizeof(a));
312 a.index = i;
313 MLVC_RKNN_CHECK(rknn_query(m->ctx, RKNN_QUERY_NATIVE_NC1HWC2_OUTPUT_ATTR, &a, sizeof(a)));
314 a.index = i;
315 a.pass_through = 1;
316 m->native_out_attr[i] = a;
317 m->out_mem[i] = rknn_create_mem(m->ctx, a.size_with_stride);
318 if (!m->out_mem[i]) return RKVC_ERR_HW;
319 MLVC_RKNN_CHECK(rknn_set_io_mem(m->ctx, m->out_mem[i], &a));
320 }
321 return RKVC_OK;

332static void rknn_write_input(mlvc_rknn_model *m, int i, const uint16_t *data)
333{
334 if (!m || i < 0 || (uint32_t)i >= m->io_num.n_input || !m->in_mem[i] || !data)
335 return;
336 rknn_tensor_attr *a = &m->in_attr[i];
337 if (a->w_stride == 0 || a->w_stride == (uint32_t)a->dims[2]) {
338 memcpy(m->in_mem[i]->virt_addr, data, (size_t)a->n_elems * sizeof(uint16_t));
339 } else {
340 int H = a->dims[1], W = a->dims[2], C = a->dims[3];
341 uint16_t *dst = (uint16_t *)m->in_mem[i]->virt_addr;
342 for (int h = 0; h < H; h++)
343 memcpy(dst + (size_t)h * a->w_stride * C,
344 data + (size_t)h * W * C,
345 (size_t)W * C * sizeof(uint16_t));
346 }
347 rknn_mem_sync(m->ctx, m->in_mem[i], RKNN_MEMORY_SYNC_TO_DEVICE);
348}

424static rkvc_err enc_resolve_geom(rkvc_mlvc_enc *e)
425{
426 mlvc_rknn_model *enc = &e->enc_model;
427 e->enc_x_in = rknn_find_input(enc, "x");
428 if (e->enc_x_in < 0)
429 e->enc_x_in = rknn_find_input(enc, "New_input_x");
430 e->enc_ref_in = rknn_find_input(enc, "ref_feature");
431 if (e->enc_x_in < 0 || e->enc_ref_in < 0)
432 return RKVC_ERR_FORMAT;
433
434 e->IMG_H = enc->in_attr[e->enc_x_in].dims[1];
435 e->IMG_W = enc->in_attr[e->enc_x_in].dims[2];
436 e->REF_H = enc->in_attr[e->enc_ref_in].dims[1];
437 e->REF_W = enc->in_attr[e->enc_ref_in].dims[2];
438 e->REF_C = enc->in_attr[e->enc_ref_in].dims[3];
439
440 e->enc_feat_out = rknn_find_output(enc, "feature");
441 e->enc_z_out = rknn_find_output(enc, "z_raw");
442 e->enc_y0_out = rknn_find_output(enc, "y_raw_0");
443 e->enc_y1_out = rknn_find_output(enc, "y_raw_1");
444 if (e->enc_feat_out < 0 || e->enc_z_out < 0 ||
445 e->enc_y0_out < 0 || e->enc_y1_out < 0)
446 return RKVC_ERR_FORMAT;
447
448 int zC1 = enc->native_out_attr[e->enc_z_out].dims[1];
449 e->ZC = zC1 * 8;
450 e->ZH = enc->native_out_attr[e->enc_z_out].dims[2];
451 e->ZW = enc->native_out_attr[e->enc_z_out].dims[3];
452
453 int yC1 = enc->native_out_attr[e->enc_y0_out].dims[1];
454 e->YC = yC1 * 8;
455 e->YH = enc->native_out_attr[e->enc_y0_out].dims[2];
456 e->YW = enc->native_out_attr[e->enc_y0_out].dims[3];
457
458 return RKVC_OK;
459}

461static rkvc_err enc_alloc_bufs(rkvc_mlvc_enc *e)
462{
463 size_t img_sz = (size_t)e->IMG_H * e->IMG_W;
464 size_t ref_sz = (size_t)e->REF_C * e->REF_H * e->REF_W;
465 size_t z_sz = (size_t)e->ZC * e->ZH * e->ZW;
466 size_t y_sz = (size_t)e->YC * e->YH * e->YW;
467
468 e->x_nhwc = rkvc_malloc(img_sz * 3 * 2);
469 e->ref_nhwc = rkvc_malloc(ref_sz * 2);
470 e->z_r = rkvc_malloc(z_sz * 4);
471 e->y0_r = rkvc_malloc(y_sz * 4);
472 e->y1_r = rkvc_malloc(y_sz * 4);
473 e->s0 = rkvc_malloc(y_sz * 4);
474 e->s1 = rkvc_malloc(y_sz * 4);
475 e->z_idx = rkvc_malloc(z_sz * 4);
476 if (!e->x_nhwc || !e->ref_nhwc || !e->z_r || !e->y0_r || !e->y1_r ||
477 !e->s0 || !e->s1 || !e->z_idx)
478 return RKVC_ERR_NOMEM;
479
480 memset(e->ref_nhwc, 0, ref_sz * 2);
481 /* z_idx 在通道平面内为常量：按平面填充（原逐元素三重循环）*/
482 for (int c = 0; c < e->ZC; c++) {
483 int32_t *p = e->z_idx + (size_t)c * e->ZH * e->ZW;
484 size_t n = (size_t)e->ZH * e->ZW;
485 for (size_t i = 0; i < n; i++)
486 p[i] = e->qp * e->ZC + c;
487 }
488 return RKVC_OK;
489}

503rkvc_err rkvc_mlvc_enc_open(rkvc_mlvc_enc **out, const rkvc_mlvc_enc_config *cfg)
504{
505 if (!out || !cfg || !cfg->enc_model_path ||
506 !cfg->gaussian_pmf_path || !cfg->bitest_pmf_path)
507 return RKVC_ERR_INVALID;
508 *out = NULL;
509
510 rkvc_mlvc_enc *e = rkvc_calloc(1, sizeof(*e));
511 if (!e) return RKVC_ERR_NOMEM;
512 e->qp = cfg->qp > 0 ? cfg->qp : 21;
513
514 /* PMF 表 */
515 mlvc_pmf gpmf = {0}, bpmf = {0};
516 rkvc_err err = load_pmf(&gpmf, cfg->gaussian_pmf_path);
517 if (err) { rkvc_free(e); return err; }
518 err = load_pmf(&bpmf, cfg->bitest_pmf_path);
519 if (err) { free_pmf(&gpmf); rkvc_free(e); return err; }
520
521 if (!gpmf.index_space) {
522 RKVC_LOG("gaussian.bin: expect index_space=True");
523 free_pmf(&gpmf); free_pmf(&bpmf); rkvc_free(e);
524 return RKVC_ERR_FORMAT;
525 }
526
527 /* 熵编解码器 */
528 int rc = rkvc_rans_coder_init(&e->g_coder, RKVC_RANS_BYTE,
529 gpmf.lengths, gpmf.num_lengths,
530 gpmf.offsets, gpmf.num_offsets,
531 gpmf.table, gpmf.num_table, 16, 2);
532 if (rc) { free_pmf(&gpmf); free_pmf(&bpmf); rkvc_free(e); return RKVC_ERR_INTERNAL; }
533 rc = rkvc_rans_coder_init(&e->b_coder, RKVC_RANS_BYTE,
534 bpmf.lengths, bpmf.num_lengths,
535 bpmf.offsets, bpmf.num_offsets,
536 bpmf.table, bpmf.num_table, 16, 2);
537 if (rc) { rkvc_rans_coder_free(&e->g_coder); free_pmf(&gpmf); free_pmf(&bpmf); rkvc_free(e); return RKVC_ERR_INTERNAL; }
538
539 free_pmf(&gpmf);
540 free_pmf(&bpmf);
541
542 /* 模型 */
543 err = rknn_model_init(&e->enc_model, cfg->enc_model_path,
544 cfg->qp_patch_path, e->qp);
545 if (err) { rkvc_rans_coder_free(&e->g_coder); rkvc_rans_coder_free(&e->b_coder); rkvc_free(e); return err; }
546
547 err = rknn_alloc_input_mems(&e->enc_model);
548 if (err) goto fail_model;
549 err = rknn_alloc_output_mems(&e->enc_model);
550 if (err) goto fail_model;
551
552 err = enc_resolve_geom(e);
553 if (err) goto fail_model;
554 err = enc_alloc_bufs(e);
555 if (err) goto fail_model;
556
557 *out = e;
558 return RKVC_OK;
559
560fail_model:
561 enc_free_bufs(e);
562 rknn_model_cleanup(&e->enc_model);
563 rkvc_rans_coder_free(&e->g_coder);
564 rkvc_rans_coder_free(&e->b_coder);
565 rkvc_free(e);
566 return err;
567}

583rkvc_err rkvc_mlvc_enc_send_frame(rkvc_mlvc_enc *e, rkvc_buffer *frame)
584{
585 if (!e) return RKVC_ERR_INVALID;
586 if (e->flushed) return RKVC_ERR_EOF;
587 if (!frame) return rkvc_mlvc_enc_drain(e);
588 if (e->pending_pkt) return RKVC_ERR_AGAIN;
589
590 if (frame->width != (uint32_t)e->IMG_W || frame->height != (uint32_t)e->IMG_H) {
591 RKVC_LOG("frame size mismatch: model expects %dx%d", e->IMG_W, e->IMG_H);
592 return RKVC_ERR_FORMAT;
593 }
594
595 yuv_to_nhwc_fp16(frame, e->x_nhwc);
596
597 /* encoder NPU */
598 rknn_write_input(&e->enc_model, e->enc_x_in, e->x_nhwc);
599 rknn_write_input(&e->enc_model, e->enc_ref_in, e->ref_nhwc);
600 MLVC_RKNN_CHECK(rknn_run(e->enc_model.ctx, NULL));
601
602 const uint16_t *bz = rknn_output_data(&e->enc_model, e->enc_z_out);
603 const uint16_t *by0 = rknn_output_data(&e->enc_model, e->enc_y0_out);
604 const uint16_t *by1 = rknn_output_data(&e->enc_model, e->enc_y1_out);
605 if (!bz || !by0 || !by1)
606 return RKVC_ERR_HW;
607
608 mlvc_px_nc1hwc2_to_nchw(bz, e->z_r, e->ZC, e->ZH, e->ZW);
609 mlvc_px_nc1hwc2_to_nchw(by0, e->y0_r, e->YC, e->YH, e->YW);
610 mlvc_px_nc1hwc2_to_nchw(by1, e->y1_r, e->YC, e->YH, e->YW);
611
612 /* 熵编码: push y1, y0, z */
613 mlvc_px_extract_scales(e->z_r, e->s0, e->s1, e->YC, e->YH, e->YW, e->ZH, e->ZW);
614
615 rkvc_rans_enc_stream stream;
616 rkvc_rans_enc_stream_init(&stream, RKVC_RANS_BYTE, 65536);
617 int rc = rkvc_rans_enc_stream_encode(&stream, &e->g_coder, e->s1, e->y1_r,
618 (size_t)e->YC * e->YH * e->YW);
619 if (rc == RKVC_RANS_OK)
620 rc = rkvc_rans_enc_stream_encode(&stream, &e->g_coder, e->s0, e->y0_r,
621 (size_t)e->YC * e->YH * e->YW);
622 if (rc == RKVC_RANS_OK)
623 rc = rkvc_rans_enc_stream_encode(&stream, &e->b_coder, e->z_idx, e->z_r,
624 (size_t)e->ZC * e->ZH * e->ZW);
625 if (rc != RKVC_RANS_OK) {
626 rkvc_rans_enc_stream_free(&stream);
627 return RKVC_ERR_INTERNAL;
628 }
629
630 size_t bits_size;
631 const uint8_t *bits = rkvc_rans_enc_stream_flush(&stream, &bits_size);
632 if (!bits) {
633 rkvc_rans_enc_stream_free(&stream);
634 return RKVC_ERR_INTERNAL;
635 }
636
637 rkvc_buffer *pkt = NULL;
638 rkvc_err err = rkvc_buffer_alloc_bitstream(&pkt, bits, bits_size, 1);
639 rkvc_rans_enc_stream_free(&stream);
640 if (err != RKVC_OK) return err;
641
642 pkt->pts = frame->pts;
643 pkt->dts = frame->pts;
644 pkt->key_frame = (e->frame_count == 0);
645 e->pending_pkt = pkt;
646
647 /* encoder feature 输出 → 下一帧参考（native NC1HWC2 直接拷贝）*/
648 const uint16_t *fv = rknn_output_data(&e->enc_model, e->enc_feat_out);
649 if (!fv)
650 return RKVC_ERR_HW;
651 memcpy(e->ref_nhwc, fv, (size_t)e->REF_C * e->REF_H * e->REF_W * sizeof(uint16_t));
652
653 e->frame_count++;
654 return RKVC_OK;
655}

703static rkvc_err dec_resolve_geom(rkvc_mlvc_dec *d)
704{
705 mlvc_rknn_model *dec = &d->dec_model;
706 d->dec_z_in = rknn_find_input(dec, "z_raw");
707 d->dec_y0_in = rknn_find_input(dec, "y_raw_0");
708 d->dec_y1_in = rknn_find_input(dec, "y_raw_1");
709 d->dec_ref_in = rknn_find_input(dec, "ref_feature");
710 d->dec_x_out = rknn_find_output(dec, "x_hat");
711 d->dec_ref_out = rknn_find_output(dec, "feature");
712 if (d->dec_z_in < 0 || d->dec_y0_in < 0 || d->dec_y1_in < 0 ||
713 d->dec_ref_in < 0 || d->dec_x_out < 0 || d->dec_ref_out < 0)
714 return RKVC_ERR_FORMAT;
715
716 int C1 = (int)dec->native_out_attr[d->dec_x_out].dims[1];
717 int native_h = (int)dec->native_out_attr[d->dec_x_out].dims[2];
718 int native_w = (int)dec->native_out_attr[d->dec_x_out].dims[3];
719 d->OUT_C2 = (dec->native_out_attr[d->dec_x_out].n_dims >= 5)
720 ? (int)dec->native_out_attr[d->dec_x_out].dims[4] : 8;
721 if (d->OUT_C2 < 3) d->OUT_C2 = 3;
722 d->x_pre_c = C1 * d->OUT_C2;
723 d->x_pre_h = native_h;
724 d->x_pre_w = native_w;
725 d->x_d2s_bs = 0;
726 d->IMG_H = native_h;
727 d->IMG_W = native_w;
728 if (d->x_pre_c % 3 == 0) {
729 int pix = d->x_pre_c / 3;
730 for (int s = 2; s <= 16; s++) {
731 if (s * s == pix) {
732 d->x_d2s_bs = s;
733 d->IMG_H = native_h * s;
734 d->IMG_W = native_w * s;
735 RKVC_LOG("MLVC decoder x_hat is pre-shuffle %dx%dx%d, CPU DepthToSpace DCR bs=%d → %dx%d",
736 d->x_pre_c, native_h, native_w, s, d->IMG_W, d->IMG_H);
737 break;
738 }
739 }
740 }
741 d->REF_C = dec->in_attr[d->dec_ref_in].dims[3];
742 d->REF_H = dec->in_attr[d->dec_ref_in].dims[1];
743 d->REF_W = dec->in_attr[d->dec_ref_in].dims[2];
744
745 d->ZC = dec->in_attr[d->dec_z_in].dims[3];
746 d->ZH = dec->in_attr[d->dec_z_in].dims[1];
747 d->ZW = dec->in_attr[d->dec_z_in].dims[2];
748 d->YC = dec->in_attr[d->dec_y0_in].dims[3];
749 d->YH = dec->in_attr[d->dec_y0_in].dims[1];
750 d->YW = dec->in_attr[d->dec_y0_in].dims[2];
751 return RKVC_OK;
752}

754static rkvc_err dec_alloc_bufs(rkvc_mlvc_dec *d)
755{
756 size_t ref_sz = (size_t)d->REF_C * d->REF_H * d->REF_W;
757 size_t z_sz = (size_t)d->ZC * d->ZH * d->ZW;
758 size_t y_sz = (size_t)d->YC * d->YH * d->YW;
759
760 d->ref_nhwc = rkvc_malloc(ref_sz * 2);
761 d->z_d = rkvc_malloc(z_sz * 4);
762 d->y0_d = rkvc_malloc(y_sz * 4);
763 d->y1_d = rkvc_malloc(y_sz * 4);
764 d->s0 = rkvc_malloc(y_sz * 4);
765 d->s1 = rkvc_malloc(y_sz * 4);
766 d->z_idx = rkvc_malloc(z_sz * 4);
767 d->z_d_nhwc = rkvc_malloc(z_sz * 2);
768 d->y0_d_nhwc = rkvc_malloc(y_sz * 2);
769 d->y1_d_nhwc = rkvc_malloc(y_sz * 2);
770 d->x_img = NULL;
771 if (d->x_d2s_bs > 0)
772 d->x_img = rkvc_malloc(3ull * d->IMG_H * d->IMG_W * 2);
773 if (!d->ref_nhwc || !d->z_d || !d->y0_d || !d->y1_d || !d->s0 || !d->s1 ||
774 !d->z_idx || !d->z_d_nhwc || !d->y0_d_nhwc || !d->y1_d_nhwc ||
775 (d->x_d2s_bs > 0 && !d->x_img))
776 return RKVC_ERR_NOMEM;
777
778 memset(d->ref_nhwc, 0, ref_sz * 2);
779 /* z_idx 在通道平面内为常量：按平面填充（原逐元素三重循环）*/
780 for (int c = 0; c < d->ZC; c++) {
781 int32_t *p = d->z_idx + (size_t)c * d->ZH * d->ZW;
782 size_t n = (size_t)d->ZH * d->ZW;
783 for (size_t i = 0; i < n; i++)
784 p[i] = d->qp * d->ZC + c;
785 }
786 return RKVC_OK;
787}

889
890rkvc_err rkvc_mlvc_dec_receive_frame(rkvc_mlvc_dec *d, rkvc_buffer **frame)
891{
892 if (!d || !frame) return RKVC_ERR_INVALID;
893 *frame = NULL;
894 rkvc_buffer_unref(d->out_frame);
895 d->out_frame = NULL;
896
897 if (!d->pending_pkt) {
898 if (d->input_eof) return RKVC_ERR_EOF;
899 return RKVC_ERR_AGAIN;
900 }
901
902 const rkvc_buffer *pkt = d->pending_pkt;
903
904 /* 熵解码 */
905 rkvc_rans_dec_stream ds;
906 rkvc_rans_dec_stream_init(&ds, RKVC_RANS_BYTE);
907 int rc = rkvc_rans_dec_stream_open(&ds, pkt->data, pkt->size);
908 if (rc != RKVC_RANS_OK) { RKVC_LOG("dec stream open failed"); return RKVC_ERR_FORMAT; }
909
910 size_t z_n = (size_t)d->ZC * d->ZH * d->ZW;
911 size_t y_n = (size_t)d->YC * d->YH * d->YW;
912
913 rc = rkvc_rans_dec_stream_decode(&ds, &d->b_coder, d->z_d, d->z_idx, z_n);
914 if (rc != RKVC_RANS_OK) { rkvc_rans_dec_stream_close(&ds); return RKVC_ERR_INTERNAL; }
915
916 mlvc_px_extract_scales(d->z_d, d->s0, d->s1, d->YC, d->YH, d->YW, d->ZH, d->ZW);
917
918 rc = rkvc_rans_dec_stream_decode(&ds, &d->g_coder, d->y0_d, d->s0, y_n);
919 if (rc == RKVC_RANS_OK)
920 rc = rkvc_rans_dec_stream_decode(&ds, &d->g_coder, d->y1_d, d->s1, y_n);
921 if (rc != RKVC_RANS_OK) { rkvc_rans_dec_stream_close(&ds); return RKVC_ERR_INTERNAL; }
922
923 rkvc_rans_dec_stream_close(&ds);
924
925 /* NCHW → NC1HWC2 fp16 */
926 mlvc_px_nchw_to_nc1hwc2_fp16(d->z_d, d->z_d_nhwc, d->ZC, d->ZH, d->ZW);
927 mlvc_px_nchw_to_nc1hwc2_fp16(d->y0_d, d->y0_d_nhwc, d->YC, d->YH, d->YW);
928 mlvc_px_nchw_to_nc1hwc2_fp16(d->y1_d, d->y1_d_nhwc, d->YC, d->YH, d->YW);
929
930 /* decoder NPU */
931 rknn_write_input(&d->dec_model, d->dec_z_in, d->z_d_nhwc);
932 rknn_write_input(&d->dec_model, d->dec_y0_in, d->y0_d_nhwc);
933 rknn_write_input(&d->dec_model, d->dec_y1_in, d->y1_d_nhwc);
934 rknn_write_input(&d->dec_model, d->dec_ref_in, d->ref_nhwc);
935 MLVC_RKNN_CHECK(rknn_run(d->dec_model.ctx, NULL));
936
937 const uint16_t *xh = rknn_output_data(&d->dec_model, d->dec_x_out);
938 if (!xh)
939 return RKVC_ERR_HW;
940 rkvc_err err;
941 if (d->x_d2s_bs > 0) {
942 /* NC1HWC2 → DepthToSpace 融合（免中间 NCHW 遍）；clip(0,1)
943 * 融合进饱和 NV12 转换（越界值由整数饱和覆盖）*/
944 int C1 = d->x_pre_c / d->OUT_C2;
