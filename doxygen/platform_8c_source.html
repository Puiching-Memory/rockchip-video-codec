36/*
37 * device-tree compatible 以 '\0' 分隔多个条目，板型条目在前、通用 SoC
38 * 条目在后（如 "rockchip,rk3588-evb7-v11\0rockchip,rk3588"）。
39 * 取最后一个 "rockchip,<soc>" 条目（截断至首个 '-'）即 SoC 名。
40 */
41static void probe_soc_name(char out[32])
42{
43 out[0] = '\0';
44 int fd = open("/proc/device-tree/compatible", O_RDONLY | O_CLOEXEC);
45 if (fd < 0)
46 return;
47
48 char buf[256];
49 ssize_t n = read(fd, buf, sizeof(buf) - 1);
50 close(fd);
51 if (n <= 0)
52 return;
53 buf[n] = '\0';
54
55 size_t off = 0;
56 while (off < (size_t)n) {
57 const char *entry = buf + off;
58 size_t len = strlen(entry);
59 const char *tok = strstr(entry, "rockchip,");
60 if (tok) {
61 tok += strlen("rockchip,");
62 size_t k = 0;
63 while (tok[k] && tok[k] != '-' && k + 1 < 32) {
64 out[k] = (char)tolower((unsigned char)tok[k]);
65 k++;
66 }
67 out[k] = '\0';
68 }
69 off += len + 1; /* 跳过本条目及其结尾 '\0' */
70 }
71}

79/* NPU 存在性（存在即可，不要求访问权限）：rknpu debugfs 节点或 DRI NPU 节点 */
80static int probe_npu_present(void)
81{
82 if (path_exists("/sys/kernel/debug/rknpu/version"))
83 return 1;
84
85 DIR *dir = opendir("/dev/dri/by-path");
86 if (!dir)
87 return 0;
88 int found = 0;
89 struct dirent *ent;
90 while ((ent = readdir(dir)) != NULL) {
91 if (strstr(ent->d_name, "npu")) {
92 found = 1;
93 break;
94 }
95 }
96 closedir(dir);
97 return found;
98}

100/*
101 * NPU 核心数：rknpu debugfs `load` 按核打印负载
102 *（"NPU load: Core0: 0%, Core1: 0%, ..."），数 CoreN: 条目个数。
103 * debugfs 未挂载 / 不可读时返回 0 表示未知。
104 */
105static int probe_npu_cores(void)
106{
107 FILE *fp = fopen("/sys/kernel/debug/rknpu/load", "r");
108 if (!fp)
109 return 0;
110 char buf[256];
111 size_t n = fread(buf, 1, sizeof(buf) - 1, fp);
112 fclose(fp);
113 buf[n] = '\0';
114
115 int cores = 0;
116 const char *p = buf;
117 while ((p = strstr(p, "Core")) != NULL) {
118 p += 4;
119 if (!isdigit((unsigned char)*p))
120 continue;
121 while (isdigit((unsigned char)*p))
122 p++;
123 if (*p == ':')
124 cores++;
125 }
126 return cores;
127}

129/*
130 * VPU 能力位 → 编解码映射（MPP 引擎语义）：
131 * 解码 VDPU1/VDPU2 仅 H.264，RKVDEC 覆盖 H.264/HEVC，AV1DEC 为 AV1；
132 * 编码 VEPU1/VEPU2 仅 H.264，VEPU22 与 RKVENC 覆盖 H.264/HEVC。
133 * Rockchip 无 AV1 硬件编码引擎（MPP 无对应 client 类型），
134 * AV1 编码始终走软件（SVT-AV1），故无对应字段。
135 */
136static void probe_vpu(rkvc_platform_info *info)
137{
138 const rk_u32 v = mpp_get_vcodec_type();
139
140 info->vpu_h264_dec_hw = !!(v & (HAVE_VDPU1 | HAVE_VDPU2 | HAVE_RKVDEC));
141 info->vpu_hevc_dec_hw = !!(v & (HAVE_HEVC_DEC | HAVE_RKVDEC));
142 info->vpu_av1_dec_hw = !!(v & HAVE_AV1DEC);
143 info->vpu_h264_enc_hw = !!(v & (HAVE_VEPU1 | HAVE_VEPU2 | HAVE_VEPU22 | HAVE_RKVENC));
144 info->vpu_hevc_enc_hw = !!(v & (HAVE_VEPU22 | HAVE_RKVENC));
145}

147const rkvc_platform_info *rkvc_platform_probe(void)
148{
149 pthread_mutex_lock(&g_lock);
150 if (!g_done) {
151 probe_soc_name(g_info.soc);
152 g_info.has_npu = probe_npu_present();
153 g_info.npu_cores = probe_npu_cores();
154 g_info.has_rga = path_exists("/dev/rga");
155 probe_vpu(&g_info);
156 g_done = 1;
157 }
158 pthread_mutex_unlock(&g_lock);
159 return &g_info;
160}
