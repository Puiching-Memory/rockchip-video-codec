106static rkvc_err session_open_nodes(rkvc_session *s)
107{
108 /* 允许 start 失败后重试：先关掉上次半开的节点，避免 FD/指针泄漏 */
109 session_close_nodes(s);
110
111 const rkvc_pipeline_desc *d = &s->desc;
112
113 if (d->template_id == RKVC_TEMPLATE_LIVE_CAPTURE) {
114 if (!d->capture_device || !d->capture_device[0] || !d->output_path)
115 return RKVC_ERR_INVALID;
116
117 rkvc_v4l2_config vc = {
118 .device = d->capture_device,
119 .width = d->width,
120 .height = d->height,
121 .fps_num = d->fps_num,
122 .fps_den = d->fps_den,
123 };
124 rkvc_err err = rkvc_v4l2_open(&s->v4l2, &vc);
125 if (err != RKVC_OK)
126 return err;
127
128 int aw = 0, ah = 0;
129 rkvc_v4l2_get_size(s->v4l2, &aw, &ah);
130 if (aw > 0 && ah > 0) {
131 s->desc.width = aw;
132 s->desc.height = ah;
133 s->desc.pixel_format = RKVC_PIX_FMT_NV12;
134 }
135 }
136
137 /* ── MLVC 输入侧：.mlvc 文件 → mlvc_demux + mlvc_dec ── */
138#ifdef RKVC_ENABLE_MLVC
139 int input_is_mlvc = (d->input_path &&
140 session_is_mlvc_file(d->input_path));
141 if (input_is_mlvc &&
142 (d->template_id == RKVC_TEMPLATE_FILE_DECODE ||
143 d->template_id == RKVC_TEMPLATE_FILE_TRANSCODE ||
144 d->template_id == RKVC_TEMPLATE_MLVC_STORAGE)) {
145 if (!d->input_path)
146 return RKVC_ERR_INVALID;
147 rkvc_mlvc_demux_config mdc = { .input_path = d->input_path };
148 rkvc_err err = rkvc_mlvc_demux_open(&s->mlvc_demux, &mdc);
149 if (err != RKVC_OK)
150 return err;
151 int dw = 0, dh = 0;
152 char dec_patch_buf[768];
153 const char *dec_patch = NULL;
154 rkvc_err perr = rkvc_qppatch_resolve(d->mlvc_qp_patch_dir, "dec",
155 rkvc_mlvc_demux_qp(s->mlvc_demux),
156 dec_patch_buf, sizeof dec_patch_buf,
157 &dec_patch);
158 if (perr != RKVC_OK)
159 return perr;
160 rkvc_mlvc_dec_config dcfg = {
161 .dec_model_path = d->mlvc_dec_model_path,
162 .gaussian_pmf_path = d->mlvc_gaussian_pmf_path,
163 .bitest_pmf_path = d->mlvc_bitest_pmf_path,
164 .qp = rkvc_mlvc_demux_qp(s->mlvc_demux),
165 .qp_patch_path = dec_patch,
166 };
167 err = rkvc_mlvc_dec_open(&s->mlvc_dec, &dcfg);
168 if (err != RKVC_OK)
169 return err;
170 dw = rkvc_mlvc_dec_width(s->mlvc_dec);
171 dh = rkvc_mlvc_dec_height(s->mlvc_dec);
172 if (dw > 0 && dh > 0) {
173 s->desc.width = dw;
174 s->desc.height = dh;
175 }
176 }
177#endif
178
179 if (d->template_id == RKVC_TEMPLATE_FILE_TRANSCODE ||
180 d->template_id == RKVC_TEMPLATE_FILE_DECODE ||
181 d->template_id == RKVC_TEMPLATE_AV1_STORAGE ||
182 d->template_id == RKVC_TEMPLATE_MLVC_STORAGE) {
183 /* .mlvc 输入由上面 MLVC 分支处理 */
184#ifdef RKVC_ENABLE_MLVC
185 if (s->mlvc_demux)
186 goto mlvc_input_done;
187#endif
188 if (!d->input_path)
189 return RKVC_ERR_INVALID;
190
191 rkvc_demux_config dc = {
192 .input_path = d->input_path,
193 .format_opts = d->codec_opts,
194 };
195 rkvc_err err = rkvc_demux_open(&s->demux, &dc);
196 if (err != RKVC_OK)
197 return err;
198
199 AVCodecParameters *par = rkvc_demux_video_par(s->demux);
200 rkvc_mpp_dec_config mdc = {
201 .route = &s->route,
202 .output_format = d->pixel_format,
203 .low_latency = d->low_latency,
204 .codec_opts = d->codec_opts,
205 };
206 err = rkvc_mpp_dec_open(&s->dec, &mdc, par);
207 if (err != RKVC_OK)
208 return err;
209 }
210mlvc_input_done:
211 ;
212
213 if (d->template_id == RKVC_TEMPLATE_FILE_ENCODE) {
214 if (!d->output_path)
215 return RKVC_ERR_INVALID;
216 }
217
218 if (d->template_id == RKVC_TEMPLATE_FILE_TRANSCODE ||
219 d->template_id == RKVC_TEMPLATE_FILE_ENCODE ||
220 d->template_id == RKVC_TEMPLATE_LIVE_CAPTURE ||
221 d->template_id == RKVC_TEMPLATE_AV1_STORAGE ||
222 d->template_id == RKVC_TEMPLATE_MLVC_STORAGE) {
223 if (!d->output_path)
224 return RKVC_ERR_INVALID;
225
226#ifdef RKVC_ENABLE_MLVC
227 int output_is_mlvc = (d->output_path &&
228 session_is_mlvc_file(d->output_path));
229 /* MLVC 编码器 + .mlvc 容器（不走 FFmpeg mux）。
230 * 编解码器独立选择：解码器由输入决定（.mlvc 输入 → MLVC 解码），
231 * 编码器由输出决定（.mlvc 输出 → MLVC 编码；标准容器 → 标准编码器）。
232 * .mlvc → .yuv 纯解码不经过此输出块（FILE_DECODE 模板，无编码器）。 */
233 if (s->route.enc_backend == RKVC_ENC_BACKEND_MLVC && output_is_mlvc) {
234 if (!d->mlvc_enc_model_path ||
235 !d->mlvc_gaussian_pmf_path || !d->mlvc_bitest_pmf_path)
236 return RKVC_ERR_INVALID;
237 char enc_patch_buf[768];
238 const char *enc_patch = NULL;
239 rkvc_err perr = rkvc_qppatch_resolve(d->mlvc_qp_patch_dir, "enc",
240 d->mlvc_qp, enc_patch_buf,
241 sizeof enc_patch_buf, &enc_patch);
242 if (perr != RKVC_OK)
243 return perr;
244 rkvc_mlvc_enc_config ec = {
245 .enc_model_path = d->mlvc_enc_model_path,
246 .gaussian_pmf_path = d->mlvc_gaussian_pmf_path,
247 .bitest_pmf_path = d->mlvc_bitest_pmf_path,
248 .qp = d->mlvc_qp,
249 .qp_patch_path = enc_patch,
250 };
251 rkvc_err err = rkvc_mlvc_enc_open(&s->mlvc_enc, &ec);
252 if (err != RKVC_OK)
253 return err;
254 int ew = rkvc_mlvc_enc_width(s->mlvc_enc);
255 int eh = rkvc_mlvc_enc_height(s->mlvc_enc);
256 if (ew > 0 && eh > 0) {
257 s->desc.width = ew;
258 s->desc.height = eh;
259 }
260 rkvc_mlvc_mux_config mc = {
261 .output_path = d->output_path,
262 .width = ew,
263 .height = eh,
264 .fps_num = d->fps_num,
265 .fps_den = d->fps_den,
266 .qp = d->mlvc_qp,
267 };
268 err = rkvc_mlvc_mux_open(&s->mlvc_mux, &mc);
269 if (err != RKVC_OK)
270 return err;
271 goto mlvc_output_done;
272 }
273#endif
274
275 if (s->route.enc_backend == RKVC_ENC_BACKEND_MPP) {
276 int ew = 0, eh = 0;
277 rkvc_err szerr = session_enc_size(s, &ew, &eh);
278 if (szerr != RKVC_OK)
279 return szerr;
280 rkvc_mpp_enc_config ec = {
281 .route = &s->route,
282 .width = ew,
283 .height = eh,
284 .fps_num = d->fps_num,
285 .fps_den = d->fps_den,
286 .bitrate = d->bitrate,
287 .input_format = d->pixel_format,
288 .gop_size = d->gop_size,
289 .low_latency = d->low_latency,
290 .rc_mode = d->rc_mode,
291 .qp_init = d->qp_init,
292 .codec_opts = d->codec_opts,
293 };
294 rkvc_err err = rkvc_mpp_enc_open(&s->enc, &ec);
295 if (err != RKVC_OK)
296 return err;
297 } else {
298 int ew = 0, eh = 0;
299 rkvc_err szerr = session_enc_size(s, &ew, &eh);
300 if (szerr != RKVC_OK)
301 return szerr;
302 rkvc_svt_enc_config sc = {
303 .width = ew,
304 .height = eh,
305 .fps_num = d->fps_num,
306 .fps_den = d->fps_den,
307 .bitrate = d->bitrate,
308 .input_format = d->pixel_format,
309 .gop_size = d->gop_size,
310 .svt_preset = s->route.svt_preset,
311 .svt_lp = d->svt_lp,
312 .svt_rtc = d->svt_rtc,
313 .rc_mode = d->rc_mode,
314 };
315 rkvc_err err = rkvc_svt_enc_open(&s->svt, &sc);
316 if (err != RKVC_OK)
317 return err;
318 }
319
320 int ew = 0, eh = 0;
321 rkvc_err szerr = session_enc_size(s, &ew, &eh);
322 if (szerr != RKVC_OK)
323 return szerr;
324
325 rkvc_mux_config mc = {
326 .output_path = d->output_path,
327 .route = &s->route,
328 .width = ew,
329 .height = eh,
330 .fps_num = d->fps_num,
331 .fps_den = d->fps_den,
332 .bitrate = d->bitrate,
333 .pixel_format = d->pixel_format,
334 };
335
336 AVCodecParameters *par = avcodec_parameters_alloc();
337 if (!par)
338 return RKVC_ERR_NOMEM;
339 par->codec_type = AVMEDIA_TYPE_VIDEO;
340 par->width = ew;
341 par->height = eh;
342 par->bit_rate = d->bitrate;
343 if (s->route.codec == RKVC_CODEC_HEVC)
344 par->codec_id = AV_CODEC_ID_HEVC;
345 else if (s->route.codec == RKVC_CODEC_AV1)
346 par->codec_id = AV_CODEC_ID_AV1;
347 else
348 par->codec_id = AV_CODEC_ID_H264;
349
350 if (s->svt)
351 rkvc_svt_enc_write_header(s->svt, par);
352
353 rkvc_err err = rkvc_mux_open(&s->mux, &mc, par);
354 avcodec_parameters_free(&par);
355 if (err != RKVC_OK) {
356 return err;
357 }
358 }
359mlvc_output_done:
360 ;
361
362 return RKVC_OK;
363}

365rkvc_err rkvc_session_create(const rkvc_pipeline_desc *desc,
366 rkvc_session **out)
367{
368 if (!desc || !out)
369 return RKVC_ERR_INVALID;
370
371 if (desc->post_upscale_algo == RKVC_UPSCALE_AI_SR &&
372 (!desc->post_upscale_rkvc_model_path ||
373 !desc->post_upscale_rkvc_model_path[0]))
374 return RKVC_ERR_INVALID;
375
376 *out = NULL;
377 rkvc_err init_err = rkvc_init();
378 if (init_err != RKVC_OK)
379 return init_err;
380
381 int rt_flags = 0;
382 rkvc_err rerr = rkvc_runtime_try_register(desc, &rt_flags);
383 if (rerr != RKVC_OK)
384 return rerr;
385
386 rkvc_session *s = rkvc_calloc(1, sizeof(*s));
387 if (!s) {
388 rkvc_runtime_unregister(rt_flags);
389 return RKVC_ERR_NOMEM;
390 }
391
392 s->desc = *desc;
393 s->runtime_flags = rt_flags;
394 if (s->desc.queue_depth <= 0)
395 s->desc.queue_depth = RKVC_PORT_QUEUE_DEFAULT;
396 if (s->desc.capture_timeout_ms <= 0)
397 s->desc.capture_timeout_ms = 1000;
398
399 pthread_mutex_init(&s->lock, NULL);
400
401 if (s->desc.enc_scale_denom > 1) {
402 int ew = 0, eh = 0;
403 if (session_enc_size(s, &ew, &eh) != RKVC_OK) {
404 rkvc_session_destroy(s);
405 return RKVC_ERR_INVALID;
406 }
407 }
408
409 rkvc_err err = rkvc_route_resolve(&s->desc, &s->route);
410 if (err != RKVC_OK) {
411 rkvc_session_destroy(s);
412 return err;
413 }
414
415 /* ── 编解码器独立选择（按方向修正路由）──────────────────────────
416 * MLVC 是与 264/265 平行的端到端 AI 编解码器，而非转码中间件：
417 * 解码后端 ← 输入（.mlvc 输入 → MLVC 解码；否则标准解码）
418 * 编码后端 ← 输出（.mlvc 输出 → MLVC 编码；.yuv 输出 → 无编码器；
419 * 标准容器 → 标准编码器）
420 * 三种一等操作：编码 video→.mlvc / 解码 .mlvc→.yuv / 转码 .mlvc→容器 */
421#ifdef RKVC_ENABLE_MLVC
422 {
423 int in_mlvc = session_is_mlvc_file(s->desc.input_path);
424 int out_mlvc = session_is_mlvc_file(s->desc.output_path);
425 int out_raw = session_is_raw_out(s->desc.output_path);
426 if (in_mlvc) {
427 s->route.dec_backend = RKVC_DEC_BACKEND_MLVC;
428 s->route.dec_name = "mlvc";
429 } else if (out_mlvc) {
430 /* 编码：标准视频输入由 MPP 解码 */
431 s->route.dec_name = "mpp";
432 }
433 if (out_mlvc) {
434 s->route.enc_backend = RKVC_ENC_BACKEND_MLVC;
435 s->route.enc_name = "mlvc";
436 s->route.reason = "mlvc neural codec encode";
437 } else if (in_mlvc && out_raw) {
438 s->route.enc_name = "raw";
439 s->route.reason = "mlvc decode -> raw yuv";
440 } else if (in_mlvc) {
441 s->route.reason = "mlvc decode + standard encode (transcode)";
442 }
443 }
444#endif
445
446 s->pool = rkvc_buffer_pool_create();
447 if (!s->pool) {
448 rkvc_session_destroy(s);
449 return RKVC_ERR_NOMEM;
450 }
451
452 if (port_init(&s->port_capture, "capture", s->desc.queue_depth, s) != RKVC_OK ||
453 port_init(&s->port_output, "output", s->desc.queue_depth, s) != RKVC_OK ||
454 port_init(&s->port_preview, "preview", s->desc.queue_depth, s) != RKVC_OK) {
455 rkvc_session_destroy(s);
456 return RKVC_ERR_NOMEM;
457 }
458
459 s->stats.route = s->route;
460 *out = s;
461 return RKVC_OK;
462}

464rkvc_err rkvc_session_start(rkvc_session *session)
465{
466 if (!session)
467 return RKVC_ERR_INVALID;
468 if (session->running)
469 return RKVC_OK;
470
471 rkvc_err err = session_open_nodes(session);
472 if (err != RKVC_OK) {
473 session_close_nodes(session);
474 return err;
475 }
476
477 pthread_mutex_lock(&session->lock);
478 session->running = 1;
479 session->stats.running = 1;
480 pthread_mutex_unlock(&session->lock);
481 return RKVC_OK;
482}
