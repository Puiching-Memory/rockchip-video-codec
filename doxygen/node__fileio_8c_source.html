50static int source_configure(rkvc_node *node, rkvc_diag **diag) {
51 file_source *src = node->priv;
52 rkvc_frame_spec out = {0};
53
54 switch (src->request.operation) {
55 case RKVC_OPERATION_DECODE:
56 case RKVC_OPERATION_TRANSCODE:
57 out.fmt = RKVC_FRAME_FMT_BITSTREAM;
58 out.domain = RKVC_MEM_DOMAIN_HOST;
59 break;
60 case RKVC_OPERATION_UPSCALE:
61 if (!src->request.width || !src->request.height) {
62 /* 几何未知：整文件按 BITSTREAM 产出。当前没有图像 demux，
63 * transform 节点会在协商/处理期拒绝并留下诊断。 */
64 out.fmt = RKVC_FRAME_FMT_BITSTREAM;
65 out.domain = RKVC_MEM_DOMAIN_HOST;
66 break;
67 }
68 /* 几何已知：与 ENCODE 同路径，输入按 NV12 原始帧解释。 */
69 out.fmt = RKVC_FRAME_FMT_NV12;
70 out.domain = RKVC_MEM_DOMAIN_HOST;
71 out.width = src->request.width;
72 out.height = src->request.height;
73 out.stride = src->request.width;
74 break;
75 case RKVC_OPERATION_ENCODE:
76 if (!src->request.width || !src->request.height) {
77 if (diag)
78 rkvc_diag_push(diag, RKVC_STATUS_FORMAT, 1, node->ops->id,
79 "encode source requires width/height");
80 return (int)RKVC_STATUS_FORMAT;
81 }
82 out.fmt = RKVC_FRAME_FMT_NV12;
83 out.domain = RKVC_MEM_DOMAIN_HOST;
84 out.width = src->request.width;
85 out.height = src->request.height;
86 out.stride = src->request.width;
87 break;
88 default:
89 return (int)RKVC_STATUS_INVALID;
90 }
91 rkvc_port_set_desired(&node->out_ports[0], &out);
92 return 0;
93}

110static int source_emit(rkvc_node *node, const rkvc_frame_spec *spec,
111 void *data, size_t size) {
112 rkvc_frame_desc desc;
113 rkvc_frame *frame = NULL;
114 rkvc_status st;
115 int rc;
116
117 rkvc_frame_desc_init(&desc, sizeof(desc));
118 desc.spec = *spec;
119 desc.data = data;
120 desc.size = size;
121 st = rkvc_backend_frame_create(&desc, fileio_frame_release, data, &frame);
122 if (st != RKVC_STATUS_OK) {
123 rkvc_g_free(data);
124 return (int)st;
125 }
126 rc = rkvc_node_emit(node, 0, frame);
127 if (rc != 0)
128 rkvc_frame_release(frame);
129 return rc;
130}

133static int source_flush(rkvc_node *node, rkvc_diag **diag) {
134 file_source *src = node->priv;
135 const rkvc_frame_spec *spec = &node->out_ports[0].fmt;
136 size_t chunk = FILEIO_BITSTREAM_CHUNK;
137 int rc = 0;
138
139 if (!src->fp)
140 return (int)RKVC_STATUS_INVALID;
141 if (spec->fmt == RKVC_FRAME_FMT_NV12) {
142 size_t row = spec->stride ? spec->stride : spec->width;
143 chunk = row * spec->height * 3 / 2;
144 }
145
146 for (;;) {
147 void *data = rkvc_g_calloc(1, chunk);
148 size_t got;
149 if (!data)
150 return (int)RKVC_STATUS_NOMEM;
151 got = fread(data, 1, chunk, src->fp);
152 if (got == 0) {
153 rkvc_g_free(data);
154 if (ferror(src->fp)) {
155 if (diag)
156 rkvc_diag_push(diag, RKVC_STATUS_IO, 3, node->ops->id,
157 "input read failed");
158 return (int)RKVC_STATUS_IO;
159 }
160 break; /* EOF */
161 }
162 if (spec->fmt == RKVC_FRAME_FMT_NV12 && got != chunk) {
163 rkvc_g_free(data);
164 if (diag)
165 rkvc_diag_push(diag, RKVC_STATUS_FORMAT, 3, node->ops->id,
166 "trailing partial raw frame");
167 return (int)RKVC_STATUS_FORMAT;
168 }
169 rc = source_emit(node, spec, data, got);
170 if (rc != 0)
171 return rc; /* 取消/错误时所有权已归还，缓冲已释放 */
172 }
173 return 0;
174}

283static int sink_write_video(file_sink *sink, const rkvc_frame_desc *desc,
284 const unsigned char *base) {
285 const rkvc_frame_spec *spec = &desc->spec;
286 size_t bpp = spec->fmt == RKVC_FRAME_FMT_P010 ? 2 : 1;
287 size_t visible = (size_t)spec->width * bpp;
288 size_t row = spec->stride ? spec->stride : visible;
289 size_t vstride = spec->ver_stride ? spec->ver_stride : spec->height;
290 uint32_t y;
291 int rc;
292
293 if (!spec->width || !spec->height)
294 return (int)RKVC_STATUS_FORMAT;
295 if ((size_t)spec->height > SIZE_MAX / row)
296 return (int)RKVC_STATUS_FORMAT;
297
298 for (y = 0; y < spec->height; ++y) {
299 rc = sink_write(sink, base + (size_t)y * row, visible);
300 if (rc != 0)
301 return rc;
302 }
303 if (spec->fmt == RKVC_FRAME_FMT_NV12 ||
304 spec->fmt == RKVC_FRAME_FMT_NV21 ||
305 spec->fmt == RKVC_FRAME_FMT_P010) {
306 const unsigned char *uv = base + row * vstride;
307 for (y = 0; y < spec->height / 2; ++y) {
308 rc = sink_write(sink, uv + (size_t)y * row, visible);
309 if (rc != 0)
310 return rc;
311 }
312 } else if (spec->fmt == RKVC_FRAME_FMT_YUV420P) {
313 const unsigned char *u = base + row * vstride;
314 const unsigned char *v = u + row * vstride / 4;
315 for (y = 0; y < spec->height / 2; ++y) {
316 rc = sink_write(sink, u + (size_t)y * row / 2, visible / 2);
317 if (rc != 0)
318 return rc;
319 }
320 for (y = 0; y < spec->height / 2; ++y) {
321 rc = sink_write(sink, v + (size_t)y * row / 2, visible / 2);
322 if (rc != 0)
323 return rc;
324 }
325 } else {
326 return (int)RKVC_STATUS_FORMAT;
327 }
328 return 0;
329}

340static int sink_process(rkvc_node *node, rkvc_frame *input,
341 rkvc_diag **diag) {
342 file_sink *sink = node->priv;
343 rkvc_frame_desc desc;
344 const void *base;
345 void *mapped = NULL;
346 int rc;
347 (void)diag;
348
349 if (!sink->fp)
350 return (int)RKVC_STATUS_INVALID;
351 if (rkvc_frame_get_desc(input, &desc) != RKVC_STATUS_OK)
352 return (int)RKVC_STATUS_FORMAT;
353
354 if (desc.spec.domain == RKVC_MEM_DOMAIN_DMABUF) {
355#ifdef __linux__
356 if (desc.size == 0)
357 return (int)RKVC_STATUS_FORMAT;
358 mapped = mmap(NULL, desc.size, PROT_READ, MAP_SHARED, desc.fd, 0);
359 if (mapped == MAP_FAILED)
360 return (int)RKVC_STATUS_IO;
361 dmabuf_read_sync(desc.fd, DMA_BUF_SYNC_START | DMA_BUF_SYNC_READ);
362 base = mapped;
363#else
364 return (int)RKVC_STATUS_UNSUPPORTED;
365#endif
366 } else {
367 if (!desc.data && desc.size)
368 return (int)RKVC_STATUS_FORMAT;
369 base = desc.data;
370 }
371
372 if (desc.spec.fmt == RKVC_FRAME_FMT_BITSTREAM)
373 rc = sink_write(sink, base, desc.size);
374 else
375 rc = sink_write_video(sink, &desc, base);
376
377 if (mapped) {
378#ifdef __linux__
379 dmabuf_read_sync(desc.fd, DMA_BUF_SYNC_END | DMA_BUF_SYNC_READ);
380 munmap(mapped, desc.size);
381#endif
382 }
383 if (rc == 0)
384 sink->frames++;
385 return rc;
386}
