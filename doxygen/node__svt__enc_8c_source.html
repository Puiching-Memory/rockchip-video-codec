26static rkvc_err ensure_yuv420p(rkvc_svt_enc *enc, AVFrame *src, AVFrame **out)
27{
28 if (!src)
29 return RKVC_ERR_INVALID;
30
31 if (src->format == AV_PIX_FMT_YUV420P) {
32 *out = src;
33 return RKVC_OK;
34 }
35
36 if (enc->sws && enc->sws_src_fmt != src->format) {
37 sws_freeContext(enc->sws);
38 enc->sws = NULL;
39 }
40 if (!enc->sws) {
41 enc->sws = sws_getContext(enc->width, enc->height, src->format,
42 enc->width, enc->height, AV_PIX_FMT_YUV420P,
43 SWS_BILINEAR, NULL, NULL, NULL);
44 if (!enc->sws)
45 return RKVC_ERR_INTERNAL;
46 enc->sws_src_fmt = src->format;
47 }
48
49 AVFrame *dst = av_frame_alloc();
50 if (!dst)
51 return RKVC_ERR_NOMEM;
52 dst->format = AV_PIX_FMT_YUV420P;
53 dst->width = enc->width;
54 dst->height = enc->height;
55 if (av_frame_get_buffer(dst, 32) < 0) {
56 av_frame_free(&dst);
57 return RKVC_ERR_NOMEM;
58 }
59 if (sws_scale(enc->sws, (const uint8_t *const *)src->data, src->linesize,
60 0, enc->height, dst->data, dst->linesize) < 0) {
61 av_frame_free(&dst);
62 return RKVC_ERR_INTERNAL;
63 }
64 dst->pts = src->pts;
65 av_frame_free(&src);
66 *out = dst;
67 return RKVC_OK;
68}

70rkvc_err rkvc_svt_enc_open(rkvc_svt_enc **out, const rkvc_svt_enc_config *cfg)
71{
72 if (!out || !cfg || cfg->width <= 0 || cfg->height <= 0)
73 return RKVC_ERR_INVALID;
74
75 *out = NULL;
76 rkvc_err init_err = rkvc_init();
77 if (init_err != RKVC_OK)
78 return init_err;
79
80 const AVCodec *codec = avcodec_find_encoder_by_name("libsvtav1");
81 if (!codec)
82 return RKVC_ERR_NOT_FOUND;
83
84 rkvc_svt_enc *enc = rkvc_calloc(1, sizeof(*enc));
85 if (!enc)
86 return RKVC_ERR_NOMEM;
87
88 enc->width = cfg->width;
89 enc->height = cfg->height;
90 enc->ctx = avcodec_alloc_context3(codec);
91 if (!enc->ctx) {
92 rkvc_svt_enc_close(enc);
93 return RKVC_ERR_NOMEM;
94 }
95
96 enc->ctx->width = cfg->width;
97 enc->ctx->height = cfg->height;
98 enc->ctx->time_base = (AVRational){cfg->fps_den, cfg->fps_num};
99 enc->ctx->framerate = (AVRational){cfg->fps_num, cfg->fps_den};
100 enc->ctx->bit_rate = cfg->bitrate;
101 enc->ctx->gop_size = cfg->gop_size;
102 enc->ctx->pix_fmt = AV_PIX_FMT_YUV420P;
103 enc->ctx->flags |= AV_CODEC_FLAG_GLOBAL_HEADER;
104 /* 不设 rc_max_rate==bit_rate，使 libsvtav1 走 VBR（含 CBR 请求）。 */
105 enc->ctx->rc_max_rate = 0;
106
107 if (!rkvc_is_valid_rc_mode(cfg->rc_mode)) {
108 rkvc_svt_enc_close(enc);
109 return RKVC_ERR_INVALID;
110 }
111
112 AVDictionary *enc_opts = NULL;
113 int preset = cfg->svt_preset > 0 ? cfg->svt_preset : RKVC_SVT_PRESET_PERF;
114 rkvc_err perr = rkvc_dict_set_int(&enc_opts, "preset", preset);
115 if (perr != RKVC_OK) {
116 rkvc_svt_enc_close(enc);
117 return perr;
118 }
119
120 /* 关闭 Dolby Vision 自动探测，避免无 RPU 时失败。 */
121 if (av_dict_set(&enc_opts, "dolbyvision", "0", 0) < 0) {
122 rkvc_dict_free(&enc_opts);
123 rkvc_svt_enc_close(enc);
124 return RKVC_ERR_NOMEM;
125 }
126
127 char params[64];
128 int lp = cfg->svt_lp >= 0 ? cfg->svt_lp : RKVC_SVT_LP_AUTO;
129 if (cfg->svt_rtc)
130 snprintf(params, sizeof(params), "lp=%d:rtc=1", lp);
131 else
132 snprintf(params, sizeof(params), "lp=%d", lp);
133 if (av_dict_set(&enc_opts, "svtav1-params", params, 0) < 0) {
134 rkvc_dict_free(&enc_opts);
135 rkvc_svt_enc_close(enc);
136 return RKVC_ERR_NOMEM;
137 }
138
139 if (cfg->rc_mode == RKVC_RC_CQP) {
140 /* 无独立 qp_init 时用中等 QP；可用后续 codec_opts 覆盖。 */
141 perr = rkvc_dict_set_int(&enc_opts, "qp", 30);
142 if (perr != RKVC_OK) {
143 rkvc_dict_free(&enc_opts);
144 rkvc_svt_enc_close(enc);
145 return perr;
146 }
147 }
148
149 perr = rkvc_opt_set_dict(enc->ctx->priv_data, &enc_opts);
150 if (perr != RKVC_OK) {
151 rkvc_dict_free(&enc_opts);
152 rkvc_svt_enc_close(enc);
153 return perr;
154 }
155
156 rkvc_err err = rkvc_codec_open2(enc->ctx, codec, &enc_opts, "svt_enc");
157 rkvc_dict_free(&enc_opts);
158 if (err != RKVC_OK) {
159 rkvc_svt_enc_close(enc);
160 return err;
161 }
162
163 enc->pkt = av_packet_alloc();
164 if (!enc->pkt) {
165 rkvc_svt_enc_close(enc);
166 return RKVC_ERR_NOMEM;
167 }
168
169 *out = enc;
170 return RKVC_OK;
171}

186rkvc_err rkvc_svt_enc_write_header(rkvc_svt_enc *enc, AVCodecParameters *par)
187{
188 if (!enc || !enc->ctx || !par)
189 return RKVC_ERR_INVALID;
190
191 if (enc->ctx->extradata && enc->ctx->extradata_size > 0) {
192 av_freep(&par->extradata);
193 par->extradata = av_mallocz((size_t)enc->ctx->extradata_size +
194 AV_INPUT_BUFFER_PADDING_SIZE);
195 if (!par->extradata)
196 return RKVC_ERR_NOMEM;
197 memcpy(par->extradata, enc->ctx->extradata,
198 (size_t)enc->ctx->extradata_size);
199 par->extradata_size = enc->ctx->extradata_size;
200 return RKVC_OK;
201 }
202
203 /* GLOBAL_HEADER 已开但仍无 extradata 时，由 mux 侧容忍空头。 */
204 return RKVC_OK;
205}

207rkvc_err rkvc_svt_enc_send_frame(rkvc_svt_enc *enc, rkvc_buffer *frame)
208{
209 if (!enc || !enc->ctx)
210 return RKVC_ERR_INVALID;
211 if (enc->flushed)
212 return RKVC_ERR_EOF;
213 if (!frame)
214 return rkvc_svt_enc_drain(enc);
215
216 AVFrame *avf = NULL;
217 rkvc_err err = rkvc_buffer_to_host_frame(frame, &avf);
218 if (err != RKVC_OK)
219 return err;
220
221 err = ensure_yuv420p(enc, avf, &avf);
222 if (err != RKVC_OK) {
223 av_frame_free(&avf);
224 return err;
225 }
226
227 /* 解复用 PTS 与编码器 time_base 不同；统一用单调帧序号，避免 MP4 帧率错乱。 */
228 avf->pts = enc->next_pts;
229
230 int ret = avcodec_send_frame(enc->ctx, avf);
231 av_frame_free(&avf);
232 if (ret == AVERROR(EAGAIN))
233 return RKVC_ERR_AGAIN;
234 if (ret >= 0)
235 enc->next_pts++;
236 return rkvc_from_averror(ret);
237}

239rkvc_err rkvc_svt_enc_receive_packet(rkvc_svt_enc *enc, rkvc_buffer **out)
240{
241 if (!enc || !enc->ctx || !out)
242 return RKVC_ERR_INVALID;
243
244 *out = NULL;
245 av_packet_unref(enc->pkt);
246 int ret = avcodec_receive_packet(enc->ctx, enc->pkt);
247 if (ret < 0) {
248 if (ret == AVERROR(EAGAIN))
249 return RKVC_ERR_AGAIN;
250 if (ret == AVERROR_EOF)
251 return RKVC_ERR_EOF;
252 return rkvc_from_averror(ret);
253 }
254
255 rkvc_buffer *b = rkvc_buffer_from_avpacket(enc->pkt);
256 if (!b)
257 return RKVC_ERR_NOMEM;
258
259 *out = b;
260 return RKVC_OK;
261}

263rkvc_err rkvc_svt_enc_drain(rkvc_svt_enc *enc)
264{
265 if (!enc || !enc->ctx)
266 return RKVC_ERR_INVALID;
267 if (enc->flushed)
268 return RKVC_OK;
269 int ret = avcodec_send_frame(enc->ctx, NULL);
270 if (ret == 0 || ret == AVERROR_EOF) {
271 enc->flushed = 1;
272 return RKVC_OK;
273 }
274 return rkvc_from_averror(ret);
