36static void pubkey_decrypt(uint8_t *out, size_t len)

83{
84 return (uint32_t)p[0] | ((uint32_t)p[1] << 8) |

103/* ── Ed25519 签名校验 ──────────────────────────────────────────── */
104
105static int ed25519_verify(const uint8_t *data, size_t data_len,
106 const uint8_t *sig, size_t sig_len)
107{
108 if (rkvc_license_pubkey_len != crypto_sign_PUBLICKEYBYTES ||
109 sig_len != crypto_sign_BYTES)
110 return 0;
111
112 uint8_t pubkey[32];
113 pubkey_decrypt(pubkey, sizeof(pubkey));
114
115 /* libsodium 原生支持 Ed25519 验签，无需 DER/PEM 包装 */
116 return crypto_sign_verify_detached(sig, data, (unsigned long)data_len,
117 pubkey) == 0 ? 1 : 0;
118}

129rkvc_err rkvc_license_verify_blob(const uint8_t *blob, size_t len,
130 rkvc_license_info *info)
131{
132 if (!blob)
133 return RKVC_ERR_INVALID;
134 if (len != RKVC_LICENSE_BLOB_SIZE)
135 return RKVC_ERR_INVALID;
136
137 if (sodium_init() < 0)
138 return RKVC_ERR_INTERNAL;
139
140 if (info)
141 memset(info, 0, sizeof(*info));
142
143 const uint8_t *signed_region = blob;
144 const uint8_t *sig = blob + RKVC_LICENSE_SIGNED_LEN;
145
146 uint32_t magic = get_u32_le(blob);
147 uint32_t product_id = get_u32_le(blob + 4);
148 const uint8_t *license_mid = blob + 8;
149
150 /* 1) 签名校验 */
151 int sig_ok = ed25519_verify(signed_region, RKVC_LICENSE_SIGNED_LEN,
152 sig, RKVC_LICENSE_SIG_LEN);
153 if (info) {
154 info->product_id = product_id;
155 bytes_to_hex(license_mid, 32, info->machine_id);
156 info->signature_valid = sig_ok;
157 }
158 if (!sig_ok)
159 return RKVC_ERR_LICENSE;
160
161 /* 2) magic / product */
162 if (magic != RKVC_LICENSE_MAGIC || product_id != RKVC_PRODUCT_RKVC)
163 return RKVC_ERR_LICENSE;
164
165 /* 3) 机器码匹配 */
166 /* 统一比较 lic_hex：info 是否为 NULL 均成立（info!=NULL 时与已填充的
167 info->machine_id 字段等价）。 */
168 char lic_hex[RKVC_MACHINE_ID_HEX_LEN];
169 bytes_to_hex(license_mid, 32, lic_hex);
170
171 char local_hex[RKVC_MACHINE_ID_HEX_LEN] = {0};
172 rkvc_err e = rkvc_machine_id(local_hex, sizeof(local_hex));
173 if (e != RKVC_OK) {
174 if (info)
175 memcpy(info->local_machine_id, local_hex, sizeof(local_hex));
176 return e;
177 }
178 int match = (strcmp(local_hex, lic_hex) == 0);
179 if (info) {
180 memcpy(info->local_machine_id, local_hex, sizeof(local_hex));
181 info->machine_matches = match;
182 }
183 if (!match)
184 return RKVC_ERR_LICENSE;
185
186 /* 4) 通过：授权一经签发永久有效，无有效期校验 */
187 if (info)
188 info->valid = 1;
189 return RKVC_OK;
190}

192rkvc_err rkvc_license_default_path(char *out_path, size_t out_size)
193{
194 if (!out_path || out_size == 0)
195 return RKVC_ERR_INVALID;
196
197 const char *env = getenv("RKVC_LICENSE_FILE");
198 if (env && env[0] != '\0') {
199 if (strlen(env) >= out_size)
200 return RKVC_ERR_INVALID;
201 strcpy(out_path, env);
202 return RKVC_OK;
203 }
204
205 const char *home = getenv("HOME");
206 if (!home || home[0] == '\0')
207 return RKVC_ERR_NOT_FOUND;
208
209 /* 默认路径后缀 "/.config/rkvc/license.lic"，XOR 混淆存储。 */
210 static const uint8_t lic_path_suffix_obf[] = {
211 0x94, 0x92, 0xde, 0xd1, 0xd1, 0xa6, 0xa8, 0xa5,
212 0xec, 0xb6, 0xae, 0xb0, 0xa4, 0xe7, 0xa5, 0xa3,
213 0xa8, 0xa9, 0xa3, 0xbd, 0xaa, 0xfe, 0xbd, 0xbb,
214 0xb0
215 };
216 char suffix[32];
217 lic_deobf_str(lic_path_suffix_obf, sizeof(lic_path_suffix_obf), suffix, 0xBBu);
218
219 int n = snprintf(out_path, out_size, "%s%s", home, suffix);
220 if (n < 0 || (size_t)n >= out_size)
221 return RKVC_ERR_INVALID;
222 return RKVC_OK;
223}

225rkvc_err rkvc_license_verify_file(const char *path, rkvc_license_info *info)
226{
227 if (!path)
228 return RKVC_ERR_INVALID;
229
230 FILE *f = fopen(path, "rb");
231 if (!f)
232 return RKVC_ERR_IO;
233
234 /* 注册码文本最长 ~256 字符即可装下 104 字节 base64 */
235 char text[512];
236 size_t n = fread(text, 1, sizeof(text) - 1, f);
237 fclose(f);
238 text[n] = '\0';
239
240 uint8_t blob[RKVC_LICENSE_BLOB_SIZE];
241 size_t blob_len = 0;
242 if (sodium_base642bin(blob, sizeof(blob), text, n, " \t\r\n", &blob_len,
243 NULL, sodium_base64_VARIANT_ORIGINAL) != 0 ||
244 blob_len != RKVC_LICENSE_BLOB_SIZE)
245 return RKVC_ERR_LICENSE;
246
247 return rkvc_license_verify_blob(blob, blob_len, info);
248}
