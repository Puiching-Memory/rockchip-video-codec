52static inline uint16_t mlvc_px_f32_to_f16(float f)
53{
54 uint32_t x;
55 memcpy(&x, &f, 4);
56 uint32_t sign = (x >> 16) & 0x8000;
57 int exp = (int)((x >> 23) & 0xff) - 127 + 15;
58 uint32_t mant = x & 0x7fffff;
59 if (exp <= 0) {
60 if (exp < -10) return (uint16_t)sign;
61 mant |= 0x800000;
62 int shift = 14 - exp;
63 uint32_t hm = mant >> shift;
64 uint32_t rem = mant & ((1u << shift) - 1);
65 uint32_t half_ = 1u << (shift - 1);
66 if (rem > half_ || (rem == half_ && (hm & 1))) hm++;
67 return (uint16_t)(sign | hm);
68 }
69 if (exp >= 31) return (uint16_t)(sign | 0x7bff);
70 uint32_t h = sign | ((uint32_t)exp << 10) | (mant >> 13);
71 uint32_t rem = mant & 0x1fff;
72 if (rem > 0x1000 || (rem == 0x1000 && (h & 1))) h++;
73 return (uint16_t)h;
74}

76static inline float mlvc_px_f16_to_f32(uint16_t h)
77{
78 uint32_t sign = ((uint32_t)h & 0x8000) << 16;
79 uint32_t exp = (h >> 10) & 0x1f;
80 uint32_t mant = h & 0x3ff;
81 uint32_t x;
82 if (exp == 0) {
83 if (mant == 0) {
84 x = sign;
85 } else {
86 exp = 1;
87 while (!(mant & 0x400)) { mant <<= 1; exp--; }
88 mant &= 0x3ff;
89 x = sign | ((exp + 127 - 15) << 23) | (mant << 13);
90 }
91 } else if (exp == 31) {
92 x = sign | 0x7f800000u | (mant << 13);
93 } else {
94 x = sign | ((exp + 127 - 15) << 23) | (mant << 13);
95 }
96 float f;
97 memcpy(&f, &x, 4);
98 return f;
99}
