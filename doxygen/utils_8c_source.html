20{
21 int should_fail = 0;
22

25 if (s_alloc_fail_after == 0) {

34}

77static int annexb_start_code_at(const uint8_t *data, size_t size,
78 size_t pos, size_t *nal_offset)
79{
80 if (pos + 4 <= size && data[pos] == 0x00 && data[pos + 1] == 0x00 &&
81 data[pos + 2] == 0x01) {
82 *nal_offset = pos + 3;
83 return 1;
84 }
85
86 if (pos + 5 <= size && data[pos] == 0x00 && data[pos + 1] == 0x00 &&
87 data[pos + 2] == 0x00 && data[pos + 3] == 0x01) {
88 *nal_offset = pos + 4;
89 return 1;
90 }
91
92 return 0;
93}

95static int has_annexb_start_code(const uint8_t *data, size_t size,

98 return annexb_start_code_at(data, size, 0, nal_offset);

101static int is_h264_or_h265_annexb(const uint8_t *data, size_t size)
102{
103 size_t nal_offset = 0;
104 int h264_sps = 0;
105 int h264_pps = 0;
106 int h265_vps = 0;
107 int h265_sps = 0;
108 int h265_pps = 0;
109
110 if (!has_annexb_start_code(data, size, &nal_offset) ||
111 nal_offset >= size)
112 return 0;
113
114 for (size_t pos = 0; pos + 4 <= size; pos++) {
115 if (!annexb_start_code_at(data, size, pos, &nal_offset) ||
116 nal_offset >= size)
117 continue;
118
119 uint8_t h264_type = data[nal_offset] & 0x1f;
120 uint8_t h265_type = (data[nal_offset] >> 1) & 0x3f;
121
122 h264_sps |= (h264_type == 7);
123 h264_pps |= (h264_type == 8);
124 h265_vps |= (h265_type == 32);
125 h265_sps |= (h265_type == 33);
126 h265_pps |= (h265_type == 34);
127
128 if ((h264_sps && h264_pps) ||
129 (h265_vps && (h265_sps || h265_pps)) ||
130 (h265_sps && h265_pps))
131 return 1;
132
133 pos = nal_offset;
134 }
135
136 return 0;
137}

139int rkvc_buffer_looks_compressed_video(const uint8_t *data, size_t size)
140{
141 if (!data || size < 4)
142 return 0;
143
144 if (is_h264_or_h265_annexb(data, size))
145 return 1;
146
147 if (size >= 12 && memcmp(data + 4, "ftyp", 4) == 0)
148 return 1;
149
150 if (size >= 4 && data[0] == 0x1a && data[1] == 0x45 &&
151 data[2] == 0xdf && data[3] == 0xa3)
152 return 1;
153
154 if (size > 188 && data[0] == 0x47 && data[188] == 0x47)
155 return 1;
156
157 return 0;
158}

174rkvc_err rkvc_from_averror(int av_err)
175{
176 if (av_err >= 0)
177 return RKVC_OK;
178
179 switch (av_err) {
180 case AVERROR(ENOMEM): return RKVC_ERR_NOMEM;
181 case AVERROR(EINVAL): return RKVC_ERR_INVALID;
182 case AVERROR(ENOENT): return RKVC_ERR_NOT_FOUND;
183 case AVERROR(EIO): return RKVC_ERR_IO;
184 case AVERROR(EACCES): return RKVC_ERR_PERMISSION;
185 case AVERROR(EPERM): return RKVC_ERR_PERMISSION;
186 case AVERROR_EOF: return RKVC_ERR_EOF;
187 case AVERROR(EAGAIN): return RKVC_ERR_AGAIN;
188 case AVERROR(ENODEV): return RKVC_ERR_HW;
189 case AVERROR_UNKNOWN: return RKVC_ERR_INTERNAL;
190 default:
191 /* AVERROR_DECODER_NOT_FOUND / ENOMEM 等 */
192 if (av_err == AVERROR_DECODER_NOT_FOUND ||
193 av_err == AVERROR_ENCODER_NOT_FOUND)
194 return RKVC_ERR_NOT_FOUND;
195 return RKVC_ERR_INTERNAL;
196 }
197}

199/* ── RKMPP 硬件设备上下文单例 ──────────────────────────────────────── */
200
201static AVBufferRef *s_hw_device_ctx = NULL;
202static pthread_mutex_t s_hw_lock = PTHREAD_MUTEX_INITIALIZER;
203static int s_hw_init_ok = 0;
204
205rkvc_err rkvc_get_hw_device_ctx(AVBufferRef **out)
206{
207 if (!out)
208 return RKVC_ERR_INVALID;
209
210 pthread_mutex_lock(&s_hw_lock);
211
212 if (!s_hw_init_ok) {
213 int ret = av_hwdevice_ctx_create(&s_hw_device_ctx,
214 AV_HWDEVICE_TYPE_RKMPP,
215 NULL, NULL, 0);
216 if (ret < 0) {
217 RKVC_LOG("RKMPP hw device create failed: %s", av_err2str(ret));
218 s_hw_device_ctx = NULL;
219 pthread_mutex_unlock(&s_hw_lock);
220 return rkvc_from_averror(ret);
221 }
222 s_hw_init_ok = 1;
223 RKVC_LOG("RKMPP hw device created");
224 }
225
226 *out = av_buffer_ref(s_hw_device_ctx);
227 pthread_mutex_unlock(&s_hw_lock);
228
229 if (!*out)
230 return RKVC_ERR_NOMEM;
231 return RKVC_OK;
232}

273{
274 switch (fmt) {
275 case AV_PIX_FMT_NV12: return RKVC_PIX_FMT_NV12;
276 case AV_PIX_FMT_YUV420P: return RKVC_PIX_FMT_YUV420P;
277 case AV_PIX_FMT_NV16: return RKVC_PIX_FMT_NV16;
278 case AV_PIX_FMT_P010LE:
279 case AV_PIX_FMT_P010BE: return RKVC_PIX_FMT_P010;
280 /* DRM_PRIME 是 dmabuf 容器格式（实际像素格式在 AVDRMFrameDescriptor 中）；
281 本项目仅消费 DRM_FORMAT_NV12 的 dmabuf（rkvc_buffer_from_drm_frame 校验），
282 故显式映射为 NV12，避免落入兜底分支输出误导性告警。 */
283 case AV_PIX_FMT_DRM_PRIME: return RKVC_PIX_FMT_NV12;
284 default:
285 rkvc_log_print(AV_LOG_WARNING,
286 "unknown AVPixelFormat %d, falling back to NV12\n",
287 (int)fmt);
288 return RKVC_PIX_FMT_NV12;
289 }
