13static uint32_t rd_u32le(const uint8_t *p)
14{
15 return (uint32_t)p[0]
16 | ((uint32_t)p[1] << 8)
17 | ((uint32_t)p[2] << 16)
18 | ((uint32_t)p[3] << 24);
19}
20
21static uint64_t rd_u64le(const uint8_t *p)
22{
23 return (uint64_t)rd_u32le(p) | ((uint64_t)rd_u32le(p + 4) << 32);
24}

26static uint32_t crc32_buf(const uint8_t *p, size_t n)
27{
28 uLong c = crc32(0L, Z_NULL, 0);
29 while (n) {
30 uInt chunk = n > 0x40000000u ? 0x40000000u : (uInt)n;
31 c = crc32(c, p, chunk);
32 p += chunk;
33 n -= chunk;
34 }
35 return (uint32_t)c;
36}

38rkvc_err rkvc_qppatch_build_path(char *buf, size_t cap, const char *dir,
39 const char *part, int qp)
40{
41 if (!buf || cap < 8 || !dir || !dir[0] || !part || !part[0] || qp < 0)
42 return RKVC_ERR_INVALID;
43 size_t n = strlen(dir);
44 while (n > 0 && (dir[n - 1] == '/' || dir[n - 1] == '\\'))
45 n--;
46 int wr = snprintf(buf, cap, "%.*s/%s_qp%d.qppatch", (int)n, dir, part, qp);
47 if (wr < 0 || (size_t)wr >= cap)
48 return RKVC_ERR_INVALID;
49 return RKVC_OK;
50}

71rkvc_err rkvc_qppatch_apply(uint8_t *base, size_t base_size,
72 const uint8_t *patch, size_t patch_size,
73 int expected_qp)
74{
75 if (!base || !patch)
76 return RKVC_ERR_INVALID;
77 if (patch_size < RKVC_QPPATCH_HEADER_SIZE)
78 return RKVC_ERR_FORMAT;
79 if (memcmp(patch, RKVC_QPPATCH_MAGIC, 4) != 0)
80 return RKVC_ERR_FORMAT;
81
82 uint32_t version = rd_u32le(patch + 4);
83 if (version != RKVC_QPPATCH_VERSION)
84 return RKVC_ERR_FORMAT;
85
86 uint64_t declared = rd_u64le(patch + 8);
87 if (declared != (uint64_t)base_size)
88 return RKVC_ERR_FORMAT;
89
90 uint32_t qp = rd_u32le(patch + 16);
91 uint32_t num_ranges = rd_u32le(patch + 20);
92 uint32_t flags = rd_u32le(patch + 24);
93 uint32_t base_crc = rd_u32le(patch + 28);
94 uint32_t payload_crc = rd_u32le(patch + 32);
95 (void)rd_u32le(patch + 36); /* coalesce_gap，仅记录 */
96 if (flags != 0)
97 return RKVC_ERR_FORMAT;
98 if (expected_qp >= 0 && qp != (uint32_t)expected_qp) {
99 RKVC_LOG("MLVC qp patch qp=%u, expected %d", qp, expected_qp);
100 return RKVC_ERR_FORMAT;
101 }
102
103 if (crc32_buf(base, base_size) != base_crc) {
104 RKVC_LOG("MLVC qp patch base crc mismatch");
105 return RKVC_ERR_FORMAT;
106 }
107
108 if (num_ranges > (SIZE_MAX - RKVC_QPPATCH_HEADER_SIZE) / RKVC_QPPATCH_RANGE_SIZE)
109 return RKVC_ERR_FORMAT;
110 size_t ranges_bytes = (size_t)num_ranges * RKVC_QPPATCH_RANGE_SIZE;
111 if (patch_size < RKVC_QPPATCH_HEADER_SIZE + ranges_bytes)
112 return RKVC_ERR_FORMAT;
113
114 const uint8_t *ranges = patch + RKVC_QPPATCH_HEADER_SIZE;
115 const uint8_t *payload = ranges + ranges_bytes;
116 size_t payload_size = patch_size - (RKVC_QPPATCH_HEADER_SIZE + ranges_bytes);
117
118 uint32_t i;
119 size_t need = 0;
120 for (i = 0; i < num_ranges; i++) {
121 const uint8_t *r = ranges + (size_t)i * RKVC_QPPATCH_RANGE_SIZE;
122 uint32_t off = rd_u32le(r);
123 uint32_t len = rd_u32le(r + 4);
124 if ((uint64_t)off + (uint64_t)len > (uint64_t)base_size)
125 return RKVC_ERR_FORMAT;
126 if (len > SIZE_MAX - need)
127 return RKVC_ERR_FORMAT;
128 need += len;
129 }
130 if (need != payload_size)
131 return RKVC_ERR_FORMAT;
132 if (crc32_buf(payload, payload_size) != payload_crc)
133 return RKVC_ERR_FORMAT;
134
135 size_t cursor = 0;
136 for (i = 0; i < num_ranges; i++) {
137 const uint8_t *r = ranges + (size_t)i * RKVC_QPPATCH_RANGE_SIZE;
138 uint32_t off = rd_u32le(r);
139 uint32_t len = rd_u32le(r + 4);
140 if (len)
141 memcpy(base + off, payload + cursor, len);
142 cursor += len;
143 }
144 return RKVC_OK;
145}

147rkvc_err rkvc_qppatch_apply_file(uint8_t *base, size_t base_size,
148 const char *patch_path, int expected_qp)
149{
150 if (!patch_path || !patch_path[0])
151 return RKVC_ERR_INVALID;
152 FILE *fp = fopen(patch_path, "rb");
153 if (!fp)
154 return RKVC_ERR_IO;
155 if (fseek(fp, 0, SEEK_END) != 0) {
156 fclose(fp);
157 return RKVC_ERR_IO;
158 }
159 long sz = ftell(fp);
160 if (sz < 0 || (size_t)sz < RKVC_QPPATCH_HEADER_SIZE) {
161 fclose(fp);
162 return RKVC_ERR_FORMAT;
163 }
164 if (fseek(fp, 0, SEEK_SET) != 0) {
165 fclose(fp);
166 return RKVC_ERR_IO;
167 }
168 uint8_t *blob = rkvc_malloc((size_t)sz);
169 if (!blob) {
170 fclose(fp);
171 return RKVC_ERR_NOMEM;
172 }
173 if (fread(blob, 1, (size_t)sz, fp) != (size_t)sz) {
174 fclose(fp);
175 rkvc_free(blob);
176 return RKVC_ERR_IO;
177 }
178 fclose(fp);
179 rkvc_err err = rkvc_qppatch_apply(base, base_size, blob, (size_t)sz, expected_qp);
180 rkvc_free(blob);
181 return err;
182}
