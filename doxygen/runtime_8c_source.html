63rkvc_err rkvc_runtime_try_register(const rkvc_pipeline_desc *desc,
64 int *out_flags)
65{
66 if (!desc || !out_flags)
67 return RKVC_ERR_INVALID;
68
69 int flags = RKVC_RT_FLAG_SESSION;
70 if (template_has_enc(desc->template_id))
71 flags |= RKVC_RT_FLAG_ENC;
72 if (template_has_npu(desc))
73 flags |= RKVC_RT_FLAG_NPU;
74
75 pthread_mutex_lock(&s_rt_lock);
76 if (s_quota.max_sessions > 0 && s_sessions >= s_quota.max_sessions) {
77 pthread_mutex_unlock(&s_rt_lock);
78 return RKVC_ERR_AGAIN;
79 }
80 if ((flags & RKVC_RT_FLAG_ENC) && s_quota.max_enc_sessions > 0 &&
81 s_enc_sessions >= s_quota.max_enc_sessions) {
82 pthread_mutex_unlock(&s_rt_lock);
83 return RKVC_ERR_AGAIN;
84 }
85 if ((flags & RKVC_RT_FLAG_NPU) && s_quota.max_npu_sessions > 0 &&
86 s_npu_sessions >= s_quota.max_npu_sessions) {
87 pthread_mutex_unlock(&s_rt_lock);
88 return RKVC_ERR_AGAIN;
89 }
90
91 s_sessions++;
92 if (flags & RKVC_RT_FLAG_ENC)
93 s_enc_sessions++;
94 if (flags & RKVC_RT_FLAG_NPU)
95 s_npu_sessions++;
96 pthread_mutex_unlock(&s_rt_lock);
97
98 *out_flags = flags;
99 return RKVC_OK;
100}

102void rkvc_runtime_unregister(int flags)
103{
104 if (!(flags & RKVC_RT_FLAG_SESSION))
105 return;
106 pthread_mutex_lock(&s_rt_lock);
107 if (s_sessions > 0)
108 s_sessions--;
109 if ((flags & RKVC_RT_FLAG_ENC) && s_enc_sessions > 0)
110 s_enc_sessions--;
111 if ((flags & RKVC_RT_FLAG_NPU) && s_npu_sessions > 0)
112 s_npu_sessions--;
113 pthread_mutex_unlock(&s_rt_lock);
114}
