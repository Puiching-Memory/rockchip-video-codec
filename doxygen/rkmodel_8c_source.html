41rkvc_status rkvc_rkmodel_open(const char *path, rkvc_rkmodel *out,
42 char *errbuf, size_t errcap) {
43 FILE *f = NULL;
44 rkmodel_fixed fixed;
45 uint8_t *tlv = NULL;
46 size_t table_bytes;
47 uint64_t data_start;
48 off_t file_end;
49 rkvc_status rc = RKVC_STATUS_OK;
50
51 if (!path || !out) {
52 fail(errbuf, errcap, "invalid argument");
53 return RKVC_STATUS_INVALID;
54 }
55 memset(out, 0, sizeof(*out));
56 {
57 size_t plen = strlen(path);
58 if (plen >= sizeof(out->path))
59 plen = sizeof(out->path) - 1;
60 memcpy(out->path, path, plen);
61 out->path[plen] = '\0';
62 }
63
64 f = fopen(path, "rb");
65 if (!f) {
66 fail(errbuf, errcap, "open: %s", strerror(errno));
67 return RKVC_STATUS_IO;
68 }
69 if (fread(&fixed, 1, sizeof(fixed), f) != sizeof(fixed)) {
70 fail(errbuf, errcap, "short fixed header");
71 rc = RKVC_STATUS_INVALID;
72 goto out;
73 }
74 if (fixed.magic != RKMODEL_MAGIC) {
75 fail(errbuf, errcap, "bad magic 0x%08" PRIx32, fixed.magic);
76 rc = RKVC_STATUS_INVALID;
77 goto out;
78 }
79 if (fixed.format_version != RKMODEL_VERSION) {
80 fail(errbuf, errcap, "unsupported version %" PRIu32,
81 fixed.format_version);
82 rc = RKVC_STATUS_INVALID;
83 goto out;
84 }
85 if (fixed.header_len > RKMODEL_MAX_HEADER) {
86 fail(errbuf, errcap, "header_len %" PRIu32 " exceeds bound",
87 fixed.header_len);
88 rc = RKVC_STATUS_INVALID;
89 goto out;
90 }
91 if (fixed.payload_count == 0 ||
92 fixed.payload_count > RKMODEL_MAX_PAYLOADS) {
93 fail(errbuf, errcap, "payload_count %" PRIu32 " outside bounds",
94 fixed.payload_count);
95 rc = RKVC_STATUS_INVALID;
96 goto out;
97 }
98 if (fixed.payload_entry_size != sizeof(rkmodel_payload_entry)) {
99 fail(errbuf, errcap, "unsupported payload entry size %" PRIu32,
100 fixed.payload_entry_size);
101 rc = RKVC_STATUS_INVALID;
102 goto out;
103 }
104 for (size_t i = 0; i < sizeof(fixed.reserved); ++i) {
105 if (fixed.reserved[i] != 0) {
106 fail(errbuf, errcap, "nonzero reserved bytes");
107 rc = RKVC_STATUS_INVALID;
108 goto out;
109 }
110 }
111
112 table_bytes = (size_t)fixed.payload_count * sizeof(rkmodel_payload_entry);
113 data_start = (uint64_t)sizeof(fixed) + fixed.header_len + table_bytes;
114 tlv = rkvc_g_calloc(1, fixed.header_len ? fixed.header_len : 1);
115 if (!tlv) {
116 fail(errbuf, errcap, "out of memory");
117 rc = RKVC_STATUS_NOMEM;
118 goto out;
119 }
120 if (fread(tlv, 1, fixed.header_len, f) != fixed.header_len) {
121 fail(errbuf, errcap, "short TLV region");
122 rc = RKVC_STATUS_INVALID;
123 goto out;
124 }
125
126 {
127 size_t pos = 0;
128 while (pos + 6 <= fixed.header_len) {
129 uint16_t tag = (uint16_t)(tlv[pos] | ((uint16_t)tlv[pos + 1] << 8));
130 uint32_t len = (uint32_t)tlv[pos + 2] |
131 ((uint32_t)tlv[pos + 3] << 8) |
132 ((uint32_t)tlv[pos + 4] << 16) |
133 ((uint32_t)tlv[pos + 5] << 24);
134 const uint8_t *val;
135 pos += 6;
136 if (len > fixed.header_len - pos) {
137 fail(errbuf, errcap, "TLV tag %u overruns header", tag);
138 rc = RKVC_STATUS_INVALID;
139 goto out;
140 }
141 val = tlv + pos;
142 switch (tag) {
143 case RKMODEL_TAG_FAMILY:
144 tlv_str(val, len, out->info.family, sizeof(out->info.family));
145 break;
146 case RKMODEL_TAG_ROLE:
147 tlv_str(val, len, out->info.role, sizeof(out->info.role));
148 break;
149 case RKMODEL_TAG_ID:
150 tlv_str(val, len, out->info.id, sizeof(out->info.id));
151 break;
152 case RKMODEL_TAG_VERSION:
153 tlv_str(val, len, out->info.version, sizeof(out->info.version));
154 break;
155 case RKMODEL_TAG_RKNN_TARGET:
156 tlv_str(val, len, out->info.rknn_target,
157 sizeof(out->info.rknn_target));
158 break;
159 case RKMODEL_TAG_MIN_ABI:
160 if (len != 4) {
161 fail(errbuf, errcap, "invalid min ABI TLV length");
162 rc = RKVC_STATUS_INVALID;
163 goto out;
164 }
165 out->min_runtime_abi = (uint32_t)val[0] |
166 ((uint32_t)val[1] << 8) |
167 ((uint32_t)val[2] << 16) |
168 ((uint32_t)val[3] << 24);
169 break;
170 default:
171 break;
172 }
173 pos += len;
174 }
175 if (pos != fixed.header_len) {
176 fail(errbuf, errcap, "trailing %zu bytes in TLV region",
177 fixed.header_len - pos);
178 rc = RKVC_STATUS_INVALID;
179 goto out;
180 }
181 }
182 if (out->info.id[0] == '\0' || out->info.family[0] == '\0' ||
183 out->info.role[0] == '\0' || out->info.version[0] == '\0') {
184 fail(errbuf, errcap, "missing mandatory TLV (id/family/role/version)");
185 rc = RKVC_STATUS_INVALID;
186 goto out;
187 }
188
189 if (fread(out->payloads, sizeof(rkmodel_payload_entry),
190 fixed.payload_count, f) != fixed.payload_count) {
191 fail(errbuf, errcap, "short payload table");
192 rc = RKVC_STATUS_INVALID;
193 goto out;
194 }
195 if (fseeko(f, 0, SEEK_END) != 0 || (file_end = ftello(f)) < 0) {
196 fail(errbuf, errcap, "cannot determine file size");
197 rc = RKVC_STATUS_IO;
198 goto out;
199 }
200 if ((uint64_t)file_end < data_start) {
201 fail(errbuf, errcap, "payload table exceeds file size");
202 rc = RKVC_STATUS_INVALID;
203 goto out;
204 }
205
206 out->payload_count = fixed.payload_count;
207 for (uint32_t i = 0; i < fixed.payload_count; ++i) {
208 const rkmodel_payload_entry *e = &out->payloads[i];
209 uint32_t kind_bit;
210 if (e->kind == 0 || e->kind > 31) {
211 fail(errbuf, errcap, "invalid payload kind %" PRIu32, e->kind);
212 rc = RKVC_STATUS_INVALID;
213 goto out;
214 }
215 kind_bit = (uint32_t)1 << e->kind;
216 if (out->info.payload_mask & kind_bit) {
217 fail(errbuf, errcap, "duplicate payload kind %" PRIu32, e->kind);
218 rc = RKVC_STATUS_INVALID;
219 goto out;
220 }
221 if (e->flags != 0) {
222 fail(errbuf, errcap, "unsupported payload flags 0x%08" PRIx32,
223 e->flags);
224 rc = RKVC_STATUS_INVALID;
225 goto out;
226 }
227 if (e->offset < data_start || e->offset > (uint64_t)file_end ||
228 e->length > (uint64_t)file_end - e->offset ||
229 e->length > ((uint64_t)1 << 30)) {
230 fail(errbuf, errcap, "payload %" PRIu32 " outside file bounds",
231 e->kind);
232 rc = RKVC_STATUS_INVALID;
233 goto out;
234 }
235 for (uint32_t j = 0; j < i; ++j) {
236 if (payload_ranges_overlap(e, &out->payloads[j])) {
237 fail(errbuf, errcap, "overlapping payload ranges");
238 rc = RKVC_STATUS_INVALID;
239 goto out;
240 }
241 }
242 out->info.payload_mask |= kind_bit;
243 }
244
245out:
246 rkvc_g_free(tlv);
247 fclose(f);
248 return rc;
249}

251rkvc_status rkvc_rkmodel_load_payload(const rkvc_rkmodel *m, uint32_t kind,
252 void **buf, size_t *size) {
253 const rkmodel_payload_entry *e = NULL;
254 FILE *f;
255 uint8_t *data;
256
257 if (!m || !buf || !size || !m->path[0])
258 return RKVC_STATUS_INVALID;
259 *buf = NULL;
260 *size = 0;
261 for (uint32_t i = 0; i < m->payload_count; ++i) {
262 if (m->payloads[i].kind == kind) {
263 e = &m->payloads[i];
264 break;
265 }
266 }
267 if (!e)
268 return RKVC_STATUS_NOT_FOUND;
269
270 f = fopen(m->path, "rb");
271 if (!f)
272 return RKVC_STATUS_IO;
273 if (fseeko(f, (off_t)e->offset, SEEK_SET) != 0) {
274 fclose(f);
275 return RKVC_STATUS_IO;
276 }
277 data = rkvc_g_calloc(1, (size_t)e->length ? (size_t)e->length : 1);
278 if (!data) {
279 fclose(f);
280 return RKVC_STATUS_NOMEM;
281 }
282 if (fread(data, 1, (size_t)e->length, f) != (size_t)e->length) {
283 rkvc_g_free(data);
284 fclose(f);
285 return RKVC_STATUS_IO;
286 }
287 fclose(f);
288 *buf = data;
289 *size = (size_t)e->length;
290 return RKVC_STATUS_OK;
291}
